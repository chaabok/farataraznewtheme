    14→   163→        }
    15→   164→        if (ids.size) ids.add(node.record.id);
    16→   165→      }
    17→   166→      return false;
    18→   167→    };
    19→   168→    walk(nodes);
    20→   169→    return ids;
    21→   170→  }, [nodes, selectedId]);

    40→   189→        code.toLowerCase().includes(needle) ||
    41→   190→        title.toLowerCase().includes(needle)

    43→   192→    },
    44→   193→    [codeField, needle, titleField],

    54→   203→        const childMatch = match(node.children);
    55→   204→        if (childMatch || nodeMatches(node)) {
    56→   205→          keep.add(node.record.id);
    57→   206→          any = true;
    58→   207→        }
    59→   208→      }
    60→   209→      return any;
    61→   210→    };
    62→   211→    match(nodes);
    63→   212→    return nodes.filter((node) => keep.has(node.record.id));
    64→   213→  }, [isSearching, nodeMatches, nodes]);

   450→   599→interface ParentPickerProps {
   451→   600→  field: FormFieldDef;
   452→   601→  nodes: TreeNode[];
   453→   602→  /** Record ids that may not become a parent (self + descendants). */
   454→   603→  excludeIds: ReadonlySet<string>;
   455→   604→  value: string | null;
   456→   605→  onChange: (value: FormFieldValue) => void;
   457→   606→  disabled?: boolean;
   458→   607→}
   459→   608→
   460→   609→/**
   461→   610→ * Self-referencing parent selector of tree forms: lists the form's own
   462→   611→ * records in hierarchy order (indented, code + title) and stores the
   463→   612→ * chosen record's id — the same value the tree hierarchy is built from.
   464→   613→ */
   465→   614→function ParentPicker({
   466→   615→  field,
   467→   616→  nodes,
   468→   617→  excludeIds,
   469→   618→  value,
   470→   619→  onChange,
   471→   620→  disabled,
   472→   621→}: ParentPickerProps) {
   473→   622→  const { locale, t } = useTranslation();
   474→   623→  const listboxId = useId();
   475→   624→  const [open, setOpen] = useState(false);
   476→   625→  const [query, setQuery] = useState('');
   477→   626→  const [activeIndex, setActiveIndex] = useState(0);
   478→   627→  const rootRef = useDismiss<HTMLDivElement>(
   479→   628→    open,
   480→   629→    useCallback(() => setOpen(false), []),
   481→   630→  );
   482→   631→
   483→   632→  const options = useMemo(
   484→   633→    () =>
   485→   634→      nodes
   486→   635→        .filter((node) => !excludeIds.has(node.record.id))
   487→   636→        .map((node) => ({ node, depth: node.depth })),
   488→   637→    [excludeIds, nodes],
   489→   638→  );
   490→   639→
   491→   640→  const needle = query.trim().toLowerCase();
   492→   641→  const filtered = useMemo(() => {
   493→   642→    if (!needle) return options;
   494→   643→    return options.filter(({ node }) => {
   495→   644→      const title = String(node.record.data.title ?? '');
   496→   645→      const code = String(node.record.data.code ?? '');
   497→   646→      return (
   498→   647→        title.toLowerCase().includes(needle) ||
   499→   648→        code.toLowerCase().includes(needle)
   500→   649→      );
   501→   650→    });
   502→   651→  }, [needle, options]);
   503→   652→
   504→   653→  const labelOf = useCallback(
   505→   654→    (node: TreeNode) => {
   506→   655→      const title = String(node.record.data.title ?? '').trim();
   507→   656→      const code = String(node.record.data.code ?? '').trim();
   508→   657→      return code && title ? `${code} — ${title}` : title || code || node.record.id;
   509→   658→    },
   510→   659→    [],
   511→   660→  );
   512→   661→
   513→   662→  const selectedNode = useMemo(
   514→   663→    () => nodes.find((node) => node.record.id === value) ?? null,
   515→   664→    [nodes, value],
   516→   665→  );
   517→   666→
   518→   667→  const openPanel = () => {
   519→   668→    setQuery('');
   520→   669→    setActiveIndex(0);
   521→   670→    setOpen(true);
   522→   671→  };
   523→   672→
   524→   673→  const pick = (node: TreeNode) => {
   525→   674→    onChange(node.record.id);
   526→   675→    setOpen(false);
   527→   676→  };
   528→   677→
   529→   678→  const onKeyDown = (event: React.KeyboardEvent) => {
   530→   679→    if (!open) {
   531→   680→      if (event.key === 'ArrowDown' || event.key === 'Enter') {
   532→   681→        event.preventDefault();
   533→   682→        openPanel();
   534→   683→      }
   535→   684→      return;
   536→   685→    }
   537→   686→    if (event.key === 'ArrowDown') {
   538→   687→      event.preventDefault();
   539→   688→      setActiveIndex((index) => Math.min(filtered.length - 1, index + 1));
   540→   689→    } else if (event.key === 'ArrowUp') {
   541→   690→      event.preventDefault();
   542→   691→      setActiveIndex((index) => Math.max(0, index - 1));
   543→   692→    } else if (event.key === 'Enter') {
   544→   693→      event.preventDefault();
   545→   694→      const target = filtered[activeIndex];
   546→   695→      if (target) pick(target.node);
   547→   696→    } else if (event.key === 'Escape') {
   548→   697→      event.stopPropagation();
   549→   698→      setOpen(false);
   550→   699→    }
   551→   700→  };
   552→   701→
   553→   702→  return (
   554→   703→    <div
   555→   704→      className={cn(
   556→   705→        'flex items-start gap-2',
   557→   706→        field.span === 2 && 'col-span-full',
   558→   707→      )}
   559→   708→      ref={rootRef}
   560→   709→      onKeyDown={onKeyDown}
   561→   710→    >
   562→   711→      {/* Label at the inline start: right side in RTL, left side in LTR. */}
   563→   712→      <label
   564→   713→        htmlFor={field.fieldKey}
   565→   714→        className="w-36 shrink-0 text-[12px] leading-6 text-white/70"
   566→   715→      >
   567→   716→        {fieldLabel(field, locale)}
   568→   717→        {field.isRequired && (
   569→   718→          <span
   570→   719→            title={t('formWindow.requiredHint')}
   571→   720→            aria-hidden="true"
   572→   721→            className="ms-1 text-[13px] leading-none text-red-400"
   573→   722→          >
   574→   723→            *
   575→   724→          </span>
   576→   725→        )}
   577→   726→      </label>
   578→   727→      <div className="relative min-w-0 flex-1">
   579→   728→        <div className="flex items-center gap-1.5">
   580→   729→          <button
   581→   730→            type="button"
   582→   731→            id={field.fieldKey}
   583→   732→            role="combobox"
   584→   733→            aria-expanded={open}
   585→   734→            aria-controls={listboxId}
   586→   735→            aria-haspopup="listbox"
   587→   736→            disabled={disabled}
   588→   737→            onClick={() => (open ? setOpen(false) : openPanel())}
   589→   738→            className={cn(
   590→   739→              controlClasses,
   591→   740→              'flex items-center justify-between gap-2 pe-2 text-start',
   592→   741→            )}
   593→   742→          >
   594→   743→            <span
   595→   744→              className={cn(
   596→   745→                'min-w-0 flex-1 truncate',
   597→   746→                selectedNode ? 'text-white' : 'text-white/30',
   598→   747→              )}
   599→   748→            >
   600→   749→              {selectedNode
   601→   750→                ? labelOf(selectedNode)
   602→   751→                : t('formWindow.selectPlaceholder')}
   603→   752→            </span>
   604→   753→            <ChevronsUpDown className="h-3.5 w-3.5 shrink-0 text-white/35" />
   605→   754→          </button>
   606→   755→        </div>
   607→   756→
   608→   757→      <AnimatePresence>
   609→   758→        {open && (
   610→   759→          <motion.div
   611→   760→            initial={{ opacity: 0, scale: 0.97, y: -4 }}
   612→   761→            animate={{ opacity: 1, scale: 1, y: 0 }}
   613→   762→            exit={{ opacity: 0, scale: 0.97, y: -4 }}
   614→   763→            transition={{ duration: 0.14, ease: 'easeOut' }}
   615→   764→            className="absolute top-full z-50 mt-1.5 w-full min-w-[220px] overflow-hidden rounded-xl bg-[#1b1e26]/[0.97] shadow-2xl shadow-black/50 ring-1 ring-white/10 backdrop-blur-xl"
   616→   765→          >
   617→   766→            <div className="flex items-center gap-2 border-b border-white/10 px-3 py-2">
   618→   767→              <Search className="h-3.5 w-3.5 shrink-0 text-white/35" />
   619→   768→              <input
   620→   769→                autoFocus
   621→   770→                type="text"
   622→   771→                value={query}
   623→   772→                onChange={(event) => {
   624→   773→                  setQuery(event.target.value);
   625→   774→                  setActiveIndex(0);
   626→   775→                }}
   627→   776→                placeholder={t('formWindow.search')}
   628→   777→                className="h-7 w-full bg-transparent text-sm text-white placeholder:text-white/30 focus:outline-none"
   629→   778→              />
   630→   779→            </div>
   631→   780→            <div
   632→   781→              id={listboxId}
   633→   782→              role="listbox"
   634→   783→              aria-label={fieldLabel(field, locale)}
   635→   784→              className="max-h-64 overflow-y-auto py-1 [scrollbar-width:thin]"
   636→   785→            >
   637→   786→              {filtered.length === 0 && (
   638→   787→                <p className="px-3 py-4 text-center text-xs text-white/40">
   639→   788→                  {t('formWindow.noMatch')}
   640→   789→                </p>
   641→   790→              )}
   642→   791→              {filtered.map(({ node }, index) => {
   643→   792→                const isSelected = node.record.id === value;
   644→   793→                const isActive = index === activeIndex;
   645→   794→                return (
   646→   795→                  <button
   647→   796→                    key={node.record.id}
   648→   797→                    type="button"
   649→   798→                    role="option"
   650→   799→                    aria-selected={isSelected}
   651→   800→                    onMouseEnter={() => setActiveIndex(index)}
   652→   801→                    onClick={() => pick(node)}
   653→   802→                    style={{ paddingInlineStart: 12 + node.depth * 14 }}
   654→   803→                    className={cn(
   655→   804→                      'flex w-full items-center gap-2 pe-3 py-1.5 text-start text-[12px] transition-colors',
   656→   805→                      isActive
   657→   806→                        ? 'bg-sky-500/15 text-white'
   658→   807→                        : 'text-white/80 hover:bg-white/[0.05]',
   659→   808→                      isSelected && !isActive && 'bg-white/[0.04]',
   660→   809→                    )}
   661→   810→                  >
   662→   811→                    <span className="min-w-0 flex-1 truncate">
   663→   812→                      {labelOf(node)}
   664→   813→                    </span>
   665→   814→                    {isSelected && (
   666→   815→                      <Check className="h-3.5 w-3.5 shrink-0 text-sky-400" />
   667→   816→                    )}
   668→   817→                  </button>
   669→   818→                );
   670→   819→              })}
   671→   820→            </div>
   672→   821→            {value !== null && value !== '' && (
   673→   822→              <div className="border-t border-white/10 p-1.5">
   674→   823→                <button
   675→   824→                  type="button"
   676→   825→                  onClick={() => {
   677→   826→                    onChange(null);
   678→   827→                    setOpen(false);
   679→   828→                  }}
   680→   829→                  className="flex w-full items-center gap-2 rounded-lg px-2.5 py-1.5 text-xs text-rose-300 transition-colors hover:bg-rose-500/15 focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-rose-400/50"
   681→   830→                >
   682→   831→                  <X className="h-3.5 w-3.5" />
   683→   832→                  {t('formWindow.clear')}
   684→   833→                </button>
   685→   834→              </div>
   686→   835→            )}
   687→   836→          </motion.div>
   688→   837→        )}
   689→   838→        </AnimatePresence>
   690→   839→      </div>
   691→   840→    </div>
   692→   841→  );
   693→   842→}
