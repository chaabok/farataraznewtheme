   365→   365→/**
   366→   366→ * SimpleFormTwoRowTabbed (Task 24) — the legacy «فرم سادهٔ دو ردیفه» shell:
   367→   367→ * gradient form header (title + Task-28 window caption: minimize / maximize /
   368→   368→ * close — the strip doubles as the drag handle of a chromeless host window),
   369→   369→ * the Task-20 legacy action bar (plus the Task-28 redesigned help tile) and
   370→   370→ * record navigation, a two-row field grid (with Task-25 automatic label
   371→   371→ * alignment via useAlignedLabels — default scope "container": straight
   372→   372→ * vertical input columns; opt-in labelScope="row" for per-row alignment),
   373→   373→ * and OPTIONAL tabbed detail grids with an empty state and a footer
   374→   374→ * row-actions strip (+ / × / copy / save). All action-bar handlers are
   375→   375→ * pass-throughs — the caller owns persistence and toasts.
   376→   376→ */
   377→   377→export function SimpleFormTwoRowTabbed({
   378→   378→  title,
   379→   379→  badge,
   380→   380→  icon,
   381→   381→  fields,
   382→   382→  onFieldChange,
   383→   383→  labelScope = "container",
   384→   384→  tabs,
   385→   385→  defaultTabId,
   386→   386→  onTabRowsChange,
   387→   387→  onSaveRows,
   388→   388→  onFirst,
   389→   389→  onPrev,
   390→   390→  onNext,
   391→   391→  onLast,
   392→   392→  onReset,
   393→   393→  onSave,
   394→   394→  onSaveNew,
   395→   395→  onSaveClose,
   396→   396→  onCopy,
   397→   397→  onDelete,
   398→   398→  onDeleteNew,
   399→   399→  onDeleteClose,
   400→   400→  onPrint,
   401→   401→  onClose,
   402→   402→  onHelp,
   403→   403→  listItems,
   404→   404→  listLoading,
   405→   405→  onOpenListItem,
   406→   406→  canSave = true,
   407→   407→  canCopy = true,
   408→   408→  canDelete = true,
   409→   409→  canPrint = true,
   410→   410→  onHeaderClose,
   411→   411→  onHeaderMaximize,
   412→   412→  onHeaderMinimize,
   413→   413→  headerMaximized,
   414→   414→  onHeaderDragStart,
   415→   415→  onHeaderDragMove,
   416→   416→  onHeaderDragEnd,
   417→   417→  onHeaderDoubleClick,
   418→   418→  className,
   419→   419→}: SimpleFormTwoRowTabbedProps) {
   420→   420→  const t = useT();
   421→   421→  const tabsProp = React.useMemo(() => tabs ?? [], [tabs]);
   422→   422→
   423→   423→  /* Task 25.1 — automatic label alignment for the field grid. Default scope
   424→   424→     "container": one shared label width ⇒ every input starts on the same
   425→   425→     edge within its cell AND the edges stack into straight vertical columns
   426→   426→     across rows (the owner's expectation). Pass labelScope="row" for
   427→   427→     per-row alignment. */
   428→   428→  const fieldsGridRef = React.useRef<HTMLDivElement | null>(null);
   429→   429→  useAlignedLabels(fieldsGridRef, { scope: labelScope });
   430→   430→
   431→   431→  /* detail-grid rows: initialized once per mount from the tabs prop */
   432→   432→  const [rowsByTab, setRowsByTab] = React.useState<Record<string, Record<string, string>[]>>(
   433→   433→    () =>
   434→   434→      Object.fromEntries(
   435→   435→        tabsProp.map((tb) => [tb.id, (tb.rows ?? []).map((r) => ({ ...r }))]),
   436→   436→      ),
   437→   437→  );
   438→   438→  const [activeTab, setActiveTab] = React.useState(
   439→   439→    () => defaultTabId ?? tabsProp[0]?.id ?? "",
   440→   440→  );
   441→   441→  const [selected, setSelected] = React.useState<{ tabId: string; index: number } | null>(null);
   442→   442→
   443→   443→  const hasTabs = tabsProp.length > 0;
   444→   444→  const hasNav = Boolean(onFirst || onPrev || onNext || onLast);
   445→   445→
   446→   446→  const commitRows = (tabId: string, rows: Record<string, string>[]) => {
   447→   447→    setRowsByTab((prev) => ({ ...prev, [tabId]: rows }));
   448→   448→    onTabRowsChange?.(tabId, rows);
   449→   449→  };
   450→   450→
   451→   451→  const addRow = () => {
   452→   452→    const tb = tabsProp.find((x) => x.id === activeTab);
   453→   453→    if (!tb) return;
   454→   454→    const rows = rowsByTab[activeTab] ?? [];
   455→   455→    const blank = Object.fromEntries(tb.columns.map((c) => [c.id, ""])) as Record<string, string>;
   456→   456→    const next = [...rows, blank];
   457→   457→    commitRows(activeTab, next);
   458→   458→    setSelected({ tabId: activeTab, index: next.length - 1 });
   459→   459→  };
   460→   460→
   461→   461→  const removeSelectedRow = () => {
   462→   462→    if (!selected || selected.tabId !== activeTab) return;
   463→   463→    const rows = [...(rowsByTab[activeTab] ?? [])];
   464→   464→    rows.splice(selected.index, 1);
   465→   465→    commitRows(activeTab, rows);
   466→   466→    setSelected(null);
   467→   467→  };
   468→   468→
   469→   469→  const copySelectedRow = () => {
   470→   470→    if (!selected || selected.tabId !== activeTab) return;
   471→   471→    const rows = [...(rowsByTab[activeTab] ?? [])];
   472→   472→    const clone = { ...rows[selected.index] };
   473→   473→    rows.splice(selected.index + 1, 0, clone);
   474→   474→    commitRows(activeTab, rows);
   475→   475→    setSelected({ tabId: activeTab, index: selected.index + 1 });
   476→   476→  };
   477→   477→
   478→   478→  const setCell = (tabId: string, rowIndex: number, colId: string, value: string) => {
   479→   479→    const rows = [...(rowsByTab[tabId] ?? [])];
   480→   480→    if (!rows[rowIndex]) return;
   481→   481→    rows[rowIndex] = { ...rows[rowIndex], [colId]: value };
   482→   482→    commitRows(tabId, rows);
   483→   483→  };
   484→   484→
   485→   485→  const selectedInActive = Boolean(selected && selected.tabId === activeTab);
   486→   486→
