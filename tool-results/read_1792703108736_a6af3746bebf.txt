    56→    56→export interface DataFormWindowProps {
    57→    57→  /** Window id — `form:<formKey>` (moduleKey defaults to `settings`). */
    58→    58→  moduleId: string;
    59→    59→  /** Descriptor attached by `openFormWindow` in the desktop store. */
    60→    60→  form?: FormWindowInfo;
    61→    61→  /** Closes the hosting window ("save & close" / "delete & close"). */
    62→    62→  onClose?: () => void;
    63→    63→}
    64→    64→
    65→    65→/** Which combined action a pending delete confirmation belongs to. */
    66→    66→type DeleteIntent = 'current' | 'new' | 'close' | null;
    67→    67→
    68→    68→/** Outcome of the shared save pipeline. */
    69→    69→type SaveResult = 'saved' | 'clean' | 'invalid' | 'error';
    70→    70→
    71→    71→/** Fallback parsing of the window id when no descriptor is attached. */
    72→    72→function parseModuleId(moduleId: string): {
    73→    73→  moduleKey: string | null;
    74→    74→  formKey: string | null;
    75→    75→} {
    76→    76→  if (!moduleId.startsWith('form:')) {
    77→    77→    return { moduleKey: null, formKey: null };
    78→    78→  }
    79→    79→  const formKey = moduleId.slice('form:'.length);
    80→    80→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    81→    81→}
    82→    82→
    83→    83→/** Sentinel kept in state so load errors render through the dictionaries. */
    84→    84→const GENERIC_LOAD_ERROR = 'formWindow.loadError';
    85→    85→
    86→    86→/**
    87→    87→ * Generic, database-driven form window: loads a FormDef + its records and
    88→    88→ * drives the whole view/new/edit state machine (dirty tracking, CRUD,
    89→    89→ * record navigation, Ctrl+S, records drawer) through the shared form
    90→    90→ * component library. One implementation renders every ERP form.
    91→    91→ */
    92→    92→export function DataFormWindow({ moduleId, form, onClose }: DataFormWindowProps) {
    93→    93→  const { locale, direction, t, tf, formatNumber } = useTranslation();
    94→    94→
    95→    95→  const fallback = useMemo(() => parseModuleId(moduleId), [moduleId]);
    96→    96→  const moduleKey = form?.moduleKey ?? fallback.moduleKey;
    97→    97→  const formKey = form?.formKey ?? fallback.formKey;
    98→    98→  const basePath =
    99→    99→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
   100→   100→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   101→   101→
   102→   102→  /** Record preselection / start-mode requested by a data list (if any). */
   103→   103→  const wantedRecordId = form?.recordId ?? null;
   104→   104→  const wantsStartInNew = form?.startInNew ?? false;
   105→   105→  /** Changes on every list-driven open → forces a fresh load. */
   106→   106→  const formNonce = form?.nonce;
   107→   107→
   108→   108→  // ── Data ──
   109→   109→  const [def, setDef] = useState<FormDef | null>(null);
   110→   110→  const [records, setRecords] = useState<FormRecord[]>([]);
   111→   111→  const [isLoading, setIsLoading] = useState(true);
   112→   112→  const [loadError, setLoadError] = useState<string | null>(null);
   113→   113→
   114→   114→  // ── Record state machine ──
   115→   115→  const [mode, setMode] = useState<FormMode>('view');
   116→   116→  const [currentIndex, setCurrentIndex] = useState(-1);
   117→   117→  const [draft, setDraft] = useState<FormDataMap>({});
   118→   118→  const [baseline, setBaseline] = useState<FormDataMap>({});
   119→   119→  const [isSaving, setIsSaving] = useState(false);
   120→   120→  const [isDeleting, setIsDeleting] = useState(false);
   121→   121→  /**
   122→   122→   * Pending delete request awaiting the confirm dialog — `current` =
   123→   123→   * delete and stay, `new` = delete then reset for a fresh entry,
   124→   124→   * `close` = delete then close the window.
   125→   125→   */
   126→   126→  const [deleteIntent, setDeleteIntent] = useState<DeleteIntent>(null);
   127→   127→  /** Record the confirmed delete targets (defaults to the loaded one). */
   128→   128→  const [deleteTargetId, setDeleteTargetId] = useState<string | null>(null);
   129→   129→  const [isListOpen, setIsListOpen] = useState(false);
   130→   130→  const [activeTabKey, setActiveTabKey] = useState<string | null>(null);
   131→   131→
   132→   132→  const current =
   133→   133→    currentIndex >= 0 && currentIndex < records.length
   134→   134→      ? records[currentIndex]
   135→   135→      : null;
   136→   136→
   137→   137→  const isDirty = useMemo(() => {
   138→   138→    const keys = new Set([...Object.keys(baseline), ...Object.keys(draft)]);
   139→   139→    for (const key of keys) {
   140→   140→      if (isChangedValue(baseline[key], draft[key])) return true;
   141→   141→    }
   142→   142→    return false;
   143→   143→  }, [baseline, draft]);
   144→   144→
   145→   145→  const status: FormStatus =
   146→   146→    mode === 'new' ? (isDirty ? 'unsaved' : 'new') : isDirty ? 'unsaved' : 'saved';
   147→   147→
   148→   148→  // ── Loading ──
   149→   149→  const load = useCallback(async () => {
   150→   150→    if (!basePath) {
   151→   151→      setLoadError(GENERIC_LOAD_ERROR);
   152→   152→      setIsLoading(false);
   153→   153→      return;
   154→   154→    }
   155→   155→    setIsLoading(true);
   156→   156→    setLoadError(null);
   157→   157→    try {
   158→   158→      const [defRes, recordsRes] = await Promise.all([
   159→   159→        apiFetch<{ data: FormDef }>(
   160→   160→          `${basePath}?locale=${encodeURIComponent(locale)}`,
   161→   161→        ),
   162→   162→        apiFetch<{ data: FormRecord[] }>(`${basePath}/records`),
   163→   163→      ]);
   164→   164→      const nextDef = defRes.data;
   165→   165→      const nextRecords = recordsRes.data ?? [];
   166→   166→      setDef(nextDef);
   167→   167→      setRecords(nextRecords);
   168→   168→
   169→   169→      if (wantsStartInNew) {
   170→   170→        // Opened from a data list in "new record" mode.
   171→   171→        const defaults = buildDefaults(nextDef.fields);
   172→   172→        setCurrentIndex(-1);
   173→   173→        setDraft({ ...defaults });
   174→   174→        setBaseline({ ...defaults });
   175→   175→        setMode('new');
   176→   176→      } else if (nextRecords.length) {
   177→   177→        // Default: the last record. When opened from a data list, the
   178→   178→        // requested record (wantedRecordId) is selected instead.
   179→   179→        let index = nextRecords.length - 1;
   180→   180→        if (wantedRecordId) {
   181→   181→          const found = nextRecords.findIndex(
   182→   182→            (record) => record.id === wantedRecordId,
   183→   183→          );
   184→   184→          if (found >= 0) index = found;
   185→   185→        }
   186→   186→        const target = nextRecords[index];
   187→   187→        setCurrentIndex(index);
   188→   188→        setDraft({ ...target.data });
   189→   189→        setBaseline({ ...target.data });
   190→   190→        setMode('view');
   191→   191→      } else {
   192→   192→        const defaults = buildDefaults(nextDef.fields);
   193→   193→        setCurrentIndex(-1);
   194→   194→        setDraft({ ...defaults });
   195→   195→        setBaseline({ ...defaults });
   196→   196→        setMode('new');
   197→   197→      }
   198→   198→
   199→   199→      const derived = deriveTabs(nextDef, locale);
   200→   200→      // The first group is the always-visible header section; the active
   201→   201→      // tab state only drives the remaining body tabs (default: first one).
   202→   202→      const firstBodyTab = derived[1] ?? null;
   203→   203→      setActiveTabKey(firstBodyTab ? firstBodyTab.key : null);
   204→   204→    } catch (error) {
   205→   205→      if (error instanceof SessionExpiredError) return;
   206→   206→      setLoadError(
   207→   207→        error instanceof Error && error.message
   208→   208→          ? error.message
   209→   209→          : GENERIC_LOAD_ERROR,
   210→   210→      );
   211→   211→    } finally {
   212→   212→      setIsLoading(false);
   213→   213→    }
   214→   214→  }, [basePath, formNonce, locale, wantedRecordId, wantsStartInNew]);
   215→   215→
   216→   216→  useEffect(() => {
   217→   217→    void load();
   218→   218→  }, [load]);
   219→   219→
   220→   220→  const refreshRecords = useCallback(async (): Promise<FormRecord[]> => {
   221→   221→    if (!basePath) return [];
   222→   222→    const res = await apiFetch<{ data: FormRecord[] }>(`${basePath}/records`);
   223→   223→    const list = res.data ?? [];
   224→   224→    setRecords(list);
   225→   225→    return list;
   226→   226→  }, [basePath]);
   227→   227→
   228→   228→  // ── Record transitions ──
   229→   229→  const applyRecord = useCallback((index: number, list: FormRecord[]) => {
   230→   230→    const target = list[index];
   231→   231→    setCurrentIndex(index);
   232→   232→    setMode('view');
   233→   233→    if (target) {
   234→   234→      setDraft({ ...target.data });
   235→   235→      setBaseline({ ...target.data });
   236→   236→    }
   237→   237→  }, []);
   238→   238→
   239→   239→  const navigateTo = useCallback(
   240→   240→    (index: number) => {
   241→   241→      if (isSaving || isDeleting) return;
   242→   242→      if (index < 0 || index >= records.length) return;
   243→   243→      if (mode !== 'view' && isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) {
   244→   244→        return;
   245→   245→      }
   246→   246→      applyRecord(index, records);
   247→   247→    },
   248→   248→    [applyRecord, isDeleting, isDirty, isSaving, mode, records, t],
   249→   249→  );
   250→   250→
   251→   251→  const startNew = useCallback(
   252→   252→    (preset?: FormDataMap) => {
   253→   253→      if (!def) return;
   254→   254→      const base = { ...buildDefaults(def.fields), ...(preset ?? {}) };
   255→   255→      setDraft(base);
   256→   256→      setBaseline({ ...base });
   257→   257→      setMode('new');
   258→   258→      setCurrentIndex(-1);
   259→   259→    },
   260→   260→    [def],
   261→   261→  );
   262→   262→
   263→   263→  const handleNew = useCallback(() => {
   264→   264→    if (mode === 'new') return;
   265→   265→    if (isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) return;
   266→   266→    startNew();
   267→   267→  }, [isDirty, mode, startNew, t]);
   268→   268→
   269→   269→  const handleCopy = useCallback(() => {
   270→   270→    if (mode !== 'view' || !current) return;
   271→   271→    startNew({ ...current.data });
   272→   272→    toast({ title: t('formWindow.copyToast') });
   273→   273→  }, [current, mode, startNew, t]);
   274→   274→
   275→   275→  /**
   276→   276→   * Shared save pipeline for every save action: validates required
   277→   277→   * fields, writes the draft to the database and refreshes the record
   278→   278→   * list. `keepRecord: false` skips re-selecting the saved record
   279→   279→   * (used by "save & new", which resets the form right after).
   280→   280→   */
   281→   281→  const performSave = useCallback(
   282→   282→    async (options?: { keepRecord?: boolean }): Promise<SaveResult> => {
   283→   283→      if (!def || !basePath || isSaving || isDeleting) return 'error';
   284→   284→      if (mode === 'view' || !isDirty) return 'clean';
   285→   285→
   286→   286→      const missingField = def.fields.find((field) =>
   287→   287→        isEmptyRequiredValue(field, draft[field.fieldKey]),
   288→   288→      );
   289→   289→      if (missingField) {
   290→   290→        toast({
   291→   291→          title: fieldLabel(missingField, locale),
   292→   292→          description: t('formWindow.requiredHint'),
   293→   293→          variant: 'destructive',
   294→   294→        });
   295→   295→        return 'invalid';
   296→   296→      }
   297→   297→
   298→   298→      setIsSaving(true);
   299→   299→      try {
   300→   300→        const saved =
   301→   301→          mode === 'new'
   302→   302→            ? (
   303→   303→                await apiFetch<{ data: FormRecord }>(`${basePath}/records`, {
   304→   304→                  method: 'POST',
   305→   305→                  body: { data: draft },
   306→   306→                })
   307→   307→              ).data
   308→   308→            : (
   309→   309→                await apiFetch<{ data: FormRecord }>(
   310→   310→                  `${basePath}/records/${encodeURIComponent(current?.id ?? '')}`,
   311→   311→                  { method: 'PUT', body: { data: draft } },
   312→   312→                )
   313→   313→              ).data;
   314→   314→
   315→   315→        const list = await refreshRecords();
   316→   316→        if (options?.keepRecord !== false) {
   317→   317→          const index = list.findIndex((record) => record.id === saved.id);
   318→   318→          if (index >= 0) {
   319→   319→            applyRecord(index, list);
   320→   320→          } else {
   321→   321→            setMode('view');
   322→   322→          }
   323→   323→        }
   324→   324→        // Open records lists of this form refresh themselves.
   325→   325→        if (moduleKey && formKey) {
   326→   326→          notifyRecordsChanged({ moduleKey, formKey });
   327→   327→        }
   328→   328→        return 'saved';
   329→   329→      } catch (error) {
   330→   330→        if (!(error instanceof SessionExpiredError)) {
   331→   331→          toast({
   332→   332→            title: t('formWindow.saveError'),
   333→   333→            description: error instanceof Error ? error.message : undefined,
   334→   334→            variant: 'destructive',
   335→   335→          });
   336→   336→        }
   337→   337→        return 'error';
   338→   338→      } finally {
   339→   339→        setIsSaving(false);
   340→   340→      }
   341→   341→    },
   342→   342→    [
   343→   343→      applyRecord,
   344→   344→      basePath,
   345→   345→      current,
   346→   346→      def,
   347→   347→      draft,
   348→   348→      formKey,
   349→   349→      isDeleting,
   350→   350→      isDirty,
   351→   351→      isSaving,
   352→   352→      locale,
   353→   353→      mode,
   354→   354→      moduleKey,
   355→   355→      refreshRecords,
   356→   356→      t,
   357→   357→    ],
   358→   358→  );
   359→   359→
   360→   360→  const handleSave = useCallback(async () => {
   361→   361→    const result = await performSave();
   362→   362→    if (result === 'saved') toast({ title: t('formWindow.saveSuccess') });
   363→   363→  }, [performSave, t]);
   364→   364→
   365→   365→  /** Save & new: write the record, then reset the form for a fresh entry. */
   366→   366→  const handleSaveNew = useCallback(async () => {
   367→   367→    if (isSaving || isDeleting) return;
   368→   368→    const result = await performSave({ keepRecord: false });
   369→   369→    if (result === 'error' || result === 'invalid') return;
   370→   370→    startNew();
   371→   371→    if (result === 'saved') toast({ title: t('formWindow.saveNewSuccess') });
   372→   372→  }, [isDeleting, isSaving, performSave, startNew, t]);
   373→   373→
   374→   374→  /** Save & close: write the record after validation, then close. */
   375→   375→  const handleSaveClose = useCallback(async () => {
   376→   376→    if (isSaving || isDeleting) return;
   377→   377→    const result = await performSave();
   378→   378→    if (result === 'error' || result === 'invalid') return;
   379→   379→    if (result === 'saved') toast({ title: t('formWindow.saveCloseSuccess') });
   380→   380→    onClose?.();
   381→   381→  }, [isDeleting, isSaving, onClose, performSave, t]);
   382→   382→
   383→   383→  /** Resets the shared delete-confirmation dialog. */
   384→   384→  const cancelDelete = useCallback(() => {
   385→   385→    setDeleteIntent(null);
   386→   386→    setDeleteTargetId(null);
   387→   387→  }, []);
   388→   388→
   389→   389→  /**
   390→   390→   * First click on any delete action opens the confirm dialog. The
   391→   391→   * target defaults to the loaded record; the tree context menu passes
   392→   392→   * the right-clicked branch id explicitly.
   393→   393→   */
   394→   394→  const requestDelete = useCallback(
   395→   395→    (intent: Exclude<DeleteIntent, null>, targetId?: string) => {
   396→   396→      if (mode === 'new' || isSaving || isDeleting) return;
   397→   397→      const target = targetId
   398→   398→        ? records.find((record) => record.id === targetId) ?? null
   399→   399→        : current;
   400→   400→      if (!target) return;
   401→   401→      setDeleteTargetId(target.id);
   402→   402→      setDeleteIntent(intent);
   403→   403→    },
   404→   404→    [current, isDeleting, isSaving, mode, records],
   405→   405→  );
   406→   406→
   407→   407→  /** Runs after the user presses «بلی» in the confirm dialog. */
   408→   408→  const confirmDelete = useCallback(async () => {
   409→   409→    if (isDeleting || !deleteIntent) return;
   410→   410→    const intent = deleteIntent;
   411→   411→    const targetId = deleteTargetId ?? current?.id ?? '';
   412→   412→    if (!basePath || !targetId) {
   413→   413→      cancelDelete();
   414→   414→      return;
   415→   415→    }
   416→   416→    setIsDeleting(true);
   417→   417→    try {
   418→   418→      // DELETE answers 204 with an empty body — nothing to parse.
   419→   419→      await apiFetch(
   420→   420→        `${basePath}/records/${encodeURIComponent(targetId)}`,
   421→   421→        {
   422→   422→          method: 'DELETE',
   423→   423→        },
   424→   424→      );
   425→   425→      cancelDelete();
   426→   426→      // Open records lists of this form refresh themselves.
   427→   427→      if (moduleKey && formKey) {
   428→   428→        notifyRecordsChanged({ moduleKey, formKey });
   429→   429→      }
   430→   430→      if (intent === 'close') {
   431→   431→        toast({ title: t('formWindow.deleteCloseSuccess') });
   432→   432→        onClose?.();
   433→   433→        return;
   434→   434→      }
   435→   435→      const list = await refreshRecords();
   436→   436→      if (intent === 'new') {
   437→   437→        startNew();
   438→   438→        toast({ title: t('formWindow.deleteNewSuccess') });
   439→   439→      } else if (list.length) {
   440→   440→        applyRecord(Math.min(currentIndex, list.length - 1), list);
   441→   441→        toast({ title: t('formWindow.deleteSuccess') });
   442→   442→      } else {
   443→   443→        setCurrentIndex(-1);
   444→   444→        startNew();
   445→   445→        toast({ title: t('formWindow.deleteSuccess') });
   446→   446→      }
   447→   447→    } catch (error) {
   448→   448→      cancelDelete();
   449→   449→      if (error instanceof SessionExpiredError) return;
   450→   450→      toast({
   451→   451→        title: error instanceof Error && error.message
   452→   452→          ? error.message
   453→   453→          : t('formWindow.loadError'),
   454→   454→        variant: 'destructive',
   455→   455→      });
   456→   456→    } finally {
   457→   457→      setIsDeleting(false);
   458→   458→    }
   459→   459→  }, [
   460→   460→    applyRecord,
   461→   461→    basePath,
   462→   462→    cancelDelete,
   463→   463→    currentIndex,
   464→   464→    current,
   465→   465→    deleteIntent,
   466→   466→    deleteTargetId,
   467→   467→    isDeleting,
   468→   468→    onClose,
   469→   469→    refreshRecords,
   470→   470→    startNew,
   471→   471→    t,
   472→   472→  ]);
   473→   473→
   474→   474→  const handlePrint = useCallback(() => {
   475→   475→    toast({ title: t('formWindow.printToast') });
   476→   476→  }, [t]);
   477→   477→
   478→   478→  const handleFieldChange = useCallback(
   479→   479→    (fieldKey: string, value: FormFieldValue) => {
   480→   480→      setDraft((prev) => ({ ...prev, [fieldKey]: value }));
   481→   481→      setMode((prev) => (prev === 'view' ? 'edit' : prev));
   482→   482→    },
   483→   483→    [],
   484→   484→  );
   485→   485→
   486→   486→  // Ctrl+S / Cmd+S saves the active draft (Escape is handled by the dialog).
   487→   487→  useEffect(() => {
   488→   488→    const onKeyDown = (event: KeyboardEvent) => {
   489→   489→      if (
   490→   490→        (event.ctrlKey || event.metaKey) &&
   491→   491→        !event.shiftKey &&
   492→   492→        !event.altKey &&
