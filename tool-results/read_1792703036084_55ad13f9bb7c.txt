   150→    return flat;
   151→  }, [records, parentField]);
   152→
   153→  /** Ids of the selected record's descendants (excluded from parent picker). */
   154→  const descendantIds = useMemo(() => {
   155→    const ids = new Set<string>();
   156→    if (!selectedId) return ids;
   157→    const walk = (list: TreeNode[]): boolean => {
   158→      for (const node of list) {
   159→        if (walk(node.children)) return true;
   160→        if (node.record.id === selectedId) {
   161→          ids.add(node.record.id);
   162→          return true;
   163→        }
   164→        if (ids.size) ids.add(node.record.id);
   165→      }
   166→      return false;
   167→    };
   168→    walk(nodes);
   169→    return ids;
   170→  }, [nodes, selectedId]);
   171→
   172→  // ── Search + expand state ───────────────────────────────────────
   173→  const [query, setQuery] = useState('');
   174→  const [collapsed, setCollapsed] = useState<ReadonlySet<string>>(new Set());
   175→
   176→  const needle = query.trim().toLowerCase();
   177→  const isSearching = needle.length > 0;
   178→
   179→  /** Node matches when its code or title contains the query. */
   180→  const nodeMatches = useCallback(
   181→    (node: TreeNode) => {
   182→      const code = codeField
   183→        ? String(node.record.data[codeField.fieldKey] ?? '')
   184→        : '';
   185→      const title = titleField
   186→        ? String(node.record.data[titleField.fieldKey] ?? '')
   187→        : '';
   188→      return (
   189→        code.toLowerCase().includes(needle) ||
   190→        title.toLowerCase().includes(needle)
   191→      );
   192→    },
   193→    [codeField, needle, titleField],
   194→  );
   195→
   196→  /** Visible nodes while searching: matches + their ancestors. */
   197→  const visibleNodes = useMemo(() => {
   198→    if (!isSearching) return nodes;
   199→    const keep = new Set<string>();
   200→    const match = (list: TreeNode[]): boolean => {
   201→      let any = false;
   202→      for (const node of list) {
   203→        const childMatch = match(node.children);
   204→        if (childMatch || nodeMatches(node)) {
   205→          keep.add(node.record.id);
   206→          any = true;
   207→        }
   208→      }
   209→      return any;
   210→    };
   211→    match(nodes);
   212→    return nodes.filter((node) => keep.has(node.record.id));
   213→  }, [isSearching, nodeMatches, nodes]);
   214→
   215→  const toggle = useCallback((recordId: string) => {
   216→    setCollapsed((prev) => {
   217→      const next = new Set(prev);
   218→      if (next.has(recordId)) next.delete(recordId);
   219→      else next.add(recordId);
   220→      return next;
   221→    });
   222→  }, []);
   223→
   224→  // ── Node context menu (زیرشاخه جدید / حذف شاخه) ─────────────────
   225→  const [menu, setMenu] = useState<{
   226→    x: number;
   227→    y: number;
   228→    recordId: string;
   229→  } | null>(null);
   230→  const closeMenu = useCallback(() => setMenu(null), []);
   231→  const menuRef = useDismiss<HTMLDivElement>(menu !== null, closeMenu);
   232→
   233→  /** Node the open menu belongs to (hidden again once it disappears). */
   234→  const menuNode = useMemo(
   235→    () =>
   236→      menu
   237→        ? nodes.find((node) => node.record.id === menu.recordId) ?? null
   238→        : null,
   239→    [menu, nodes],
   240→  );
   241→

   243→  const openMenu = useCallback((event: React.MouseEvent, recordId: string) => {
   244→    event.preventDefault();

   246→    const y = Math.min(event.clientY, window.innerHeight - 104);
   247→    setMenu({ x: Math.max(8, x), y: Math.max(8, y), recordId });
   248→  }, []);
   249→
   250→  // Fixed-position menu must not linger above a scrolled/resized tree.
   251→  useEffect(() => {
   252→    if (!menu) return;
   253→    const close = () => setMenu(null);
   254→    window.addEventListener('scroll', close, true);
   255→    window.addEventListener('resize', close);
   256→    return () => {
   257→      window.removeEventListener('scroll', close, true);
   258→      window.removeEventListener('resize', close);
   259→    };
   260→  }, [menu]);
   261→
   262→  const handleMenuNewSub = useCallback(() => {
   263→    if (!menu || !parentField) return;
   264→    onNewChild({ [parentField.fieldKey]: menu.recordId });
   265→    closeMenu();
   266→  }, [closeMenu, menu, onNewChild, parentField]);
   267→
   268→  const handleMenuDelete = useCallback(() => {
   269→    if (!menu || mode === 'new' || isBusy) return;
   270→    const node = nodes.find((item) => item.record.id === menu.recordId);
   271→    if (!node || node.children.length > 0) return;
   272→    onDeleteNode(menu.recordId);
   273→    closeMenu();
   274→  }, [closeMenu, isBusy, menu, mode, nodes, onDeleteNode]);
   275→
   276→  // ── Body tabs (tabKey !== null groups, definition order kept) ───
   277→  const bodyTabs = useMemo(() => {
   278→    const groups: Array<{ key: string; name: string; fields: FormFieldDef[] }> = [];
   279→    for (const field of sortedFields) {
   280→      if (field.tabKey === null) continue;
   281→      let group = groups.find((candidate) => candidate.key === field.tabKey);
   282→      if (!group) {
   283→        group = {
   284→          key: field.tabKey,
   285→          name: field.tabName ?? field.tabKey,
   286→          fields: [],
   287→        };
   288→        groups.push(group);
   289→      }
   290→      group.fields.push(field);
   291→    }
   292→    return groups;
   293→  }, [sortedFields]);
   294→
   295→  const headerFields = useMemo(
   296→    () => sortedFields.filter((field) => field.tabKey === null),
   297→    [sortedFields],
   298→  );
   299→
   300→  const [activeTabKey, setActiveTabKey] = useState<string | null>(null);
   301→  const activeBodyTab =
   302→    bodyTabs.find((tab) => tab.key === activeTabKey) ?? bodyTabs[0] ?? null;
   303→
   304→  const current = records.find((record) => record.id === selectedId) ?? null;
   305→  const showDetail = mode === 'new' || current !== null;
   306→
   307→  const renderFields = (list: FormFieldDef[]) => (
   308→    <div className="grid grid-cols-1 gap-x-5 gap-y-4 p-4 sm:grid-cols-2">
   309→      {list.map((field) =>
   310→        parentField && field.fieldKey === parentField.fieldKey ? (

   312→            key={field.fieldKey}
   313→            field={field}
   314→            nodes={nodes}
   315→            excludeIds={descendantIds}
   316→            value={

   318→                ? (draft[field.fieldKey] as string)
   319→                : null
   320→            }
   321→            onChange={(value) => onFieldChange(field.fieldKey, value)}
   322→            disabled={isBusy}
   323→          />
   324→        ) : (
   325→          <FieldRenderer
   326→            key={field.fieldKey}
   327→            field={field}
   328→            value={draft[field.fieldKey] ?? null}
   329→            onChange={(value) => onFieldChange(field.fieldKey, value)}
   330→            disabled={isBusy}
   331→            lookupUrlBase={lookupUrlBase}
   332→          />
   333→        ),
   334→      )}
   335→    </div>
   336→  );
   337→
   338→  return (
   339→    <div className="flex h-full min-h-0 flex-col md:flex-row">
   340→      {/* ── Box 1: tree panel (درختواره) ─────────────────────────── */}
   341→      <aside
   342→        aria-label={t('formWindow.treePanel')}
   343→        className="flex max-h-56 w-full shrink-0 flex-col border-b border-white/10 bg-white/[0.02] md:max-h-none md:w-60 md:border-b-0 md:border-e lg:w-72"
   344→      >
   345→        <div className="shrink-0 border-b border-white/10 px-3 pb-2.5 pt-3">
   346→          {/* Panel title + record count */}
   347→          <div className="mb-2.5 flex items-center gap-1.5">
   348→            <Network className="h-3.5 w-3.5 text-white/45" aria-hidden="true" />
   349→            <span className="text-[12px] font-semibold text-white/85">
   350→              {t('formWindow.treePanel')}
   351→            </span>
   352→            <span className="rounded bg-white/[0.07] px-1.5 py-px text-[10px] tabular-nums text-white/45">
   353→              {formatNumber(records.length)}
   354→            </span>
   355→          </div>
   356→
   357→          {/* Search inside the tree (CRUD lives in the action bar) */}
   358→          <div className="relative">
   359→            <Search
   360→              className="pointer-events-none absolute start-2 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-white/35"
   361→              aria-hidden="true"
   362→            />
   363→            <input
   364→              type="text"
   365→              value={query}
   366→              onChange={(event) => setQuery(event.target.value)}
   367→              placeholder={t('formWindow.treeSearch')}
   368→              aria-label={t('formWindow.treeSearch')}
   369→              className={cn(controlClasses, 'h-7 ps-7 pe-7')}
   370→            />
   371→            {isSearching && (
   372→              <button
   373→                type="button"
   374→                title={t('formWindow.clear')}
   375→                aria-label={t('formWindow.clear')}
   376→                onClick={() => setQuery('')}
   377→                className="absolute end-1 top-1/2 flex h-5 w-5 -translate-y-1/2 items-center justify-center rounded-md text-white/40 transition-colors hover:bg-white/10 hover:text-white focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-white/40"
   378→              >
   379→                <X className="h-3 w-3" />
   380→              </button>
   381→            )}
   382→          </div>
   383→        </div>
   384→
   385→        {/* Nodes */}
   386→        <div
   387→          role="tree"
   388→          aria-label={t('formWindow.treePanel')}
   389→          className="min-h-0 flex-1 overflow-y-auto p-1.5 [scrollbar-width:thin]"
   390→        >
   391→          {nodes.length === 0 ? (
   392→            <div className="flex flex-col items-center gap-2 px-3 py-8 text-center">
   393→              <Network className="h-5 w-5 text-white/25" aria-hidden="true" />
   394→              <p className="text-[12px] text-white/40">
   395→                {t('formWindow.emptyRecords')}
   396→              </p>
   397→            </div>
   398→          ) : visibleNodes.length === 0 ? (
   399→            <p className="px-3 py-8 text-center text-[12px] text-white/40">
   400→              {t('formWindow.noMatch')}
   401→            </p>
   402→          ) : (
   403→            visibleNodes.map((node) => {
   404→              const hasChildren = node.children.length > 0;
   405→              const isCollapsed = !isSearching && collapsed.has(node.record.id);
   406→              const isSelected =
   407→                mode !== 'new' && node.record.id === selectedId;
   408→              return (
   409→                <div
   410→                  key={node.record.id}
   411→                  role="treeitem"
   412→                  aria-selected={isSelected}
   413→                  aria-level={node.depth + 1}
   414→                  aria-expanded={hasChildren ? !isCollapsed : undefined}
   415→                  style={{ paddingInlineStart: node.depth * 14 }}
   416→                  onContextMenu={(event) => openMenu(event, node.record.id)}
   417→                >
   418→                  <div
   419→                    className={cn(
   420→                      'group flex items-center gap-0.5 rounded-lg pe-1 transition-colors',
   421→                      isSelected
   422→                        ? 'bg-sky-500/15 ring-1 ring-sky-400/30'
   423→                        : 'hover:bg-white/[0.05]',
   424→                    )}
   425→                  >
   426→                    <button
   427→                      type="button"
   428→                      onClick={() => onSelect(node.record.id)}
   429→                      className="flex min-w-0 flex-1 items-center gap-1.5 py-1.5 ps-1.5 text-start"
   430→                    >
   431→                      {hasChildren ? (
   432→                        <Folder
   433→                          className="h-3.5 w-3.5 shrink-0 text-amber-400/80"
   434→                          aria-hidden="true"
   435→                        />
   436→                      ) : (
   437→                        <FileText
   438→                          className="h-3.5 w-3.5 shrink-0 text-white/40"
   439→                          aria-hidden="true"
   440→                        />
   441→                      )}
   442→                      <span className="min-w-0 flex-1 truncate text-[12px] text-white/85">
   443→                        {titleField
   444→                          ? formatFieldValue(
   445→                              titleField,
   446→                              node.record.data[titleField.fieldKey],
   447→                              formatNumber,
   448→                            )
   449→                          : node.record.id}
   450→                      </span>
   451→                      {codeField && (
   452→                        <span className="shrink-0 rounded bg-white/[0.07] px-1.5 py-0.5 font-mono text-[10px] text-white/60">
   453→                          {formatFieldValue(
   454→                            codeField,
   455→                            node.record.data[codeField.fieldKey],
   456→                            formatNumber,
   457→                          )}
   458→                        </span>
   459→                      )}
   460→                    </button>
   461→                    <button
   462→                      type="button"
   463→                      title={t('formWindow.actionCopy')}
   464→                      aria-label={t('formWindow.actionCopy')}
   465→                      onClick={() => onCopyNode(node.record)}
   466→                      className="flex h-5 w-5 shrink-0 items-center justify-center rounded-md text-white/35 transition-colors hover:bg-white/10 hover:text-white focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-white/40"
   467→                    >
   468→                      <Copy className="h-3 w-3" />
   469→                    </button>
   470→                    <button
   471→                      type="button"
   472→                      aria-label={
   473→                        hasChildren ? t('formWindow.treePanel') : undefined
   474→                      }
   475→                      disabled={!hasChildren}
   476→                      onClick={() => toggle(node.record.id)}
   477→                      className={cn(
   478→                        'flex h-5 w-5 shrink-0 items-center justify-center rounded-md text-white/45 transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-white/40',
   479→                        hasChildren
   480→                          ? 'hover:bg-white/10 hover:text-white'
   481→                          : 'pointer-events-none opacity-0',
   482→                      )}
   483→                    >
   484→                      <ChevronRight
   485→                        className={cn(
   486→                          'h-3.5 w-3.5 transition-transform rtl:-scale-x-100',
   487→                          !isCollapsed && 'rotate-90 rtl:-rotate-90',
   488→                        )}
   489→                      />
   490→                    </button>
   491→                  </div>
   492→                </div>
   493→              );
   494→            })
   495→          )}
   496→        </div>
   497→      </aside>
   498→
   499→      {/* ── Box 2: detail (fields + tabs) ────────────────────────── */}
   500→      <section
   501→        aria-label={t('formWindow.mainSection')}
   502→        className="flex min-h-0 min-w-0 flex-1 flex-col overflow-y-auto [scrollbar-width:thin]"
   503→      >
   504→        {showDetail ? (
   505→          <>
   506→            <div className="bg-white/[0.02]">
   507→              {renderFields(headerFields)}
   508→            </div>
   509→            {bodyTabs.length > 0 && (
   510→              <div className="flex min-h-0 flex-col">
   511→                <FormTabs
   512→                  tabs={bodyTabs.map((tab) => ({
   513→                    key: tab.key,
   514→                    name: tab.name,
   515→                    count: tab.fields.length,
   516→                  }))}
   517→                  active={activeBodyTab?.key ?? null}
   518→                  onChange={setActiveTabKey}
   519→                  accentColor={accentColor}
   520→                  direction={direction}
   521→                >
   522→                  {renderFields(activeBodyTab?.fields ?? [])}
   523→                </FormTabs>
   524→              </div>
   525→            )}
   526→          </>
   527→        ) : records.length === 0 ? (
   528→          <div className="flex flex-1 flex-col items-center justify-center gap-2 p-6 text-center">
   529→            <Network className="h-6 w-6 text-white/25" aria-hidden="true" />
   530→            <p className="text-[13px] text-white/55">
   531→              {t('formWindow.emptyRecords')}
   532→            </p>
   533→            <p className="text-[11px] text-white/35">
   534→              {t('formWindow.emptyRecordsHint')}
   535→            </p>
   536→          </div>
   537→        ) : (
   538→          <div className="flex flex-1 flex-col items-center justify-center gap-2 p-6 text-center">
   539→            <Network className="h-6 w-6 text-white/25" aria-hidden="true" />
   540→            <p className="text-[12px] text-white/45">
   541→              {t('formWindow.treeSelectHint')}
   542→            </p>
   543→          </div>
   544→        )}
   545→      </section>
   546→
   547→      {/* ── Node context menu (right click) ─────────────────────────── */}
   548→      <AnimatePresence>
   549→        {menu && menuNode && (
   550→          <motion.div
   551→            ref={menuRef}
   552→            role="menu"
   553→            aria-label={t('formWindow.treePanel')}
   554→            initial={{ opacity: 0, scale: 0.95, y: -3 }}
   555→            animate={{ opacity: 1, scale: 1, y: 0 }}
   556→            exit={{ opacity: 0, scale: 0.95, y: -3 }}
   557→            transition={{ duration: 0.13, ease: 'easeOut' }}
   558→            style={{ left: menu.x, top: menu.y }}
   559→            className="fixed z-[70] min-w-[190px] origin-top overflow-hidden rounded-xl bg-[#1b1e26]/[0.97] p-1 shadow-2xl shadow-black/50 ring-1 ring-white/10 backdrop-blur-xl"
   560→          >
   561→            <button
   562→              type="button"
   563→              role="menuitem"
   564→              disabled={!parentField || isBusy}
   565→              onClick={handleMenuNewSub}
   566→              className="flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-start text-[12px] text-white/85 transition-colors enabled:hover:bg-emerald-500/15 enabled:hover:text-emerald-300 focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-emerald-400/50 disabled:cursor-not-allowed disabled:opacity-40"
   567→            >
   568→              <FolderPlus
   569→                className="h-3.5 w-3.5 shrink-0 text-emerald-400"
   570→                aria-hidden="true"
   571→              />
   572→              {t('formWindow.treeContextNewSub')}
   573→            </button>
   574→            <button
   575→              type="button"
   576→              role="menuitem"
   577→              title={
   578→                menuNode.children.length > 0
   579→                  ? t('formWindow.treeContextDeleteDisabled')
   580→                  : undefined
   581→              }
   582→              disabled={menuNode.children.length > 0 || mode === 'new' || isBusy}
   583→              onClick={handleMenuDelete}
   584→              className="flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-start text-[12px] text-white/85 transition-colors enabled:hover:bg-rose-500/15 enabled:hover:text-rose-300 focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-rose-400/50 disabled:cursor-not-allowed disabled:opacity-40"
   585→            >
   586→              <Trash2
   587→                className="h-3.5 w-3.5 shrink-0 text-rose-400"
   588→                aria-hidden="true"
   589→              />
   590→              {t('formWindow.treeContextDelete')}
   591→            </button>
   592→          </motion.div>
   593→        )}
   594→      </AnimatePresence>
   595→    </div>
   596→  );
   597→}
   598→
   599→interface ParentPickerProps {
   600→  field: FormFieldDef;
   601→  nodes: TreeNode[];
   602→  /** Record ids that may not become a parent (self + descendants). */
   603→  excludeIds: ReadonlySet<string>;
   604→  value: string | null;
   605→  onChange: (value: FormFieldValue) => void;
   606→  disabled?: boolean;
   607→}
   608→
   609→/**
   610→ * Self-referencing parent selector of tree forms: lists the form's own
   611→ * records in hierarchy order (indented, code + title) and stores the
   612→ * chosen record's id — the same value the tree hierarchy is built from.
   613→ */
   614→function ParentPicker({
   615→  field,
   616→  nodes,
   617→  excludeIds,
   618→  value,
   619→  onChange,
   620→  disabled,
   621→}: ParentPickerProps) {
   622→  const { locale, t } = useTranslation();
   623→  const listboxId = useId();
   624→  const [open, setOpen] = useState(false);
   625→  const [query, setQuery] = useState('');
   626→  const [activeIndex, setActiveIndex] = useState(0);
   627→  const rootRef = useDismiss<HTMLDivElement>(
   628→    open,
   629→    useCallback(() => setOpen(false), []),
   630→  );
   631→
   632→  const options = useMemo(
   633→    () =>
   634→      nodes
   635→        .filter((node) => !excludeIds.has(node.record.id))
   636→        .map((node) => ({ node, depth: node.depth })),
   637→    [excludeIds, nodes],
   638→  );
   639→
   640→  const needle = query.trim().toLowerCase();
   641→  const filtered = useMemo(() => {
   642→    if (!needle) return options;
   643→    return options.filter(({ node }) => {
   644→      const title = String(node.record.data.title ?? '');
   645→      const code = String(node.record.data.code ?? '');
   646→      return (
   647→        title.toLowerCase().includes(needle) ||
   648→        code.toLowerCase().includes(needle)
   649→      );
   650→    });
   651→  }, [needle, options]);
   652→
   653→  const labelOf = useCallback(
   654→    (node: TreeNode) => {
   655→      const title = String(node.record.data.title ?? '').trim();
   656→      const code = String(node.record.data.code ?? '').trim();
   657→      return code && title ? `${code} — ${title}` : title || code || node.record.id;
   658→    },
   659→    [],
   660→  );
   661→
   662→  const selectedNode = useMemo(
   663→    () => nodes.find((node) => node.record.id === value) ?? null,
   664→    [nodes, value],
   665→  );
   666→
   667→  const openPanel = () => {
   668→    setQuery('');
   669→    setActiveIndex(0);
   670→    setOpen(true);
   671→  };
   672→
   673→  const pick = (node: TreeNode) => {
   674→    onChange(node.record.id);
   675→    setOpen(false);
   676→  };
   677→
   678→  const onKeyDown = (event: React.KeyboardEvent) => {
   679→    if (!open) {
   680→      if (event.key === 'ArrowDown' || event.key === 'Enter') {
   681→        event.preventDefault();
   682→        openPanel();
   683→      }
   684→      return;
   685→    }
   686→    if (event.key === 'ArrowDown') {
   687→      event.preventDefault();
   688→      setActiveIndex((index) => Math.min(filtered.length - 1, index + 1));
   689→    } else if (event.key === 'ArrowUp') {
   690→      event.preventDefault();
   691→      setActiveIndex((index) => Math.max(0, index - 1));
   692→    } else if (event.key === 'Enter') {
   693→      event.preventDefault();
   694→      const target = filtered[activeIndex];
   695→      if (target) pick(target.node);
   696→    } else if (event.key === 'Escape') {
   697→      event.stopPropagation();
   698→      setOpen(false);
   699→    }
   700→  };
   701→
   702→  return (
   703→    <div
   704→      className={cn(
