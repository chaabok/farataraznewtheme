    55→    55→export interface DataFormWindowProps {
    56→    56→  /** Window id — `form:<formKey>` (moduleKey defaults to `settings`). */
    57→    57→  moduleId: string;
    58→    58→  /** Descriptor attached by `openFormWindow` in the desktop store. */
    59→    59→  form?: FormWindowInfo;
    60→    60→  /** Closes the hosting window ("save & close" / "delete & close"). */
    61→    61→  onClose?: () => void;
    62→    62→}
    63→    63→
    64→    64→/** Which combined action a pending delete confirmation belongs to. */
    65→    65→type DeleteIntent = 'current' | 'new' | 'close' | null;
    66→    66→
    67→    67→/** Outcome of the shared save pipeline. */
    68→    68→type SaveResult = 'saved' | 'clean' | 'invalid' | 'error';
    69→    69→
    70→    70→/** Fallback parsing of the window id when no descriptor is attached. */
    71→    71→function parseModuleId(moduleId: string): {
    72→    72→  moduleKey: string | null;
    73→    73→  formKey: string | null;
    74→    74→} {
    75→    75→  if (!moduleId.startsWith('form:')) {
    76→    76→    return { moduleKey: null, formKey: null };
    77→    77→  }
    78→    78→  const formKey = moduleId.slice('form:'.length);
    79→    79→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    80→    80→}
    81→    81→
    82→    82→/** Sentinel kept in state so load errors render through the dictionaries. */
    83→    83→const GENERIC_LOAD_ERROR = 'formWindow.loadError';
    84→    84→
    85→    85→/**
    86→    86→ * Generic, database-driven form window: loads a FormDef + its records and
    87→    87→ * drives the whole view/new/edit state machine (dirty tracking, CRUD,
    88→    88→ * record navigation, Ctrl+S, records drawer) through the shared form
    89→    89→ * component library. One implementation renders every ERP form.
    90→    90→ */
    91→    91→export function DataFormWindow({ moduleId, form, onClose }: DataFormWindowProps) {
    92→    92→  const { locale, direction, t, tf, formatNumber } = useTranslation();
    93→    93→
    94→    94→  const fallback = useMemo(() => parseModuleId(moduleId), [moduleId]);
    95→    95→  const moduleKey = form?.moduleKey ?? fallback.moduleKey;
    96→    96→  const formKey = form?.formKey ?? fallback.formKey;
    97→    97→  const basePath =
    98→    98→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
    99→    99→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   100→   100→
   101→   101→  // ── Data ──
   102→   102→  const [def, setDef] = useState<FormDef | null>(null);
   103→   103→  const [records, setRecords] = useState<FormRecord[]>([]);
   104→   104→  const [isLoading, setIsLoading] = useState(true);
   105→   105→  const [loadError, setLoadError] = useState<string | null>(null);
   106→   106→
   107→   107→  // ── Record state machine ──
   108→   108→  const [mode, setMode] = useState<FormMode>('view');
   109→   109→  const [currentIndex, setCurrentIndex] = useState(-1);
   110→   110→  const [draft, setDraft] = useState<FormDataMap>({});
   111→   111→  const [baseline, setBaseline] = useState<FormDataMap>({});
   112→   112→  const [isSaving, setIsSaving] = useState(false);
   113→   113→  const [isDeleting, setIsDeleting] = useState(false);
   114→   114→  /**
   115→   115→   * Pending delete request awaiting the confirm dialog — `current` =
   116→   116→   * delete and stay, `new` = delete then reset for a fresh entry,
   117→   117→   * `close` = delete then close the window.
   118→   118→   */
   119→   119→  const [deleteIntent, setDeleteIntent] = useState<DeleteIntent>(null);
   120→   120→  const [isListOpen, setIsListOpen] = useState(false);
   121→   121→  const [activeTabKey, setActiveTabKey] = useState<string | null>(null);
   122→   122→
   123→   123→  const current =
   124→   124→    currentIndex >= 0 && currentIndex < records.length
   125→   125→      ? records[currentIndex]
   126→   126→      : null;
   127→   127→
   128→   128→  const isDirty = useMemo(() => {
   129→   129→    const keys = new Set([...Object.keys(baseline), ...Object.keys(draft)]);
   130→   130→    for (const key of keys) {
   131→   131→      if (isChangedValue(baseline[key], draft[key])) return true;
   132→   132→    }
   133→   133→    return false;
   134→   134→  }, [baseline, draft]);
   135→   135→
   136→   136→  const status: FormStatus =
   137→   137→    mode === 'new' ? (isDirty ? 'unsaved' : 'new') : isDirty ? 'unsaved' : 'saved';
   138→   138→
   139→   139→  // ── Loading ──
   140→   140→  const load = useCallback(async () => {
   141→   141→    if (!basePath) {
   142→   142→      setLoadError(GENERIC_LOAD_ERROR);
   143→   143→      setIsLoading(false);
   144→   144→      return;
   145→   145→    }
   146→   146→    setIsLoading(true);
   147→   147→    setLoadError(null);
   148→   148→    try {
   149→   149→      const [defRes, recordsRes] = await Promise.all([
   150→   150→        apiFetch<{ data: FormDef }>(
   151→   151→          `${basePath}?locale=${encodeURIComponent(locale)}`,
   152→   152→        ),
   153→   153→        apiFetch<{ data: FormRecord[] }>(`${basePath}/records`),
   154→   154→      ]);
   155→   155→      const nextDef = defRes.data;
   156→   156→      const nextRecords = recordsRes.data ?? [];
   157→   157→      setDef(nextDef);
   158→   158→      setRecords(nextRecords);
   159→   159→
   160→   160→      if (nextRecords.length) {
   161→   161→        const lastIndex = nextRecords.length - 1;
   162→   162→        const last = nextRecords[lastIndex];
   163→   163→        setCurrentIndex(lastIndex);
   164→   164→        setDraft({ ...last.data });
   165→   165→        setBaseline({ ...last.data });
   166→   166→        setMode('view');
   167→   167→      } else {
   168→   168→        const defaults = buildDefaults(nextDef.fields);
   169→   169→        setCurrentIndex(-1);
   170→   170→        setDraft({ ...defaults });
   171→   171→        setBaseline({ ...defaults });
   172→   172→        setMode('new');
   173→   173→      }
   174→   174→
   175→   175→      const derived = deriveTabs(nextDef, locale);
   176→   176→      // The first group is the always-visible header section; the active
   177→   177→      // tab state only drives the remaining body tabs (default: first one).
   178→   178→      const firstBodyTab = derived[1] ?? null;
   179→   179→      setActiveTabKey(firstBodyTab ? firstBodyTab.key : null);
   180→   180→    } catch (error) {
   181→   181→      if (error instanceof SessionExpiredError) return;
   182→   182→      setLoadError(
   183→   183→        error instanceof Error && error.message
   184→   184→          ? error.message
   185→   185→          : GENERIC_LOAD_ERROR,
   186→   186→      );
   187→   187→    } finally {
   188→   188→      setIsLoading(false);
   189→   189→    }
   190→   190→  }, [basePath, locale]);
   191→   191→
   192→   192→  useEffect(() => {
   193→   193→    void load();
   194→   194→  }, [load]);
   195→   195→
   196→   196→  const refreshRecords = useCallback(async (): Promise<FormRecord[]> => {
   197→   197→    if (!basePath) return [];
   198→   198→    const res = await apiFetch<{ data: FormRecord[] }>(`${basePath}/records`);
   199→   199→    const list = res.data ?? [];
   200→   200→    setRecords(list);
   201→   201→    return list;
   202→   202→  }, [basePath]);
   203→   203→
   204→   204→  // ── Record transitions ──
   205→   205→  const applyRecord = useCallback((index: number, list: FormRecord[]) => {
   206→   206→    const target = list[index];
   207→   207→    setCurrentIndex(index);
   208→   208→    setMode('view');
   209→   209→    if (target) {
   210→   210→      setDraft({ ...target.data });
   211→   211→      setBaseline({ ...target.data });
   212→   212→    }
   213→   213→  }, []);
   214→   214→
   215→   215→  const navigateTo = useCallback(
   216→   216→    (index: number) => {
   217→   217→      if (isSaving || isDeleting) return;
   218→   218→      if (index < 0 || index >= records.length) return;
   219→   219→      if (mode !== 'view' && isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) {
   220→   220→        return;
   221→   221→      }
   222→   222→      applyRecord(index, records);
   223→   223→    },
   224→   224→    [applyRecord, isDeleting, isDirty, isSaving, mode, records, t],
   225→   225→  );
   226→   226→
   227→   227→  const startNew = useCallback(
   228→   228→    (preset?: FormDataMap) => {
   229→   229→      if (!def) return;
   230→   230→      const base = { ...buildDefaults(def.fields), ...(preset ?? {}) };
   231→   231→      setDraft(base);
   232→   232→      setBaseline({ ...base });
   233→   233→      setMode('new');
   234→   234→      setCurrentIndex(-1);
   235→   235→    },
   236→   236→    [def],
   237→   237→  );
   238→   238→
   239→   239→  const handleNew = useCallback(() => {
   240→   240→    if (mode === 'new') return;
   241→   241→    if (isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) return;
   242→   242→    startNew();
   243→   243→  }, [isDirty, mode, startNew, t]);
   244→   244→
   245→   245→  const handleCopy = useCallback(() => {
   246→   246→    if (mode !== 'view' || !current) return;
   247→   247→    startNew({ ...current.data });
   248→   248→    toast({ title: t('formWindow.copyToast') });
   249→   249→  }, [current, mode, startNew, t]);
   250→   250→
   251→   251→  /**
   252→   252→   * Shared save pipeline for every save action: validates required
   253→   253→   * fields, writes the draft to the database and refreshes the record
   254→   254→   * list. `keepRecord: false` skips re-selecting the saved record
   255→   255→   * (used by "save & new", which resets the form right after).
   256→   256→   */
   257→   257→  const performSave = useCallback(
   258→   258→    async (options?: { keepRecord?: boolean }): Promise<SaveResult> => {
   259→   259→      if (!def || !basePath || isSaving || isDeleting) return 'error';
   260→   260→      if (mode === 'view' || !isDirty) return 'clean';
   261→   261→
   262→   262→      const missingField = def.fields.find((field) =>
   263→   263→        isEmptyRequiredValue(field, draft[field.fieldKey]),
   264→   264→      );
   265→   265→      if (missingField) {
   266→   266→        toast({
   267→   267→          title: fieldLabel(missingField, locale),
   268→   268→          description: t('formWindow.requiredHint'),
   269→   269→          variant: 'destructive',
   270→   270→        });
   271→   271→        return 'invalid';
   272→   272→      }
   273→   273→
   274→   274→      setIsSaving(true);
   275→   275→      try {
   276→   276→        const saved =
   277→   277→          mode === 'new'
   278→   278→            ? (
   279→   279→                await apiFetch<{ data: FormRecord }>(`${basePath}/records`, {
   280→   280→                  method: 'POST',
   281→   281→                  body: { data: draft },
   282→   282→                })
   283→   283→              ).data
   284→   284→            : (
   285→   285→                await apiFetch<{ data: FormRecord }>(
   286→   286→                  `${basePath}/records/${encodeURIComponent(current?.id ?? '')}`,
   287→   287→                  { method: 'PUT', body: { data: draft } },
   288→   288→                )
   289→   289→              ).data;
   290→   290→
   291→   291→        const list = await refreshRecords();
   292→   292→        if (options?.keepRecord !== false) {
   293→   293→          const index = list.findIndex((record) => record.id === saved.id);
   294→   294→          if (index >= 0) {
   295→   295→            applyRecord(index, list);
   296→   296→          } else {
   297→   297→            setMode('view');
   298→   298→          }
   299→   299→        }
   300→   300→        return 'saved';
   301→   301→      } catch (error) {
   302→   302→        if (!(error instanceof SessionExpiredError)) {
   303→   303→          toast({
   304→   304→            title: t('formWindow.saveError'),
   305→   305→            description: error instanceof Error ? error.message : undefined,
   306→   306→            variant: 'destructive',
   307→   307→          });
   308→   308→        }
   309→   309→        return 'error';
   310→   310→      } finally {
   311→   311→        setIsSaving(false);
   312→   312→      }
   313→   313→    },
   314→   314→    [
   315→   315→      applyRecord,
   316→   316→      basePath,
   317→   317→      current,
   318→   318→      def,
   319→   319→      draft,
   320→   320→      isDeleting,
   321→   321→      isDirty,
   322→   322→      isSaving,
   323→   323→      locale,
   324→   324→      mode,
   325→   325→      refreshRecords,
   326→   326→      t,
   327→   327→    ],
   328→   328→  );
   329→   329→
   330→   330→  const handleSave = useCallback(async () => {
   331→   331→    const result = await performSave();
   332→   332→    if (result === 'saved') toast({ title: t('formWindow.saveSuccess') });
   333→   333→  }, [performSave, t]);
   334→   334→
   335→   335→  /** Save & new: write the record, then reset the form for a fresh entry. */
   336→   336→  const handleSaveNew = useCallback(async () => {
   337→   337→    if (isSaving || isDeleting) return;
   338→   338→    const result = await performSave({ keepRecord: false });
   339→   339→    if (result === 'error' || result === 'invalid') return;
   340→   340→    startNew();
   341→   341→    if (result === 'saved') toast({ title: t('formWindow.saveNewSuccess') });
   342→   342→  }, [isDeleting, isSaving, performSave, startNew, t]);
   343→   343→
   344→   344→  /** Save & close: write the record after validation, then close. */
   345→   345→  const handleSaveClose = useCallback(async () => {
   346→   346→    if (isSaving || isDeleting) return;
   347→   347→    const result = await performSave();
   348→   348→    if (result === 'error' || result === 'invalid') return;
   349→   349→    if (result === 'saved') toast({ title: t('formWindow.saveCloseSuccess') });
   350→   350→    onClose?.();
   351→   351→  }, [isDeleting, isSaving, onClose, performSave, t]);
   352→   352→
   353→   353→  /** First click on any delete action opens the confirm dialog. */
   354→   354→  const requestDelete = useCallback(
   355→   355→    (intent: Exclude<DeleteIntent, null>) => {
   356→   356→      if (!current || mode === 'new' || isSaving || isDeleting) return;
   357→   357→      setDeleteIntent(intent);
   358→   358→    },
   359→   359→    [current, isDeleting, isSaving, mode],
   360→   360→  );
   361→   361→
   362→   362→  /** Runs after the user presses «بلی» in the confirm dialog. */
   363→   363→  const confirmDelete = useCallback(async () => {
   364→   364→    if (isDeleting || !deleteIntent) return;
   365→   365→    const intent = deleteIntent;
   366→   366→    if (!basePath || !current) {
   367→   367→      setDeleteIntent(null);
   368→   368→      return;
   369→   369→    }
   370→   370→    setIsDeleting(true);
   371→   371→    try {
   372→   372→      // DELETE answers 204 with an empty body — nothing to parse.
   373→   373→      await apiFetch(
   374→   374→        `${basePath}/records/${encodeURIComponent(current.id)}`,
   375→   375→        {
   376→   376→          method: 'DELETE',
   377→   377→        },
   378→   378→      );
   379→   379→      setDeleteIntent(null);
   380→   380→      if (intent === 'close') {
   381→   381→        toast({ title: t('formWindow.deleteCloseSuccess') });
   382→   382→        onClose?.();
   383→   383→        return;
   384→   384→      }
   385→   385→      const list = await refreshRecords();
   386→   386→      if (intent === 'new') {
   387→   387→        startNew();
   388→   388→        toast({ title: t('formWindow.deleteNewSuccess') });
   389→   389→      } else if (list.length) {
   390→   390→        applyRecord(Math.min(currentIndex, list.length - 1), list);
   391→   391→        toast({ title: t('formWindow.deleteSuccess') });
   392→   392→      } else {
   393→   393→        setCurrentIndex(-1);
   394→   394→        startNew();
   395→   395→        toast({ title: t('formWindow.deleteSuccess') });
   396→   396→      }
   397→   397→    } catch (error) {
   398→   398→      setDeleteIntent(null);
   399→   399→      if (error instanceof SessionExpiredError) return;
   400→   400→      toast({
   401→   401→        title: error instanceof Error && error.message
   402→   402→          ? error.message
   403→   403→          : t('formWindow.loadError'),
   404→   404→        variant: 'destructive',
   405→   405→      });
   406→   406→    } finally {
   407→   407→      setIsDeleting(false);
   408→   408→    }
   409→   409→  }, [
   410→   410→    applyRecord,
   411→   411→    basePath,
   412→   412→    currentIndex,
   413→   413→    current,
   414→   414→    deleteIntent,
   415→   415→    isDeleting,
   416→   416→    onClose,
   417→   417→    refreshRecords,
   418→   418→    startNew,
   419→   419→    t,
   420→   420→  ]);
   421→   421→
   422→   422→  const handlePrint = useCallback(() => {
   423→   423→    toast({ title: t('formWindow.printToast') });
   424→   424→  }, [t]);
   425→   425→
   426→   426→  const handleFieldChange = useCallback(
   427→   427→    (fieldKey: string, value: FormFieldValue) => {
   428→   428→      setDraft((prev) => ({ ...prev, [fieldKey]: value }));
   429→   429→      setMode((prev) => (prev === 'view' ? 'edit' : prev));
   430→   430→    },
   431→   431→    [],
   432→   432→  );
   433→   433→
   434→   434→  // Ctrl+S / Cmd+S saves the active draft (Escape is handled by the dialog).
   435→   435→  useEffect(() => {
   436→   436→    const onKeyDown = (event: KeyboardEvent) => {
   437→   437→      if (
   438→   438→        (event.ctrlKey || event.metaKey) &&
   439→   439→        !event.shiftKey &&
   440→   440→        !event.altKey &&
   441→   441→        event.key.toLowerCase() === 's'
   442→   442→      ) {
   443→   443→        event.preventDefault();
   444→   444→        void handleSave();
   445→   445→      }
   446→   446→    };
   447→   447→    window.addEventListener('keydown', onKeyDown);
   448→   448→    return () => window.removeEventListener('keydown', onKeyDown);
   449→   449→  }, [handleSave]);
   450→   450→
   451→   451→  // ── Derived view models ──
   452→   452→  const sortedFields = useMemo(
   453→   453→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   454→   454→    [def],
   455→   455→  );
   456→   456→
   457→   457→  const tabs = useMemo(() => (def ? deriveTabs(def, locale) : []), [def, locale]);
   458→   458→
   459→   459→  /**
   460→   460→   * Two-part layout of tabbed forms: the first group is the header
   461→   461→   * section (its fields are always visible, without any tab strip) and
   462→   462→   * the remaining groups build the tabbed body section below it.
   463→   463→   */
   464→   464→  const isTabbed = def?.type === 'tabbed' && tabs.length > 0;
   465→   465→  const headerTab = isTabbed ? (tabs[0] ?? null) : null;
   466→   466→  const bodyTabs = isTabbed ? tabs.slice(1) : [];
   467→   467→  const activeBodyTab =
   468→   468→    bodyTabs.find((tab) => (tab.key ?? null) === (activeTabKey ?? null)) ??
   469→   469→    bodyTabs[0] ??
   470→   470→    null;
   471→   471→
   472→   472→  const listColumns = useMemo(
   473→   473→    () => pickListColumns(sortedFields),
   474→   474→    [sortedFields],
   475→   475→  );
   476→   476→
   477→   477→  const title = def
   478→   478→    ? formTitle(def, locale)
   479→   479→    : form?.name ?? t('common.loading');
   480→   480→  const subtitle =
   481→   481→    mode === 'new'
   482→   482→      ? t('formWindow.newMode')
   483→   483→      : current
   484→   484→        ? tf('formWindow.recordOf', {
   485→   485→            n: formatNumber(currentIndex + 1),
   486→   486→            total: formatNumber(records.length),
   487→   487→          })
   488→   488→        : undefined;
   489→   489→
   490→   490→  const accentColor = def?.accentColor || DEFAULT_FORM_ACCENT;
   491→   491→
