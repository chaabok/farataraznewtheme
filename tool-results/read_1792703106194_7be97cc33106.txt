    55→
    56→export interface DataFormWindowProps {
    57→  /** Window id — `form:<formKey>` (moduleKey defaults to `settings`). */
    58→  moduleId: string;
    59→  /** Descriptor attached by `openFormWindow` in the desktop store. */
    60→  form?: FormWindowInfo;
    61→  /** Closes the hosting window ("save & close" / "delete & close"). */
    62→  onClose?: () => void;
    63→}
    64→
    65→/** Which combined action a pending delete confirmation belongs to. */
    66→type DeleteIntent = 'current' | 'new' | 'close' | null;
    67→
    68→/** Outcome of the shared save pipeline. */
    69→type SaveResult = 'saved' | 'clean' | 'invalid' | 'error';
    70→
    71→/** Fallback parsing of the window id when no descriptor is attached. */
    72→function parseModuleId(moduleId: string): {
    73→  moduleKey: string | null;
    74→  formKey: string | null;
    75→} {
    76→  if (!moduleId.startsWith('form:')) {
    77→    return { moduleKey: null, formKey: null };
    78→  }
    79→  const formKey = moduleId.slice('form:'.length);
    80→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    81→}
    82→
    83→/** Sentinel kept in state so load errors render through the dictionaries. */
    84→const GENERIC_LOAD_ERROR = 'formWindow.loadError';
    85→
    86→/**
    87→ * Generic, database-driven form window: loads a FormDef + its records and
    88→ * drives the whole view/new/edit state machine (dirty tracking, CRUD,
    89→ * record navigation, Ctrl+S, records drawer) through the shared form
    90→ * component library. One implementation renders every ERP form.
    91→ */
    92→export function DataFormWindow({ moduleId, form, onClose }: DataFormWindowProps) {
    93→  const { locale, direction, t, tf, formatNumber } = useTranslation();
    94→
    95→  const fallback = useMemo(() => parseModuleId(moduleId), [moduleId]);
    96→  const moduleKey = form?.moduleKey ?? fallback.moduleKey;
    97→  const formKey = form?.formKey ?? fallback.formKey;
    98→  const basePath =
    99→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
   100→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   101→
   102→  /** Record preselection / start-mode requested by a data list (if any). */
   103→  const wantedRecordId = form?.recordId ?? null;
   104→  const wantsStartInNew = form?.startInNew ?? false;
   105→  /** Changes on every list-driven open → forces a fresh load. */
   106→  const formNonce = form?.nonce;
   107→
   108→  // ── Data ──
   109→  const [def, setDef] = useState<FormDef | null>(null);
   110→  const [records, setRecords] = useState<FormRecord[]>([]);
   111→  const [isLoading, setIsLoading] = useState(true);
   112→  const [loadError, setLoadError] = useState<string | null>(null);
   113→
   114→  // ── Record state machine ──
   115→  const [mode, setMode] = useState<FormMode>('view');
   116→  const [currentIndex, setCurrentIndex] = useState(-1);
   117→  const [draft, setDraft] = useState<FormDataMap>({});
   118→  const [baseline, setBaseline] = useState<FormDataMap>({});
   119→  const [isSaving, setIsSaving] = useState(false);
   120→  const [isDeleting, setIsDeleting] = useState(false);
   121→  /**
   122→   * Pending delete request awaiting the confirm dialog — `current` =
   123→   * delete and stay, `new` = delete then reset for a fresh entry,
   124→   * `close` = delete then close the window.
   125→   */
   126→  const [deleteIntent, setDeleteIntent] = useState<DeleteIntent>(null);
   127→  /** Record the confirmed delete targets (defaults to the loaded one). */
   128→  const [deleteTargetId, setDeleteTargetId] = useState<string | null>(null);
   129→  const [isListOpen, setIsListOpen] = useState(false);
   130→  const [activeTabKey, setActiveTabKey] = useState<string | null>(null);
   131→
   132→  const current =
   133→    currentIndex >= 0 && currentIndex < records.length
   134→      ? records[currentIndex]
   135→      : null;
   136→
   137→  const isDirty = useMemo(() => {
   138→    const keys = new Set([...Object.keys(baseline), ...Object.keys(draft)]);
   139→    for (const key of keys) {
   140→      if (isChangedValue(baseline[key], draft[key])) return true;
   141→    }
   142→    return false;
   143→  }, [baseline, draft]);
   144→
   145→  const status: FormStatus =
   146→    mode === 'new' ? (isDirty ? 'unsaved' : 'new') : isDirty ? 'unsaved' : 'saved';
   147→
   148→  // ── Loading ──
   149→  const load = useCallback(async () => {
   150→    if (!basePath) {
   151→      setLoadError(GENERIC_LOAD_ERROR);
   152→      setIsLoading(false);
   153→      return;
   154→    }
   155→    setIsLoading(true);
   156→    setLoadError(null);
   157→    try {
   158→      const [defRes, recordsRes] = await Promise.all([
   159→        apiFetch<{ data: FormDef }>(
   160→          `${basePath}?locale=${encodeURIComponent(locale)}`,
   161→        ),
   162→        apiFetch<{ data: FormRecord[] }>(`${basePath}/records`),
   163→      ]);
   164→      const nextDef = defRes.data;
   165→      const nextRecords = recordsRes.data ?? [];
   166→      setDef(nextDef);
   167→      setRecords(nextRecords);
   168→
   169→      if (wantsStartInNew) {
   170→        // Opened from a data list in "new record" mode.
   171→        const defaults = buildDefaults(nextDef.fields);
   172→        setCurrentIndex(-1);
   173→        setDraft({ ...defaults });
   174→        setBaseline({ ...defaults });
   175→        setMode('new');
   176→      } else if (nextRecords.length) {
   177→        // Default: the last record. When opened from a data list, the
   178→        // requested record (wantedRecordId) is selected instead.
   179→        let index = nextRecords.length - 1;
   180→        if (wantedRecordId) {
   181→          const found = nextRecords.findIndex(
   182→            (record) => record.id === wantedRecordId,
   183→          );
   184→          if (found >= 0) index = found;
   185→        }
   186→        const target = nextRecords[index];
   187→        setCurrentIndex(index);
   188→        setDraft({ ...target.data });
   189→        setBaseline({ ...target.data });
   190→        setMode('view');
   191→      } else {
   192→        const defaults = buildDefaults(nextDef.fields);
   193→        setCurrentIndex(-1);
   194→        setDraft({ ...defaults });
   195→        setBaseline({ ...defaults });
   196→        setMode('new');
   197→      }
   198→
   199→      const derived = deriveTabs(nextDef, locale);
   200→      // The first group is the always-visible header section; the active
   201→      // tab state only drives the remaining body tabs (default: first one).
   202→      const firstBodyTab = derived[1] ?? null;
   203→      setActiveTabKey(firstBodyTab ? firstBodyTab.key : null);
   204→    } catch (error) {
   205→      if (error instanceof SessionExpiredError) return;
   206→      setLoadError(
   207→        error instanceof Error && error.message
   208→          ? error.message
   209→          : GENERIC_LOAD_ERROR,
   210→      );
   211→    } finally {
   212→      setIsLoading(false);
   213→    }
   214→  }, [basePath, formNonce, locale, wantedRecordId, wantsStartInNew]);
   215→
   216→  useEffect(() => {
   217→    void load();
   218→  }, [load]);
   219→
   220→  const refreshRecords = useCallback(async (): Promise<FormRecord[]> => {
   221→    if (!basePath) return [];
   222→    const res = await apiFetch<{ data: FormRecord[] }>(`${basePath}/records`);
   223→    const list = res.data ?? [];
   224→    setRecords(list);
   225→    return list;
   226→  }, [basePath]);
   227→
   228→  // ── Record transitions ──
   229→  const applyRecord = useCallback((index: number, list: FormRecord[]) => {
   230→    const target = list[index];
   231→    setCurrentIndex(index);
   232→    setMode('view');
   233→    if (target) {
   234→      setDraft({ ...target.data });
   235→      setBaseline({ ...target.data });
   236→    }
   237→  }, []);
   238→
   239→  const navigateTo = useCallback(
   240→    (index: number) => {
   241→      if (isSaving || isDeleting) return;
   242→      if (index < 0 || index >= records.length) return;
   243→      if (mode !== 'view' && isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) {
   244→        return;
   245→      }
   246→      applyRecord(index, records);
   247→    },
   248→    [applyRecord, isDeleting, isDirty, isSaving, mode, records, t],
   249→  );
   250→
   251→  const startNew = useCallback(
   252→    (preset?: FormDataMap) => {
   253→      if (!def) return;
   254→      const base = { ...buildDefaults(def.fields), ...(preset ?? {}) };
   255→      setDraft(base);
   256→      setBaseline({ ...base });
   257→      setMode('new');
   258→      setCurrentIndex(-1);
   259→    },
   260→    [def],
   261→  );
   262→
   263→  const handleNew = useCallback(() => {
   264→    if (mode === 'new') return;
   265→    if (isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) return;
   266→    startNew();
   267→  }, [isDirty, mode, startNew, t]);
   268→
   269→  const handleCopy = useCallback(() => {
   270→    if (mode !== 'view' || !current) return;
   271→    startNew({ ...current.data });
   272→    toast({ title: t('formWindow.copyToast') });
   273→  }, [current, mode, startNew, t]);
   274→
   275→  /**
   276→   * Shared save pipeline for every save action: validates required
   277→   * fields, writes the draft to the database and refreshes the record
   278→   * list. `keepRecord: false` skips re-selecting the saved record
   279→   * (used by "save & new", which resets the form right after).
   280→   */
   281→  const performSave = useCallback(
   282→    async (options?: { keepRecord?: boolean }): Promise<SaveResult> => {
   283→      if (!def || !basePath || isSaving || isDeleting) return 'error';
   284→      if (mode === 'view' || !isDirty) return 'clean';
   285→
   286→      const missingField = def.fields.find((field) =>
   287→        isEmptyRequiredValue(field, draft[field.fieldKey]),
   288→      );
   289→      if (missingField) {
   290→        toast({
   291→          title: fieldLabel(missingField, locale),
   292→          description: t('formWindow.requiredHint'),
   293→          variant: 'destructive',
   294→        });
   295→        return 'invalid';
   296→      }
   297→
   298→      setIsSaving(true);
   299→      try {
   300→        const saved =
   301→          mode === 'new'
   302→            ? (
   303→                await apiFetch<{ data: FormRecord }>(`${basePath}/records`, {
   304→                  method: 'POST',
   305→                  body: { data: draft },
   306→                })
   307→              ).data
   308→            : (
   309→                await apiFetch<{ data: FormRecord }>(
   310→                  `${basePath}/records/${encodeURIComponent(current?.id ?? '')}`,
   311→                  { method: 'PUT', body: { data: draft } },
   312→                )
   313→              ).data;
   314→
   315→        const list = await refreshRecords();
   316→        if (options?.keepRecord !== false) {
   317→          const index = list.findIndex((record) => record.id === saved.id);
   318→          if (index >= 0) {
   319→            applyRecord(index, list);
   320→          } else {
   321→            setMode('view');
   322→          }
   323→        }
   324→        // Open records lists of this form refresh themselves.
   325→        if (moduleKey && formKey) {
   326→          notifyRecordsChanged({ moduleKey, formKey });
   327→        }
   328→        return 'saved';
   329→      } catch (error) {
   330→        if (!(error instanceof SessionExpiredError)) {
   331→          toast({
   332→            title: t('formWindow.saveError'),
   333→            description: error instanceof Error ? error.message : undefined,
   334→            variant: 'destructive',
   335→          });
   336→        }
   337→        return 'error';
   338→      } finally {
   339→        setIsSaving(false);
   340→      }
   341→    },
   342→    [
   343→      applyRecord,
   344→      basePath,
   345→      current,
   346→      def,
   347→      draft,
   348→      formKey,
   349→      isDeleting,
   350→      isDirty,
   351→      isSaving,
   352→      locale,
   353→      mode,
   354→      moduleKey,
   355→      refreshRecords,
   356→      t,
   357→    ],
   358→  );
   359→
   360→  const handleSave = useCallback(async () => {
   361→    const result = await performSave();
   362→    if (result === 'saved') toast({ title: t('formWindow.saveSuccess') });
   363→  }, [performSave, t]);
   364→
   365→  /** Save & new: write the record, then reset the form for a fresh entry. */
   366→  const handleSaveNew = useCallback(async () => {
   367→    if (isSaving || isDeleting) return;
   368→    const result = await performSave({ keepRecord: false });
   369→    if (result === 'error' || result === 'invalid') return;
   370→    startNew();
   371→    if (result === 'saved') toast({ title: t('formWindow.saveNewSuccess') });
   372→  }, [isDeleting, isSaving, performSave, startNew, t]);
   373→
   374→  /** Save & close: write the record after validation, then close. */
   375→  const handleSaveClose = useCallback(async () => {
   376→    if (isSaving || isDeleting) return;
   377→    const result = await performSave();
   378→    if (result === 'error' || result === 'invalid') return;
   379→    if (result === 'saved') toast({ title: t('formWindow.saveCloseSuccess') });
   380→    onClose?.();
   381→  }, [isDeleting, isSaving, onClose, performSave, t]);
   382→
   383→  /** Resets the shared delete-confirmation dialog. */
   384→  const cancelDelete = useCallback(() => {
   385→    setDeleteIntent(null);
   386→    setDeleteTargetId(null);
   387→  }, []);
   388→
   389→  /**
   390→   * First click on any delete action opens the confirm dialog. The
   391→   * target defaults to the loaded record; the tree context menu passes
   392→   * the right-clicked branch id explicitly.
   393→   */
   394→  const requestDelete = useCallback(
   395→    (intent: Exclude<DeleteIntent, null>, targetId?: string) => {
   396→      if (mode === 'new' || isSaving || isDeleting) return;
   397→      const target = targetId
   398→        ? records.find((record) => record.id === targetId) ?? null
   399→        : current;
   400→      if (!target) return;
   401→      setDeleteTargetId(target.id);
   402→      setDeleteIntent(intent);
   403→    },
   404→    [current, isDeleting, isSaving, mode, records],
   405→  );
   406→
   407→  /** Runs after the user presses «بلی» in the confirm dialog. */
   408→  const confirmDelete = useCallback(async () => {
   409→    if (isDeleting || !deleteIntent) return;
   410→    const intent = deleteIntent;
   411→    const targetId = deleteTargetId ?? current?.id ?? '';
   412→    if (!basePath || !targetId) {
   413→      cancelDelete();
   414→      return;
   415→    }
   416→    setIsDeleting(true);
   417→    try {
   418→      // DELETE answers 204 with an empty body — nothing to parse.
   419→      await apiFetch(
   420→        `${basePath}/records/${encodeURIComponent(targetId)}`,
   421→        {
   422→          method: 'DELETE',
   423→        },
   424→      );
   425→      cancelDelete();
   426→      // Open records lists of this form refresh themselves.
   427→      if (moduleKey && formKey) {
   428→        notifyRecordsChanged({ moduleKey, formKey });
   429→      }
   430→      if (intent === 'close') {
   431→        toast({ title: t('formWindow.deleteCloseSuccess') });
   432→        onClose?.();
   433→        return;
   434→      }
   435→      const list = await refreshRecords();
   436→      if (intent === 'new') {
   437→        startNew();
   438→        toast({ title: t('formWindow.deleteNewSuccess') });
   439→      } else if (list.length) {
   440→        applyRecord(Math.min(currentIndex, list.length - 1), list);
   441→        toast({ title: t('formWindow.deleteSuccess') });
   442→      } else {
   443→        setCurrentIndex(-1);
   444→        startNew();
   445→        toast({ title: t('formWindow.deleteSuccess') });
   446→      }
   447→    } catch (error) {
   448→      cancelDelete();
   449→      if (error instanceof SessionExpiredError) return;
   450→      toast({
   451→        title: error instanceof Error && error.message
   452→          ? error.message
   453→          : t('formWindow.loadError'),
   454→        variant: 'destructive',
   455→      });
   456→    } finally {
   457→      setIsDeleting(false);
   458→    }
   459→  }, [
   460→    applyRecord,
   461→    basePath,
   462→    cancelDelete,
   463→    currentIndex,
   464→    current,
   465→    deleteIntent,
   466→    deleteTargetId,
   467→    isDeleting,
   468→    onClose,
   469→    refreshRecords,
   470→    startNew,
   471→    t,
   472→  ]);
   473→
   474→  const handlePrint = useCallback(() => {
   475→    toast({ title: t('formWindow.printToast') });
   476→  }, [t]);
   477→
   478→  const handleFieldChange = useCallback(
   479→    (fieldKey: string, value: FormFieldValue) => {
   480→      setDraft((prev) => ({ ...prev, [fieldKey]: value }));
   481→      setMode((prev) => (prev === 'view' ? 'edit' : prev));
   482→    },
   483→    [],
   484→  );
   485→
   486→  // Ctrl+S / Cmd+S saves the active draft (Escape is handled by the dialog).
   487→  useEffect(() => {
   488→    const onKeyDown = (event: KeyboardEvent) => {
   489→      if (
   490→        (event.ctrlKey || event.metaKey) &&
   491→        !event.shiftKey &&
   492→        !event.altKey &&
   493→        event.key.toLowerCase() === 's'
   494→      ) {
   495→        event.preventDefault();
   496→        void handleSave();
   497→      }
   498→    };
   499→    window.addEventListener('keydown', onKeyDown);
   500→    return () => window.removeEventListener('keydown', onKeyDown);
   501→  }, [handleSave]);
   502→
   503→  // ── Derived view models ──
   504→  const sortedFields = useMemo(
   505→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   506→    [def],
   507→  );
   508→
   509→  const tabs = useMemo(() => (def ? deriveTabs(def, locale) : []), [def, locale]);
   510→
   511→  /**
   512→   * Two-part layout of tabbed forms: the first group is the header
   513→   * section (its fields are always visible, without any tab strip) and
   514→   * the remaining groups build the tabbed body section below it.
   515→   */
   516→  const isTabbed = def?.type === 'tabbed' && tabs.length > 0;
   517→  const headerTab = isTabbed ? (tabs[0] ?? null) : null;
   518→  const bodyTabs = isTabbed ? tabs.slice(1) : [];
   519→  const activeBodyTab =
   520→    bodyTabs.find((tab) => (tab.key ?? null) === (activeTabKey ?? null)) ??
   521→    bodyTabs[0] ??
   522→    null;
   523→
   524→  const listColumns = useMemo(
   525→    () => pickListColumns(sortedFields),
   526→    [sortedFields],
   527→  );
   528→
   529→  const title = def
   530→    ? formTitle(def, locale)
   531→    : form?.name ?? t('common.loading');
   532→  const subtitle =
   533→    mode === 'new'
   534→      ? t('formWindow.newMode')
   535→      : current
   536→        ? tf('formWindow.recordOf', {
   537→            n: formatNumber(currentIndex + 1),
   538→            total: formatNumber(records.length),
   539→          })
   540→        : undefined;
   541→
   542→  const accentColor = def?.accentColor || DEFAULT_FORM_ACCENT;
   543→
   544→  /** The fields grid shared by simple forms and the active tab. */
   545→  const renderFields = (fields: typeof sortedFields): ReactNode => (
   546→    <div className={fieldsGridClasses}>
   547→      {fields.map((field) => (
   548→        <FieldRenderer
   549→          key={field.fieldKey}
   550→          field={field}
   551→          value={draft[field.fieldKey] ?? null}
   552→          onChange={(value) => handleFieldChange(field.fieldKey, value)}
   553→          disabled={isSaving || isDeleting}
   554→          lookupUrlBase={lookupUrlBase}
   555→        />
   556→      ))}
   557→    </div>
   558→  );
   559→
   560→  let content: ReactNode;
   561→  if (!def) {
   562→    // Loading skeleton (error handled below).
   563→    content = (
   564→      <div className={fieldsGridClasses}>
   565→        {Array.from({ length: 6 }).map((_, index) => (
   566→          <div key={`form-skeleton-${index}`} className="space-y-1.5">
   567→            <Skeleton className="h-3 w-24 bg-white/[0.08]" />
   568→            <Skeleton className="h-6 w-full rounded-md bg-white/[0.05]" />
   569→          </div>
   570→        ))}
   571→      </div>
   572→    );
   573→  } else if (def.type === 'tree') {
   574→    content = (
   575→      <TreeForm
   576→        records={records}
   577→        fields={sortedFields}
   578→        draft={draft}
   579→        mode={mode}
   580→        selectedId={current?.id ?? null}
   581→        isBusy={isSaving || isDeleting}
   582→        accentColor={accentColor}
   583→        lookupUrlBase={lookupUrlBase}
   584→        onSelect={(recordId) => {
   585→          const index = records.findIndex((record) => record.id === recordId);
   586→          if (index >= 0) navigateTo(index);
   587→        }}
   588→        onFieldChange={handleFieldChange}
   589→        onNewChild={(preset) => {
   590→          if (
   591→            mode !== 'view' &&
   592→            isDirty &&
   593→            !window.confirm(t('formWindow.navDiscardConfirm'))
   594→          ) {
   595→            return;
   596→          }
   597→          startNew(preset);
   598→        }}
   599→        onDeleteNode={(recordId) => requestDelete('current', recordId)}
   600→        onCopyNode={(record) => {
   601→          if (
   602→            mode !== 'view' &&
   603→            isDirty &&
   604→            !window.confirm(t('formWindow.navDiscardConfirm'))
   605→          ) {
   606→            return;
   607→          }
   608→          startNew({ ...record.data });
   609→          toast({ title: t('formWindow.copyToast') });
   610→        }}
   611→      />
   612→    );
   613→  } else if (isTabbed && headerTab) {
   614→    // Header section (main fields, no tab strip) + tabbed body section.
   615→    content = (
   616→      <div className="flex min-h-0 flex-col">
   617→        <section
   618→          aria-label={headerTab.name}
   619→          className={cn(
   620→            'shrink-0 bg-white/[0.02]',
   621→            bodyTabs.length > 0 && 'border-b border-white/10',
   622→          )}
   623→        >
   624→          {renderFields(headerTab.fields)}
   625→        </section>
   626→        {bodyTabs.length > 0 && (
   627→          <section aria-label={activeBodyTab?.name ?? undefined}>
   628→            <FormTabs
   629→              tabs={bodyTabs.map((tab) => ({
   630→                key: tab.key,
   631→                name: tab.name,
   632→                count: tab.fields.length,
   633→              }))}
   634→              active={activeBodyTab?.key ?? null}
   635→              onChange={setActiveTabKey}
   636→              accentColor={accentColor}
   637→              direction={direction}
   638→            >
   639→              {renderFields(activeBodyTab?.fields ?? [])}
   640→            </FormTabs>
   641→          </section>
   642→        )}
   643→      </div>
   644→    );
   645→  } else {
   646→    content = renderFields(sortedFields);
   647→  }
   648→
   649→  const errorText =
   650→    loadError === GENERIC_LOAD_ERROR ? t('formWindow.loadError') : loadError;
   651→
   652→  return (
   653→    <>
   654→    <FormShell
   655→      title={title}
   656→      subtitle={subtitle}
   657→      iconKey={def?.iconKey ?? form?.iconKey ?? 'fileText'}
   658→      accentColor={accentColor}
   659→      actionBar={
   660→        <FormActionBar
   661→          mode={mode}
   662→          isDirty={isDirty}
   663→          isSaving={isSaving}
   664→          isDeleting={isDeleting}
   665→          hasRecord={Boolean(current)}
   666→          recordIndex={currentIndex}
   667→          recordCount={records.length}
   668→          isListOpen={isListOpen}
   669→          onNew={handleNew}
   670→          onSave={() => void handleSave()}
   671→          onSaveNew={() => void handleSaveNew()}
   672→          onSaveClose={() => void handleSaveClose()}
   673→          onCopy={handleCopy}
   674→          onDelete={() => requestDelete('current')}
   675→          onDeleteNew={() => requestDelete('new')}
   676→          onDeleteClose={() => requestDelete('close')}
   677→          onPrint={handlePrint}
   678→          onToggleList={() => setIsListOpen((open) => !open)}
   679→          onFirst={() => navigateTo(0)}
   680→          onPrev={() => navigateTo(Math.max(0, currentIndex - 1))}
   681→          onNext={() => navigateTo(Math.min(records.length - 1, currentIndex + 1))}
   682→          onLast={() => navigateTo(records.length - 1)}
   683→          hideNav={def?.type === 'tree'}
   684→        />
   685→      }
   686→      subPanel={
   687→        <AnimatePresence initial={false}>
   688→          {isListOpen && (
   689→            <RecordsDrawer
   690→              key="records-drawer"
   691→              records={records}
   692→              currentIndex={currentIndex}
   693→              codeField={listColumns.codeField}
   694→              titleField={listColumns.titleField}
   695→              onJump={navigateTo}
   696→              onClose={() => setIsListOpen(false)}
   697→            />
   698→          )}
   699→        </AnimatePresence>
   700→      }
   701→      footer={
   702→        <FormFooter
   703→          status={status}
   704→          position={
   705→            current && mode !== 'new'
   706→              ? { n: currentIndex + 1, total: records.length }
   707→              : null
   708→          }
   709→          updatedAt={current?.updatedAt ?? null}
   710→        />
   711→      }
   712→    >
   713→      {errorText && !def ? (
   714→        <div className="flex min-h-[220px] flex-col items-center justify-center gap-3 p-6 text-center">
   715→          <p className="max-w-sm text-sm text-red-200/90">{errorText}</p>
   716→          <button
   717→            type="button"
   718→            onClick={() => void load()}
   719→            className="flex items-center gap-2 rounded-lg bg-white/[0.07] px-4 py-2 text-[13px] text-white/80 transition-colors hover:bg-white/[0.12] focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-white/40"
   720→          >
   721→            <RefreshCw className="h-3.5 w-3.5" />
   722→            {t('common.retry')}
   723→          </button>
   724→        </div>
   725→      ) : (
   726→        <>
   727→          {errorText && def && (
   728→            <div className="mx-4 mt-4 rounded-lg border border-red-400/30 bg-red-950/40 px-3 py-2 text-xs text-red-200/90">
   729→              {errorText}
   730→            </div>
   731→          )}
   732→          {records.length === 0 && def && def.type !== 'tree' ? (
   733→            <div className="px-4 pb-2 pt-6 text-center">
   734→              <p className="text-[13px] text-white/55">
   735→                {t('formWindow.emptyRecords')}
   736→              </p>
   737→              <p className="mt-1 text-[11px] text-white/35">
   738→                {t('formWindow.emptyRecordsHint')}
   739→              </p>
   740→            </div>
   741→          ) : null}
   742→          {content}
   743→        </>
   744→      )}
   745→    </FormShell>
   746→
   747→    {/* Single delete-confirmation dialog shared by all three delete actions. */}
   748→    <AlertDialog
   749→      open={deleteIntent !== null}
   750→      onOpenChange={(open) => {
   751→        if (!open) cancelDelete();
   752→      }}
   753→    >
   754→      <AlertDialogContent className="max-w-[calc(100vw-2rem)] rounded-2xl border border-white/15 bg-[#1e2128]/95 text-white shadow-2xl shadow-black/60 backdrop-blur-xl sm:max-w-md">
   755→        <AlertDialogHeader className="sm:text-start">
   756→          <AlertDialogTitle className="flex items-center gap-2.5 text-base text-white">
   757→            <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-rose-500/15 text-rose-400 ring-1 ring-rose-400/30">
   758→              <Trash2 className="h-4 w-4" aria-hidden="true" />
   759→            </span>
   760→            {t('formWindow.deleteDialogTitle')}
   761→          </AlertDialogTitle>
   762→          <AlertDialogDescription className="text-[13px] leading-relaxed text-white/65">
   763→            {t('formWindow.deleteDialogMessage')}
   764→          </AlertDialogDescription>
   765→        </AlertDialogHeader>
   766→        <AlertDialogFooter className="gap-2 sm:justify-start">
   767→          <AlertDialogCancel
   768→            disabled={isDeleting}
   769→            className="h-9 rounded-lg border-white/15 bg-white/[0.06] px-4 text-[13px] text-white/80 transition-colors hover:bg-white/[0.12] hover:text-white focus-visible:ring-white/40"
   770→          >
   771→            {t('formWindow.confirmNo')}
   772→          </AlertDialogCancel>
   773→          <AlertDialogAction
   774→            disabled={isDeleting}
   775→            onClick={(event) => {
   776→              // Keep the dialog open while the delete request runs.
   777→              event.preventDefault();
   778→              void confirmDelete();
   779→            }}
   780→            className="flex h-9 items-center gap-1.5 rounded-lg bg-rose-500/90 px-4 text-[13px] text-white shadow-sm shadow-rose-950/40 transition-colors hover:bg-rose-500 focus-visible:ring-rose-300/60"
   781→          >
   782→            {isDeleting ? (
   783→              <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" />
   784→            ) : null}
   785→            {t('formWindow.confirmYes')}
   786→          </AlertDialogAction>
   787→        </AlertDialogFooter>
   788→      </AlertDialogContent>
   789→    </AlertDialog>
   790→    </>
   791→  );
   792→}
