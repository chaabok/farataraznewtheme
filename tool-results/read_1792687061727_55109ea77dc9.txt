    39→interface TreeFormProps {
    40→  records: FormRecord[];
    41→  fields: FormFieldDef[];
    42→  /** Draft values of the loaded / new entry (owned by the form window). */
    43→  draft: FormDataMap;
    44→  mode: FormMode;
    45→  /** Id of the record loaded in the detail box (null in new mode). */
    46→  selectedId: string | null;
    47→  /** True while a save/delete request is in flight. */
    48→  isBusy: boolean;
    49→  /** Accent hex color from the FormDef. */
    50→  accentColor: string;
    51→  /** Base path ending in `/lookups` for lookup fields. */
    52→  lookupUrlBase: string;
    53→  /** Clicking a tree node loads it into the detail box. */
    54→  onSelect: (recordId: string) => void;
    55→  onFieldChange: (fieldKey: string, value: FormFieldValue) => void;
    56→  /**
    57→   * Tree toolbar «جدید» — receives a preset map assigning the selected
    58→   * node as parent (empty when nothing is selected → root entry).
    59→   */
    60→  onNewChild: (preset: FormDataMap) => void;
    61→  /** Tree toolbar «تغییر» — puts the loaded record into edit mode. */
    62→  onEditNode: () => void;
    63→  /** Tree toolbar «حذف» — opens the shared delete confirmation. */
    64→  onDeleteNode: () => void;
    65→  /** Per-node copy icon — starts a new entry prefilled with that node. */
    66→  onCopyNode: (record: FormRecord) => void;
    67→}
    68→
    69→interface TreeNode {
    70→  record: FormRecord;
    71→  depth: number;
    72→  children: TreeNode[];
    73→}
    74→
    75→/**
    76→ * Two-box renderer of `type: 'tree'` ERP forms (e.g. «گروه کالا-خدمت»):
    77→ *
    78→ * - Tree box (درختواره): toolbar (جدید / تغییر / حذف), a search field and
    79→ *   the expandable hierarchy built from the records' parent field, with
    80→ *   code badges, per-node copy and folder/file glyphs.
    81→ * - Detail box: the header fields of the selected record (the parent
    82→ *   field renders as a self-referencing picker), plus the tabbed body
    83→ *   sections — sharing the exact same field components as every form.
    84→ *
    85→ * Purely presentational: the record state machine (save/delete/navigate)
    86→ * lives in the hosting DataFormWindow, so the whole CRUD surface —
    87→ * action bar, confirm dialog, status footer — stays identical.
    88→ */
    89→export function TreeForm({
    90→  records,
    91→  fields,
    92→  draft,
    93→  mode,
    94→  selectedId,
    95→  isBusy,
    96→  accentColor,
    97→  lookupUrlBase,
    98→  onSelect,
    99→  onFieldChange,
   100→  onNewChild,
   101→  onEditNode,
   102→  onDeleteNode,
   103→  onCopyNode,
   104→}: TreeFormProps) {
   105→  const { direction, formatNumber, t } = useTranslation();
   106→
   107→  const sortedFields = useMemo(
   108→    () => [...fields].sort((a, b) => a.sortOrder - b.sortOrder),
   109→    [fields],
   110→  );
   111→  const parentField = useMemo(() => pickParentField(sortedFields), [sortedFields]);
   112→  const { codeField, titleField } = useMemo(
   113→    () => pickListColumns(sortedFields),
   114→    [sortedFields],
   115→  );
   116→
   117→  // ── Hierarchy ────────────────────────────────────────────────────
   118→  const nodes = useMemo<TreeNode[]>(() => {
   119→    const byId = new Map<string, TreeNode>(
   120→      records.map((record) => [record.id, { record, depth: 0, children: [] }]),
   121→    );
   122→    const roots: TreeNode[] = [];
   123→    for (const node of byId.values()) {
   124→      const parentValue = parentField
   125→        ? node.record.data[parentField.fieldKey]
   126→        : null;
   127→      const parent =
   128→        typeof parentValue === 'string' ? byId.get(parentValue) : undefined;
   129→      if (parent && parent !== node) parent.children.push(node);
   130→      else roots.push(node);
   131→    }
   132→    const flat: TreeNode[] = [];
   133→    const walk = (list: TreeNode[], depth: number) => {
   134→      for (const node of list) {
   135→        node.depth = depth;
   136→        flat.push(node);
   137→        walk(node.children, depth + 1);
   138→      }
   139→    };
   140→    walk(roots, 0);
   141→    return flat;
   142→  }, [records, parentField]);
   143→
   144→  /** Ids of the selected record's descendants (excluded from parent picker). */
   145→  const descendantIds = useMemo(() => {
   146→    const ids = new Set<string>();
   147→    if (!selectedId) return ids;
   148→    const walk = (list: TreeNode[]): boolean => {
   149→      for (const node of list) {
   150→        if (walk(node.children)) return true;
   151→        if (node.record.id === selectedId) {
   152→          ids.add(node.record.id);
   153→          return true;
   154→        }
   155→        if (ids.size) ids.add(node.record.id);
   156→      }
   157→      return false;
   158→    };
   159→    walk(nodes);
   160→    return ids;
   161→  }, [nodes, selectedId]);
   162→
   163→  // ── Search + expand state ───────────────────────────────────────
   164→  const [query, setQuery] = useState('');
   165→  const [collapsed, setCollapsed] = useState<ReadonlySet<string>>(new Set());
   166→
   167→  const needle = query.trim().toLowerCase();
   168→  const isSearching = needle.length > 0;
   169→
   170→  /** Node matches when its code or title contains the query. */
   171→  const nodeMatches = useCallback(
   172→    (node: TreeNode) => {
   173→      const code = codeField
   174→        ? String(node.record.data[codeField.fieldKey] ?? '')
   175→        : '';
   176→      const title = titleField
   177→        ? String(node.record.data[titleField.fieldKey] ?? '')
   178→        : '';
   179→      return (
   180→        code.toLowerCase().includes(needle) ||
   181→        title.toLowerCase().includes(needle)
   182→      );
   183→    },
   184→    [codeField, needle, titleField],
   185→  );
   186→
   187→  /** Visible nodes while searching: matches + their ancestors. */
   188→  const visibleNodes = useMemo(() => {
   189→    if (!isSearching) return nodes;
   190→    const keep = new Set<string>();
   191→    const match = (list: TreeNode[]): boolean => {
   192→      let any = false;
   193→      for (const node of list) {
   194→        const childMatch = match(node.children);
   195→        if (childMatch || nodeMatches(node)) {
   196→          keep.add(node.record.id);
   197→          any = true;
   198→        }
   199→      }
   200→      return any;
   201→    };
   202→    match(nodes);
   203→    return nodes.filter((node) => keep.has(node.record.id));
   204→  }, [isSearching, nodeMatches, nodes]);
   205→
   206→  const toggle = useCallback((recordId: string) => {
   207→    setCollapsed((prev) => {
   208→      const next = new Set(prev);
   209→      if (next.has(recordId)) next.delete(recordId);
   210→      else next.add(recordId);
   211→      return next;
   212→    });
   213→  }, []);
   214→
   215→  // ── Body tabs (tabKey !== null groups, definition order kept) ───
   216→  const bodyTabs = useMemo(() => {
   217→    const groups: Array<{ key: string; name: string; fields: FormFieldDef[] }> = [];
   218→    for (const field of sortedFields) {
   219→      if (field.tabKey === null) continue;
   220→      let group = groups.find((candidate) => candidate.key === field.tabKey);
   221→      if (!group) {
   222→        group = {
   223→          key: field.tabKey,
   224→          name: field.tabName ?? field.tabKey,
   225→          fields: [],
   226→        };
   227→        groups.push(group);
   228→      }
   229→      group.fields.push(field);
   230→    }
   231→    return groups;
   232→  }, [sortedFields]);
   233→
   234→  const headerFields = useMemo(
   235→    () => sortedFields.filter((field) => field.tabKey === null),
   236→    [sortedFields],
   237→  );
   238→
   239→  const [activeTabKey, setActiveTabKey] = useState<string | null>(null);
   240→  const activeBodyTab =
   241→    bodyTabs.find((tab) => tab.key === activeTabKey) ?? bodyTabs[0] ?? null;
   242→
   243→  const current = records.find((record) => record.id === selectedId) ?? null;
   244→  const showDetail = mode === 'new' || current !== null;
   245→
   246→  const renderFields = (list: FormFieldDef[]) => (
   247→    <div className="grid grid-cols-1 gap-x-5 gap-y-4 p-4 sm:grid-cols-2">
   248→      {list.map((field) =>
   249→        parentField && field.fieldKey === parentField.fieldKey ? (
   250→          <ParentPicker
   251→            key={field.fieldKey}
   252→            field={field}
   253→            nodes={nodes}
   254→            excludeIds={descendantIds}
   255→            value={
   256→              typeof draft[field.fieldKey] === 'string'
   257→                ? (draft[field.fieldKey] as string)
   258→                : null
   259→            }
   260→            onChange={(value) => onFieldChange(field.fieldKey, value)}
   261→            disabled={isBusy}
   262→          />
   263→        ) : (
   264→          <FieldRenderer
   265→            key={field.fieldKey}
   266→            field={field}
   267→            value={draft[field.fieldKey] ?? null}
   268→            onChange={(value) => onFieldChange(field.fieldKey, value)}
   269→            disabled={isBusy}
   270→            lookupUrlBase={lookupUrlBase}
   271→          />
   272→        ),
   273→      )}
   274→    </div>
   275→  );
   276→
   277→  return (
   278→    <div className="flex h-full min-h-0 flex-col md:flex-row">
   279→      {/* ── Box 1: tree panel (درختواره) ─────────────────────────── */}
   280→      <aside
   281→        aria-label={t('formWindow.treePanel')}
   282→        className="flex max-h-56 w-full shrink-0 flex-col border-b border-white/10 bg-white/[0.02] md:max-h-none md:w-60 md:border-b-0 md:border-e lg:w-72"
   283→      >
   284→        <div className="shrink-0 border-b border-white/10 px-3 pb-2.5 pt-3">
   285→          {/* Panel title + record count */}
   286→          <div className="mb-2.5 flex items-center gap-1.5">
   287→            <Network className="h-3.5 w-3.5 text-white/45" aria-hidden="true" />
   288→            <span className="text-[12px] font-semibold text-white/85">
   289→              {t('formWindow.treePanel')}
   290→            </span>
   291→            <span className="rounded bg-white/[0.07] px-1.5 py-px text-[10px] tabular-nums text-white/45">
   292→              {formatNumber(records.length)}
   293→            </span>
   294→          </div>
   295→
   296→          {/* Toolbar: جدید / تغییر / حذف */}
   297→          <div className="mb-2.5 flex items-center gap-1.5">
   298→            <button
   299→              type="button"
   300→              title={t('formWindow.nodeNew')}
   301→              onClick={() =>
   302→                onNewChild(
   303→                  selectedId && parentField
   304→                    ? { [parentField.fieldKey]: selectedId }
   305→                    : {},
   306→                )
   307→              }
   308→              disabled={isBusy}
   309→              className="flex h-7 items-center gap-1 rounded-md bg-emerald-500/15 px-2.5 text-[11px] font-medium text-emerald-400 ring-1 ring-white/10 transition-all duration-150 hover:bg-emerald-500/25 hover:shadow-[0_0_14px_-4px_rgba(52,211,153,0.55)] active:scale-95 disabled:pointer-events-none disabled:opacity-40 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-400/50"
   310→            >
   311→              <Plus className="h-3 w-3" strokeWidth={2.4} />
   312→              {t('formWindow.nodeNew')}
   313→            </button>
   314→            <button
   315→              type="button"
   316→              title={t('formWindow.nodeEdit')}
   317→              onClick={onEditNode}
   318→              disabled={!current || mode === 'new' || isBusy}
   319→              className="flex h-7 items-center gap-1 rounded-md bg-sky-500/15 px-2.5 text-[11px] font-medium text-sky-400 ring-1 ring-white/10 transition-all duration-150 hover:bg-sky-500/25 hover:shadow-[0_0_14px_-4px_rgba(56,189,248,0.55)] active:scale-95 disabled:pointer-events-none disabled:opacity-40 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-sky-400/50"
   320→            >
   321→              <Pencil className="h-3 w-3" strokeWidth={2.2} />
   322→              {t('formWindow.nodeEdit')}
   323→            </button>
   324→            <button
   325→              type="button"
   326→              title={t('formWindow.nodeDelete')}
   327→              onClick={onDeleteNode}
   328→              disabled={!current || mode === 'new' || isBusy}
   329→              className="flex h-7 items-center gap-1 rounded-md bg-rose-500/15 px-2.5 text-[11px] font-medium text-rose-400 ring-1 ring-white/10 transition-all duration-150 hover:bg-rose-500/25 hover:shadow-[0_0_14px_-4px_rgba(251,113,133,0.55)] active:scale-95 disabled:pointer-events-none disabled:opacity-40 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-rose-400/50"
   330→            >
   331→              <Trash2 className="h-3 w-3" strokeWidth={2.2} />
   332→              {t('formWindow.nodeDelete')}
   333→            </button>
   334→          </div>
   335→
   336→          {/* Search inside the tree */}
   337→          <div className="relative">
   338→            <Search
   339→              className="pointer-events-none absolute start-2 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-white/35"
   340→              aria-hidden="true"
   341→            />
   342→            <input
   343→              type="text"
   344→              value={query}
   345→              onChange={(event) => setQuery(event.target.value)}
   346→              placeholder={t('formWindow.treeSearch')}
   347→              aria-label={t('formWindow.treeSearch')}
   348→              className={cn(controlClasses, 'h-7 ps-7 pe-7')}
   349→            />
   350→            {isSearching && (
   351→              <button
   352→                type="button"
   353→                title={t('formWindow.clear')}
   354→                aria-label={t('formWindow.clear')}
   355→                onClick={() => setQuery('')}
   356→                className="absolute end-1 top-1/2 flex h-5 w-5 -translate-y-1/2 items-center justify-center rounded-md text-white/40 transition-colors hover:bg-white/10 hover:text-white focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-white/40"
   357→              >
   358→                <X className="h-3 w-3" />
   359→              </button>
   360→            )}
   361→          </div>
   362→        </div>
   363→
   364→        {/* Nodes */}
   365→        <div
   366→          role="tree"
   367→          aria-label={t('formWindow.treePanel')}
   368→          className="min-h-0 flex-1 overflow-y-auto p-1.5 [scrollbar-width:thin]"
   369→        >
   370→          {nodes.length === 0 ? (
   371→            <div className="flex flex-col items-center gap-2 px-3 py-8 text-center">
   372→              <Network className="h-5 w-5 text-white/25" aria-hidden="true" />
   373→              <p className="text-[12px] text-white/40">
   374→                {t('formWindow.emptyRecords')}
   375→              </p>
   376→            </div>
   377→          ) : visibleNodes.length === 0 ? (
   378→            <p className="px-3 py-8 text-center text-[12px] text-white/40">
   379→              {t('formWindow.noMatch')}
   380→            </p>
   381→          ) : (
   382→            visibleNodes.map((node) => {
   383→              const hasChildren = node.children.length > 0;
   384→              const isCollapsed = !isSearching && collapsed.has(node.record.id);
   385→              const isSelected =
   386→                mode !== 'new' && node.record.id === selectedId;
   387→              return (
   388→                <div
   389→                  key={node.record.id}
   390→                  role="treeitem"
   391→                  aria-selected={isSelected}
   392→                  aria-level={node.depth + 1}
   393→                  aria-expanded={hasChildren ? !isCollapsed : undefined}
   394→                  style={{ paddingInlineStart: node.depth * 14 }}
   395→                >
   396→                  <div
   397→                    className={cn(
   398→                      'group flex items-center gap-0.5 rounded-lg pe-1 transition-colors',
   399→                      isSelected
   400→                        ? 'bg-sky-500/15 ring-1 ring-sky-400/30'
   401→                        : 'hover:bg-white/[0.05]',
   402→                    )}
   403→                  >
   404→                    <button
   405→                      type="button"
   406→                      onClick={() => onSelect(node.record.id)}
   407→                      className="flex min-w-0 flex-1 items-center gap-1.5 py-1.5 ps-1.5 text-start"
   408→                    >
   409→                      {hasChildren ? (
   410→                        <Folder
   411→                          className="h-3.5 w-3.5 shrink-0 text-amber-400/80"
   412→                          aria-hidden="true"
   413→                        />
   414→                      ) : (
   415→                        <FileText
   416→                          className="h-3.5 w-3.5 shrink-0 text-white/40"
   417→                          aria-hidden="true"
   418→                        />
   419→                      )}
   420→                      <span className="min-w-0 flex-1 truncate text-[12px] text-white/85">
   421→                        {titleField
   422→                          ? formatFieldValue(
   423→                              titleField,
   424→                              node.record.data[titleField.fieldKey],
   425→                              formatNumber,
   426→                            )
   427→                          : node.record.id}
   428→                      </span>
   429→                      {codeField && (
   430→                        <span className="shrink-0 rounded bg-white/[0.07] px-1.5 py-0.5 font-mono text-[10px] text-white/60">
   431→                          {formatFieldValue(
   432→                            codeField,
   433→                            node.record.data[codeField.fieldKey],
   434→                            formatNumber,
   435→                          )}
   436→                        </span>
   437→                      )}
   438→                    </button>
   439→                    <button
   440→                      type="button"
   441→                      title={t('formWindow.actionCopy')}
   442→                      aria-label={t('formWindow.actionCopy')}
   443→                      onClick={() => onCopyNode(node.record)}
   444→                      className="flex h-5 w-5 shrink-0 items-center justify-center rounded-md text-white/35 transition-colors hover:bg-white/10 hover:text-white focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-white/40"
   445→                    >
   446→                      <Copy className="h-3 w-3" />
   447→                    </button>
   448→                    <button
   449→                      type="button"
   450→                      aria-label={
   451→                        hasChildren ? t('formWindow.treePanel') : undefined
   452→                      }
   453→                      disabled={!hasChildren}
   454→                      onClick={() => toggle(node.record.id)}
   455→                      className={cn(
   456→                        'flex h-5 w-5 shrink-0 items-center justify-center rounded-md text-white/45 transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-white/40',
   457→                        hasChildren
   458→                          ? 'hover:bg-white/10 hover:text-white'
   459→                          : 'pointer-events-none opacity-0',
   460→                      )}
   461→                    >
   462→                      <ChevronRight
   463→                        className={cn(
   464→                          'h-3.5 w-3.5 transition-transform rtl:-scale-x-100',
   465→                          !isCollapsed && 'rotate-90 rtl:-rotate-90',
   466→                        )}
   467→                      />
   468→                    </button>
   469→                  </div>
   470→                </div>
   471→              );
   472→            })
   473→          )}
   474→        </div>
   475→      </aside>
   476→
   477→      {/* ── Box 2: detail (fields + tabs) ────────────────────────── */}
   478→      <section
   479→        aria-label={t('formWindow.mainSection')}
   480→        className="flex min-h-0 min-w-0 flex-1 flex-col overflow-y-auto [scrollbar-width:thin]"
   481→      >
   482→        {showDetail ? (
   483→          <>
   484→            <div className="bg-white/[0.02]">
   485→              {renderFields(headerFields)}
   486→            </div>
   487→            {bodyTabs.length > 0 && (
   488→              <div className="flex min-h-0 flex-col">
   489→                <FormTabs
   490→                  tabs={bodyTabs.map((tab) => ({
   491→                    key: tab.key,
   492→                    name: tab.name,
   493→                    count: tab.fields.length,
   494→                  }))}
   495→                  active={activeBodyTab?.key ?? null}
   496→                  onChange={setActiveTabKey}
   497→                  accentColor={accentColor}
   498→                  direction={direction}
   499→                >
   500→                  {renderFields(activeBodyTab?.fields ?? [])}
   501→                </FormTabs>
   502→              </div>
   503→            )}
   504→          </>
   505→        ) : records.length === 0 ? (
   506→          <div className="flex flex-1 flex-col items-center justify-center gap-2 p-6 text-center">
   507→            <Network className="h-6 w-6 text-white/25" aria-hidden="true" />
   508→            <p className="text-[13px] text-white/55">
   509→              {t('formWindow.emptyRecords')}
   510→            </p>
   511→            <p className="text-[11px] text-white/35">
   512→              {t('formWindow.emptyRecordsHint')}
   513→            </p>
   514→          </div>
   515→        ) : (
   516→          <div className="flex flex-1 flex-col items-center justify-center gap-2 p-6 text-center">
   517→            <Network className="h-6 w-6 text-white/25" aria-hidden="true" />
   518→            <p className="text-[12.5px] text-white/45">
   519→              {t('formWindow.treeSelectHint')}
   520→            </p>
   521→          </div>
   522→        )}
   523→      </section>
   524→    </div>
   525→  );
   526→}
   527→
   528→interface ParentPickerProps {
   529→  field: FormFieldDef;
   530→  nodes: TreeNode[];
   531→  /** Record ids that may not become a parent (self + descendants). */
   532→  excludeIds: ReadonlySet<string>;
   533→  value: string | null;
   534→  onChange: (value: FormFieldValue) => void;
   535→  disabled?: boolean;
   536→}
   537→
   538→/**
   539→ * Self-referencing parent selector of tree forms: lists the form's own
   540→ * records in hierarchy order (indented, code + title) and stores the
   541→ * chosen record's id — the same value the tree hierarchy is built from.
   542→ */
   543→function ParentPicker({
   544→  field,
   545→  nodes,
   546→  excludeIds,
   547→  value,
   548→  onChange,
   549→  disabled,
   550→}: ParentPickerProps) {
   551→  const { locale, t } = useTranslation();
   552→  const listboxId = useId();
   553→  const [open, setOpen] = useState(false);
   554→  const [query, setQuery] = useState('');
   555→  const [activeIndex, setActiveIndex] = useState(0);
   556→  const rootRef = useDismiss<HTMLDivElement>(
   557→    open,
   558→    useCallback(() => setOpen(false), []),
   559→  );
   560→
   561→  const options = useMemo(
   562→    () =>
   563→      nodes
   564→        .filter((node) => !excludeIds.has(node.record.id))
   565→        .map((node) => ({ node, depth: node.depth })),
   566→    [excludeIds, nodes],
   567→  );
   568→
   569→  const needle = query.trim().toLowerCase();
   570→  const filtered = useMemo(() => {
   571→    if (!needle) return options;
   572→    return options.filter(({ node }) => {
   573→      const title = String(node.record.data.title ?? '');
   574→      const code = String(node.record.data.code ?? '');
   575→      return (
   576→        title.toLowerCase().includes(needle) ||
   577→        code.toLowerCase().includes(needle)
   578→      );
   579→    });
   580→  }, [needle, options]);
   581→
   582→  const labelOf = useCallback(
   583→    (node: TreeNode) => {
   584→      const title = String(node.record.data.title ?? '').trim();
   585→      const code = String(node.record.data.code ?? '').trim();
   586→      return code && title ? `${code} — ${title}` : title || code || node.record.id;
   587→    },
   588→    [],
   589→  );
   590→
   591→  const selectedNode = useMemo(
   592→    () => nodes.find((node) => node.record.id === value) ?? null,
   593→    [nodes, value],
   594→  );
   595→
   596→  const openPanel = () => {
   597→    setQuery('');
   598→    setActiveIndex(0);
   599→    setOpen(true);
   600→  };
   601→
   602→  const pick = (node: TreeNode) => {
   603→    onChange(node.record.id);
   604→    setOpen(false);
   605→  };
   606→
   607→  const onKeyDown = (event: React.KeyboardEvent) => {
   608→    if (!open) {
   609→      if (event.key === 'ArrowDown' || event.key === 'Enter') {
   610→        event.preventDefault();
   611→        openPanel();
   612→      }
   613→      return;
   614→    }
   615→    if (event.key === 'ArrowDown') {
   616→      event.preventDefault();
   617→      setActiveIndex((index) => Math.min(filtered.length - 1, index + 1));
   618→    } else if (event.key === 'ArrowUp') {
   619→      event.preventDefault();
   620→      setActiveIndex((index) => Math.max(0, index - 1));
   621→    } else if (event.key === 'Enter') {
   622→      event.preventDefault();
   623→      const target = filtered[activeIndex];
   624→      if (target) pick(target.node);
   625→    } else if (event.key === 'Escape') {
   626→      event.stopPropagation();
   627→      setOpen(false);
   628→    }
   629→  };
   630→
   631→  return (
   632→    <div
   633→      className={cn('relative', field.span === 2 && 'col-span-full')}
   634→      ref={rootRef}
   635→      onKeyDown={onKeyDown}
   636→    >
   637→      <div className="mb-1 flex items-baseline gap-1">
   638→        <label
   639→          htmlFor={field.fieldKey}
   640→          className="text-[13px] leading-none text-white/70"
   641→        >
   642→          {fieldLabel(field, locale)}
   643→        </label>
   644→        {field.isRequired && (
   645→          <span
   646→            title={t('formWindow.requiredHint')}
   647→            aria-hidden="true"
   648→            className="text-[13px] leading-none text-red-400"
   649→          >
   650→            *
   651→          </span>
   652→        )}
   653→      </div>
   654→      <div className="flex items-center gap-1.5">
   655→        <button
   656→          type="button"
   657→          id={field.fieldKey}
   658→          role="combobox"
   659→          aria-expanded={open}
   660→          aria-controls={listboxId}
   661→          aria-haspopup="listbox"
   662→          disabled={disabled}
   663→          onClick={() => (open ? setOpen(false) : openPanel())}
   664→          className={cn(
   665→            controlClasses,
   666→            'flex items-center justify-between gap-2 pe-2 text-start',
   667→          )}
   668→        >
   669→          <span
   670→            className={cn(
   671→              'min-w-0 flex-1 truncate',
   672→              selectedNode ? 'text-white' : 'text-white/30',
   673→            )}
   674→          >
   675→            {selectedNode
   676→              ? labelOf(selectedNode)
   677→              : t('formWindow.selectPlaceholder')}
   678→          </span>
   679→          <ChevronsUpDown className="h-3.5 w-3.5 shrink-0 text-white/35" />
   680→        </button>
   681→      </div>
   682→
   683→      <AnimatePresence>
   684→        {open && (
   685→          <motion.div
   686→            initial={{ opacity: 0, scale: 0.97, y: -4 }}
   687→            animate={{ opacity: 1, scale: 1, y: 0 }}
   688→            exit={{ opacity: 0, scale: 0.97, y: -4 }}
   689→            transition={{ duration: 0.14, ease: 'easeOut' }}
   690→            className="absolute top-full z-50 mt-1.5 w-full min-w-[220px] overflow-hidden rounded-xl bg-[#1b1e26]/[0.97] shadow-2xl shadow-black/50 ring-1 ring-white/10 backdrop-blur-xl"
   691→          >
   692→            <div className="flex items-center gap-2 border-b border-white/10 px-3 py-2">
   693→              <Search className="h-3.5 w-3.5 shrink-0 text-white/35" />
   694→              <input
   695→                autoFocus
   696→                type="text"
   697→                value={query}
   698→                onChange={(event) => {
   699→                  setQuery(event.target.value);
   700→                  setActiveIndex(0);
   701→                }}
   702→                placeholder={t('formWindow.search')}
   703→                className="h-7 w-full bg-transparent text-sm text-white placeholder:text-white/30 focus:outline-none"
   704→              />
   705→            </div>
   706→            <div
   707→              id={listboxId}
   708→              role="listbox"
   709→              aria-label={fieldLabel(field, locale)}
   710→              className="max-h-64 overflow-y-auto py-1 [scrollbar-width:thin]"
   711→            >
   712→              {filtered.length === 0 && (
   713→                <p className="px-3 py-4 text-center text-xs text-white/40">
   714→                  {t('formWindow.noMatch')}
   715→                </p>
   716→              )}
   717→              {filtered.map(({ node }, index) => {
   718→                const isSelected = node.record.id === value;
   719→                const isActive = index === activeIndex;
   720→                return (
   721→                  <button
   722→                    key={node.record.id}
   723→                    type="button"
   724→                    role="option"
   725→                    aria-selected={isSelected}
   726→                    onMouseEnter={() => setActiveIndex(index)}
   727→                    onClick={() => pick(node)}
   728→                    style={{ paddingInlineStart: 12 + node.depth * 14 }}
   729→                    className={cn(
   730→                      'flex w-full items-center gap-2 pe-3 py-1.5 text-start text-[12.5px] transition-colors',
   731→                      isActive
   732→                        ? 'bg-sky-500/15 text-white'
   733→                        : 'text-white/80 hover:bg-white/[0.05]',
   734→                      isSelected && !isActive && 'bg-white/[0.04]',
   735→                    )}
   736→                  >
   737→                    <span className="min-w-0 flex-1 truncate">
   738→                      {labelOf(node)}
   739→                    </span>
   740→                    {isSelected && (
   741→                      <Check className="h-3.5 w-3.5 shrink-0 text-sky-400" />
   742→                    )}
   743→                  </button>
   744→                );
   745→              })}
   746→            </div>
   747→            {value !== null && value !== '' && (
   748→              <div className="border-t border-white/10 p-1.5">
   749→                <button
   750→                  type="button"
   751→                  onClick={() => {
   752→                    onChange(null);
   753→                    setOpen(false);
   754→                  }}
   755→                  className="flex w-full items-center gap-2 rounded-lg px-2.5 py-1.5 text-xs text-rose-300 transition-colors hover:bg-rose-500/15 focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-rose-400/50"
   756→                >
   757→                  <X className="h-3.5 w-3.5" />
   758→                  {t('formWindow.clear')}
   759→                </button>
   760→              </div>
   761→            )}
   762→          </motion.div>
   763→        )}
   764→      </AnimatePresence>
   765→    </div>
   766→  );
   767→}
   768→
