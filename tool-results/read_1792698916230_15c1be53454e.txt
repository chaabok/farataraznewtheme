   101→   101→/**
   102→   102→ * Read the rendered width of every data column from its header cell.
   103→   103→ * Used to freeze the whole grid the first time the user grabs a handle,
   104→   104→ * so resizing one column never reflows its neighbours.
   105→   105→ */
   106→   106→function measureColumns(table: HTMLTableElement | null): Record<string, number> | null {
   107→   107→  if (!table) return null;
   108→   108→  const measured: Record<string, number> = {};
   109→   109→  table
   110→   110→    .querySelectorAll<HTMLTableCellElement>('th[data-col]')
   111→   111→    .forEach((th) => {
   112→   112→      const key = th.dataset.col;
   113→   113→      if (key) measured[key] = th.getBoundingClientRect().width;
   114→   114→    });
   115→   115→  return Object.keys(measured).length ? measured : null;
   116→   116→}

   192→   192→export function DataListWindow({ moduleId, list, onClose }: DataListWindowProps) {
   193→   193→  const { locale, t, tf, formatNumber, direction } = useTranslation();

   195→   195→
   196→   196→  const fallback = useMemo(() => parseListModuleId(moduleId), [moduleId]);
   197→   197→  const moduleKey = list?.moduleKey ?? fallback.moduleKey;
   198→   198→  const formKey = list?.formKey ?? fallback.formKey;
   199→   199→  const basePath =
   200→   200→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
   201→   201→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   202→   202→
   203→   203→  // ── Data ──
   204→   204→  const [def, setDef] = useState<FormDef | null>(null);
   205→   205→  /** Full dataset of the form (parent-title context + total count). */
   206→   206→  const [allRecords, setAllRecords] = useState<FormRecord[]>([]);
   207→   207→  /** Rows currently displayed — the server-filtered slice. */
   208→   208→  const [records, setRecords] = useState<FormRecord[]>([]);
   209→   209→  const [isLoading, setIsLoading] = useState(true);
   210→   210→  const [isRefreshing, setIsRefreshing] = useState(false);
   211→   211→  /** True while a debounced server-side filter query is in flight. */
   212→   212→  const [isFiltering, setIsFiltering] = useState(false);
   213→   213→  const [loadError, setLoadError] = useState<string | null>(null);
   214→   214→  /** lookupKey → (option code → label). */
   215→   215→  const [lookupLabels, setLookupLabels] = useState<
   216→   216→    Record<string, Map<string, string>>
   217→   217→  >({});
   218→   218→
   219→   219→  // ── View state ──
   220→   220→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   221→   221→  const [filters, setFilters] = useState<Record<string, ColumnFilterState>>({});
   222→   222→  const [isFilterRowVisible, setIsFilterRowVisible] = useState(true);
   223→   223→  /** Open operator menu, anchored to the funnel button of one column. */
   224→   224→  const [filterMenu, setFilterMenu] = useState<{
   225→   225→    fieldKey: string;
   226→   226→    rect: DOMRect;
   227→   227→    left: number;
   228→   228→    openUp: boolean;
   229→   229→  } | null>(null);
   230→   230→  const [hiddenColumns, setHiddenColumns] = useState<Set<string>>(new Set());
   231→   231→  /**
   232→   232→   * Explicit px width per data column — `null` = classic auto layout that
   233→   233→   * fills the window. Set on the first resize interaction (all columns are
   234→   234→   * frozen at their rendered widths, then only the grabbed one moves).
   235→   235→   */
   236→   236→  const [columnWidths, setColumnWidths] = useState<Record<string, number> | null>(null);
   237→   237→  /** Field currently being drag-resized (drives cursors + handle hint). */
   238→   238→  const [resizingField, setResizingField] = useState<string | null>(null);
   239→   239→  const [isColumnsOpen, setIsColumnsOpen] = useState(false);
   240→   240→  const [isDeleteConfirmOpen, setIsDeleteConfirmOpen] = useState(false);
   241→   241→  const [isDeleting, setIsDeleting] = useState(false);
   242→   242→
   243→   243→  /** Latest values for callbacks that must not re-create on every change. */
   244→   244→  const activeFiltersRef = useRef<ListColumnFilter[]>([]);
   245→   245→  const allRecordsRef = useRef<FormRecord[]>([]);
   246→   246→  const filterSeqRef = useRef(0);
   247→   247→  const filterMenuRef = useRef<HTMLDivElement | null>(null);
   248→   248→  const filterInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
   249→   249→  const tableRef = useRef<HTMLTableElement | null>(null);
   250→   250→
   251→   251→  // ── Loading ──
   252→   252→  /** Records URL with the JSON `?filters=` payload (server-side filtering). */
   253→   253→  const fetchRecordsPage = useCallback(
   254→   254→    async (columns: ListColumnFilter[]): Promise<FormRecord[]> => {
   255→   255→      if (!basePath) return [];
   256→   256→      const params = new URLSearchParams({ locale });
   257→   257→      if (columns.length) {
   258→   258→        params.set('filters', JSON.stringify(columns));
   259→   259→      }
   260→   260→      const res = await apiFetch<{ data: FormRecord[] }>(
   261→   261→        `${basePath}/records?${params.toString()}`,
   262→   262→      );
   263→   263→      return res.data ?? [];
   264→   264→    },
   265→   265→    [basePath, locale],
   266→   266→  );
   267→   267→  const loadLookups = useCallback(
   268→   268→    async (formDef: FormDef) => {
   269→   269→      const keys = Array.from(
   270→   270→        new Set(
   271→   271→          formDef.fields
   272→   272→            .filter((field) => field.fieldType === 'lookup' && field.lookupKey)
   273→   273→            .map((field) => field.lookupKey as string),
   274→   274→        ),
   275→   275→      );
   276→   276→      if (!keys.length || !lookupUrlBase) return;
   277→   277→      const results = await Promise.all(
   278→   278→        keys.map((key) =>
   279→   279→          apiFetch<{ data: LookupOption[] }>(
   280→   280→            `${lookupUrlBase}/${encodeURIComponent(key)}?locale=${encodeURIComponent(locale)}`,
   281→   281→          )
   282→   282→            .then((payload): [string, Map<string, string>] => [
   283→   283→              key,
   284→   284→              new Map(
   285→   285→                (payload.data ?? []).map((option) => [
   286→   286→                  option.code,
   287→   287→                  locale === 'fa' ? option.label : (option.labelEn ?? option.label),
   288→   288→                ]),
   289→   289→              ),
   290→   290→            ])
   291→   291→            .catch((): [string, Map<string, string>] => [key, new Map()]),
   292→   292→        ),
   293→   293→      );
   294→   294→      setLookupLabels(Object.fromEntries(results));
   295→   295→    },
   296→   296→    [locale, lookupUrlBase],
   297→   297→  );
   298→   298→
   299→   299→  const load = useCallback(async () => {
   300→   300→    if (!basePath) {
   301→   301→      setLoadError(GENERIC_LOAD_ERROR);
   302→   302→      setIsLoading(false);
   303→   303→      return;
   304→   304→    }
   305→   305→    setIsLoading(true);
   306→   306→    setLoadError(null);
   307→   307→    try {
   308→   308→      const defRes = await apiFetch<{ data: FormDef }>(
   309→   309→        `${basePath}?locale=${encodeURIComponent(locale)}`,
   310→   310→      );
   311→   311→      const nextDef = defRes.data;
   312→   312→      setDef(nextDef);
   313→   313→      setSelectedIds(new Set());
   314→   314→
   315→   315→      const all = await fetchRecordsPage([]);
   316→   316→      setAllRecords(all);
   317→   317→      setRecords(all);
   318→   318→      void loadLookups(nextDef);
   319→   319→    } catch (error) {
   320→   320→      if (error instanceof SessionExpiredError) return;
   321→   321→      setLoadError(
   322→   322→        error instanceof Error && error.message
   323→   323→          ? error.message
   324→   324→          : GENERIC_LOAD_ERROR,
   325→   325→      );
   326→   326→    } finally {
   327→   327→      setIsLoading(false);
   328→   328→    }
   329→   329→  }, [basePath, fetchRecordsPage, loadLookups, locale]);
   330→   330→
   331→   331→  useEffect(() => {
   332→   332→    void load();
   333→   333→  }, [load]);
   334→   334→
   335→   335→  const refreshRecords = useCallback(async () => {
   336→   336→    if (!basePath) return;
   337→   337→    setIsRefreshing(true);
   338→   338→    try {
   339→   339→      const next = await fetchRecordsPage([]);
   340→   340→      setAllRecords(next);
   341→   341→      const columns = activeFiltersRef.current;
   342→   342→      // Keep the visible slice consistent with the active column filters.
   343→   343→      const visible = columns.length
   344→   344→        ? await fetchRecordsPage(columns)
   345→   345→        : next;
   346→   346→      setRecords(visible);
   347→   347→      setSelectedIds((prev) => {
   348→   348→        const nextIds = new Set(
   349→   349→          [...prev].filter((id) => visible.some((record) => record.id === id)),
   350→   350→        );
   351→   351→        return nextIds.size === prev.size ? prev : nextIds;
   352→   352→      });
   353→   353→    } catch (error) {
   354→   354→      if (error instanceof SessionExpiredError) return;
   355→   355→      toast({
   356→   356→        title: t('dataList.loadError'),
   357→   357→        description: error instanceof Error ? error.message : undefined,
   358→   358→        variant: 'destructive',
   359→   359→      });
   360→   360→    } finally {
   361→   361→      setIsRefreshing(false);
   362→   362→    }
   363→   363→  }, [basePath, fetchRecordsPage, t]);
   364→   364→
   365→   365→  // Auto-refresh when a form window of the same form mutates records.
   366→   366→  useEffect(() => {
   367→   367→    if (!moduleKey || !formKey) return;
   368→   368→    const handler = (event: Event) => {
   369→   369→      const detail = (
   370→   370→        event as CustomEvent<{ moduleKey?: string; formKey?: string }>
   371→   371→      ).detail;
   372→   372→      if (detail?.moduleKey === moduleKey && detail?.formKey === formKey) {
   373→   373→        void refreshRecords();
   374→   374→      }
   375→   375→    };
   376→   376→    window.addEventListener(RECORDS_CHANGED_EVENT, handler);
   377→   377→    return () => window.removeEventListener(RECORDS_CHANGED_EVENT, handler);
   378→   378→  }, [formKey, moduleKey, refreshRecords]);
   379→   379→
   380→   380→  // ── Derived view models ──
   381→   381→  const sortedFields = useMemo(
   382→   382→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   383→   383→    [def],
   384→   384→  );
   385→   385→  const parentField = useMemo(
   386→   386→    () => (def?.type === 'tree' ? pickParentField(sortedFields) : null),
   387→   387→    [def, sortedFields],
   388→   388→  );
   389→   389→  const visibleFields = useMemo(
   390→   390→    () => sortedFields.filter((field) => !hiddenColumns.has(field.fieldKey)),
   391→   391→    [hiddenColumns, sortedFields],
   392→   392→  );
   393→   393→
   394→   394→  /** recordId → display title of that record (tree parent columns). */
   395→   395→  const parentTitles = useMemo(() => {
   396→   396→    const map = new Map<string, string>();
   397→   397→    if (!parentField) return map;
   398→   398→    const titleField =
   399→   399→      sortedFields.find(
   400→   400→        (field) =>
   401→   401→          field !== parentField && /title|name|desc/i.test(field.fieldKey),
   402→   402→      ) ??
   403→   403→      sortedFields.find((field) => field !== parentField) ??
   404→   404→      null;
   405→   405→    for (const record of allRecords) {
   406→   406→      const raw = titleField ? record.data[titleField.fieldKey] : null;
   407→   407→      map.set(
   408→   408→        record.id,
   409→   409→        raw === null || raw === undefined || raw === ''
   410→   410→          ? record.id
   411→   411→          : String(raw),
   412→   412→      );
   413→   413→    }
   414→   414→    return map;
   415→   415→  }, [parentField, allRecords, sortedFields]);
   416→   416→
   417→   417→  /** Display text of one cell (checkbox → بلی/خیر, lookup → label, …). */
   418→   418→  const cellText = useCallback(
   419→   419→    (field: FormFieldDef, record: FormRecord): string => {
   420→   420→      const raw = record.data[field.fieldKey];
   421→   421→      if (field.fieldType === 'checkbox') {
   422→   422→        return raw === true ? t('dataList.yes') : t('dataList.no');
   423→   423→      }
   424→   424→      // Tree parent columns resolve the parent record's title (the
   425→   425→      // stored value is the parent record id — must run before lookup).
   426→   426→      if (parentField && field.fieldKey === parentField.fieldKey) {
   427→   427→        if (raw === null || raw === undefined || raw === '') return '—';
   428→   428→        return parentTitles.get(String(raw)) ?? String(raw);
   429→   429→      }
   430→   430→      if (field.fieldType === 'lookup') {
   431→   431→        if (raw === null || raw === undefined || raw === '') return '—';
   432→   432→        const labels = field.lookupKey
   433→   433→          ? lookupLabels[field.lookupKey]
   434→   434→          : undefined;
   435→   435→        return labels?.get(String(raw)) ?? String(raw);
   436→   436→      }
   437→   437→      return formatFieldValue(field, raw, formatNumber);
   438→   438→    },
   439→   439→    [formatNumber, lookupLabels, parentField, parentTitles, t],
   440→   440→  );
   441→   441→
   442→   442→  /**
   443→   443→   * Effective column filters sent to the service as `?filters=` JSON —
   444→   444→   * the grid itself no longer filters client-side.
   445→   445→   */
   446→   446→  const activeFilters = useMemo<ListColumnFilter[]>(
   447→   447→    () =>
   448→   448→      Object.entries(filters)
   449→   449→        .filter(([, filter]) => isColumnFilterActive(filter))
   450→   450→        .map(([field, filter]) => ({
   451→   451→          field,
   452→   452→          op: filter.op,
   453→   453→          value: filter.value,
   454→   454→        })),
   455→   455→    [filters],
   456→   456→  );
   457→   457→
   458→   458→  useEffect(() => {
   459→   459→    activeFiltersRef.current = activeFilters;
   460→   460→    allRecordsRef.current = allRecords;
   461→   461→  }, [activeFilters, allRecords]);
   462→   462→
   463→   463→  // Server-side filtering: debounce typed values, then re-query the service.
   464→   464→  useEffect(() => {
   465→   465→    const handle = window.setTimeout(() => {
   466→   466→      if (!activeFilters.length) {
   467→   467→        setRecords(allRecordsRef.current);
   468→   468→        setIsFiltering(false);
