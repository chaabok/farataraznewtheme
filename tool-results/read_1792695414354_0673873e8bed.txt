    77→    77→/** Fallback parsing of the tab id when no descriptor is attached. */
    78→    78→function parseListModuleId(moduleId: string): {
    79→    79→  moduleKey: string | null;
    80→    80→  formKey: string | null;
    81→    81→} {
    82→    82→  if (!moduleId.startsWith('list:')) {
    83→    83→    return { moduleKey: null, formKey: null };
    84→    84→  }
    85→    85→  const formKey = moduleId.slice('list:'.length);
    86→    86→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    87→    87→}

   159→   159→/**

   166→   166→export function DataListWindow({ moduleId, list, onClose }: DataListWindowProps) {
   167→   167→  const { locale, t, tf, formatNumber } = useTranslation();
   168→   168→  const addFormTab = useDesktopStore((s) => s.addFormTab);
   169→   169→
   170→   170→  const fallback = useMemo(() => parseListModuleId(moduleId), [moduleId]);
   171→   171→  const moduleKey = list?.moduleKey ?? fallback.moduleKey;
   172→   172→  const formKey = list?.formKey ?? fallback.formKey;
   173→   173→  const basePath =
   174→   174→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
   175→   175→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   176→   176→
   177→   177→  // ── Data ──
   178→   178→  const [def, setDef] = useState<FormDef | null>(null);
   179→   179→  /** Full dataset of the form (parent-title context + total count). */
   180→   180→  const [allRecords, setAllRecords] = useState<FormRecord[]>([]);
   181→   181→  /** Rows currently displayed — the server-filtered slice. */
   182→   182→  const [records, setRecords] = useState<FormRecord[]>([]);
   183→   183→  const [isLoading, setIsLoading] = useState(true);
   184→   184→  const [isRefreshing, setIsRefreshing] = useState(false);
   185→   185→  /** True while a debounced server-side filter query is in flight. */
   186→   186→  const [isFiltering, setIsFiltering] = useState(false);
   187→   187→  const [loadError, setLoadError] = useState<string | null>(null);
   188→   188→  /** lookupKey → (option code → label). */
   189→   189→  const [lookupLabels, setLookupLabels] = useState<
   190→   190→    Record<string, Map<string, string>>
   191→   191→  >({});
   192→   192→
   193→   193→  // ── View state ──
   194→   194→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   195→   195→  const [filters, setFilters] = useState<Record<string, ColumnFilterState>>({});
   196→   196→  const [isFilterRowVisible, setIsFilterRowVisible] = useState(true);
   197→   197→  /** Open operator menu, anchored to the funnel button of one column. */
   198→   198→  const [filterMenu, setFilterMenu] = useState<{
   199→   199→    fieldKey: string;
   200→   200→    rect: DOMRect;
   201→   201→    left: number;
   202→   202→    openUp: boolean;
   203→   203→  } | null>(null);
   204→   204→  const [hiddenColumns, setHiddenColumns] = useState<Set<string>>(new Set());
   205→   205→  const [isColumnsOpen, setIsColumnsOpen] = useState(false);
   206→   206→  const [isDeleteConfirmOpen, setIsDeleteConfirmOpen] = useState(false);
   207→   207→  const [isDeleting, setIsDeleting] = useState(false);
   208→   208→
   209→   209→  /** Latest values for callbacks that must not re-create on every change. */
   210→   210→  const activeFiltersRef = useRef<ListColumnFilter[]>([]);
   211→   211→  const allRecordsRef = useRef<FormRecord[]>([]);
   212→   212→  const filterSeqRef = useRef(0);
   213→   213→  const filterMenuRef = useRef<HTMLDivElement | null>(null);
   214→   214→  const filterInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
   215→   215→
   216→   216→  // ── Loading ──
   217→   217→  /** Records URL with the JSON `?filters=` payload (server-side filtering). */
   218→   218→  const fetchRecordsPage = useCallback(
   219→   219→    async (columns: ListColumnFilter[]): Promise<FormRecord[]> => {
   220→   220→      if (!basePath) return [];
   221→   221→      const params = new URLSearchParams({ locale });
   222→   222→      if (columns.length) {
   223→   223→        params.set('filters', JSON.stringify(columns));
   224→   224→      }
   225→   225→      const res = await apiFetch<{ data: FormRecord[] }>(
   226→   226→        `${basePath}/records?${params.toString()}`,
   227→   227→      );
   228→   228→      return res.data ?? [];
   229→   229→    },
   230→   230→    [basePath, locale],
   231→   231→  );
   232→   232→  const loadLookups = useCallback(
   233→   233→    async (formDef: FormDef) => {
   234→   234→      const keys = Array.from(
   235→   235→        new Set(
   236→   236→          formDef.fields
   237→   237→            .filter((field) => field.fieldType === 'lookup' && field.lookupKey)
   238→   238→            .map((field) => field.lookupKey as string),
   239→   239→        ),
   240→   240→      );
   241→   241→      if (!keys.length || !lookupUrlBase) return;
   242→   242→      const results = await Promise.all(
   243→   243→        keys.map((key) =>
   244→   244→          apiFetch<{ data: LookupOption[] }>(
   245→   245→            `${lookupUrlBase}/${encodeURIComponent(key)}?locale=${encodeURIComponent(locale)}`,
   246→   246→          )
   247→   247→            .then((payload): [string, Map<string, string>] => [
   248→   248→              key,
   249→   249→              new Map(
   250→   250→                (payload.data ?? []).map((option) => [
   251→   251→                  option.code,
   252→   252→                  locale === 'fa' ? option.label : (option.labelEn ?? option.label),
   253→   253→                ]),
   254→   254→              ),
   255→   255→            ])
   256→   256→            .catch((): [string, Map<string, string>] => [key, new Map()]),
   257→   257→        ),
   258→   258→      );
   259→   259→      setLookupLabels(Object.fromEntries(results));
   260→   260→    },
   261→   261→    [locale, lookupUrlBase],
   262→   262→  );
   263→   263→
   264→   264→  const load = useCallback(async () => {
   265→   265→    if (!basePath) {
   266→   266→      setLoadError(GENERIC_LOAD_ERROR);
   267→   267→      setIsLoading(false);
   268→   268→      return;
   269→   269→    }
   270→   270→    setIsLoading(true);
   271→   271→    setLoadError(null);
   272→   272→    try {
   273→   273→      const defRes = await apiFetch<{ data: FormDef }>(
   274→   274→        `${basePath}?locale=${encodeURIComponent(locale)}`,
   275→   275→      );
   276→   276→      const nextDef = defRes.data;
   277→   277→      setDef(nextDef);
   278→   278→      setSelectedIds(new Set());
   279→   279→
   280→   280→      const all = await fetchRecordsPage([]);
   281→   281→      setAllRecords(all);
   282→   282→      setRecords(all);
   283→   283→      void loadLookups(nextDef);
   284→   284→    } catch (error) {
   285→   285→      if (error instanceof SessionExpiredError) return;
   286→   286→      setLoadError(
   287→   287→        error instanceof Error && error.message
   288→   288→          ? error.message
   289→   289→          : GENERIC_LOAD_ERROR,
   290→   290→      );
   291→   291→    } finally {
   292→   292→      setIsLoading(false);
   293→   293→    }
   294→   294→  }, [basePath, fetchRecordsPage, loadLookups, locale]);
   295→   295→
   296→   296→  useEffect(() => {
   297→   297→    void load();
   298→   298→  }, [load]);
   299→   299→
   300→   300→  const refreshRecords = useCallback(async () => {
   301→   301→    if (!basePath) return;
   302→   302→    setIsRefreshing(true);
   303→   303→    try {
   304→   304→      const next = await fetchRecordsPage([]);
   305→   305→      setAllRecords(next);
   306→   306→      const columns = activeFiltersRef.current;
   307→   307→      // Keep the visible slice consistent with the active column filters.
   308→   308→      const visible = columns.length
   309→   309→        ? await fetchRecordsPage(columns)
   310→   310→        : next;
   311→   311→      setRecords(visible);
   312→   312→      setSelectedIds((prev) => {
   313→   313→        const nextIds = new Set(
   314→   314→          [...prev].filter((id) => visible.some((record) => record.id === id)),
   315→   315→        );
   316→   316→        return nextIds.size === prev.size ? prev : nextIds;
   317→   317→      });
   318→   318→    } catch (error) {
   319→   319→      if (error instanceof SessionExpiredError) return;
   320→   320→      toast({
   321→   321→        title: t('dataList.loadError'),
   322→   322→        description: error instanceof Error ? error.message : undefined,
   323→   323→        variant: 'destructive',
   324→   324→      });
   325→   325→    } finally {
   326→   326→      setIsRefreshing(false);
   327→   327→    }
   328→   328→  }, [basePath, fetchRecordsPage, t]);
   329→   329→
   330→   330→  // Auto-refresh when a form window of the same form mutates records.
   331→   331→  useEffect(() => {
   332→   332→    if (!moduleKey || !formKey) return;
   333→   333→    const handler = (event: Event) => {
   334→   334→      const detail = (
   335→   335→        event as CustomEvent<{ moduleKey?: string; formKey?: string }>
   336→   336→      ).detail;
   337→   337→      if (detail?.moduleKey === moduleKey && detail?.formKey === formKey) {
   338→   338→        void refreshRecords();
   339→   339→      }
   340→   340→    };
   341→   341→    window.addEventListener(RECORDS_CHANGED_EVENT, handler);
   342→   342→    return () => window.removeEventListener(RECORDS_CHANGED_EVENT, handler);
   343→   343→  }, [formKey, moduleKey, refreshRecords]);
   344→   344→
   345→   345→  // ── Derived view models ──
   346→   346→  const sortedFields = useMemo(
   347→   347→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   348→   348→    [def],
   349→   349→  );
   350→   350→  const parentField = useMemo(
   351→   351→    () => (def?.type === 'tree' ? pickParentField(sortedFields) : null),
   352→   352→    [def, sortedFields],
   353→   353→  );
   354→   354→  const visibleFields = useMemo(
   355→   355→    () => sortedFields.filter((field) => !hiddenColumns.has(field.fieldKey)),
   356→   356→    [hiddenColumns, sortedFields],
   357→   357→  );
   358→   358→
   359→   359→  /** recordId → display title of that record (tree parent columns). */
   360→   360→  const parentTitles = useMemo(() => {
   361→   361→    const map = new Map<string, string>();
   362→   362→    if (!parentField) return map;
   363→   363→    const titleField =
   364→   364→      sortedFields.find(
   365→   365→        (field) =>
   366→   366→          field !== parentField && /title|name|desc/i.test(field.fieldKey),
   367→   367→      ) ??
   368→   368→      sortedFields.find((field) => field !== parentField) ??
   369→   369→      null;
   370→   370→    for (const record of allRecords) {
   371→   371→      const raw = titleField ? record.data[titleField.fieldKey] : null;
   372→   372→      map.set(
   373→   373→        record.id,
   374→   374→        raw === null || raw === undefined || raw === ''
   375→   375→          ? record.id
   376→   376→          : String(raw),
   377→   377→      );
   378→   378→    }
   379→   379→    return map;
   380→   380→  }, [parentField, allRecords, sortedFields]);
   381→   381→
   382→   382→  /** Display text of one cell (checkbox → بلی/خیر, lookup → label, …). */
   383→   383→  const cellText = useCallback(
   384→   384→    (field: FormFieldDef, record: FormRecord): string => {
   385→   385→      const raw = record.data[field.fieldKey];
   386→   386→      if (field.fieldType === 'checkbox') {
   387→   387→        return raw === true ? t('dataList.yes') : t('dataList.no');
   388→   388→      }
   389→   389→      // Tree parent columns resolve the parent record's title (the
   390→   390→      // stored value is the parent record id — must run before lookup).
   391→   391→      if (parentField && field.fieldKey === parentField.fieldKey) {
   392→   392→        if (raw === null || raw === undefined || raw === '') return '—';
   393→   393→        return parentTitles.get(String(raw)) ?? String(raw);
   394→   394→      }
   395→   395→      if (field.fieldType === 'lookup') {
   396→   396→        if (raw === null || raw === undefined || raw === '') return '—';
   397→   397→        const labels = field.lookupKey
   398→   398→          ? lookupLabels[field.lookupKey]
   399→   399→          : undefined;
   400→   400→        return labels?.get(String(raw)) ?? String(raw);
   401→   401→      }
   402→   402→      return formatFieldValue(field, raw, formatNumber);
   403→   403→    },
   404→   404→    [formatNumber, lookupLabels, parentField, parentTitles, t],
   405→   405→  );
   406→   406→
   407→   407→  /**
   408→   408→   * Effective column filters sent to the service as `?filters=` JSON —
   409→   409→   * the grid itself no longer filters client-side.
   410→   410→   */
   411→   411→  const activeFilters = useMemo<ListColumnFilter[]>(
   412→   412→    () =>
   413→   413→      Object.entries(filters)
   414→   414→        .filter(([, filter]) => isColumnFilterActive(filter))
   415→   415→        .map(([field, filter]) => ({
   416→   416→          field,
   417→   417→          op: filter.op,
   418→   418→          value: filter.value,
   419→   419→        })),
   420→   420→    [filters],
   421→   421→  );
   422→   422→
   423→   423→  useEffect(() => {
   424→   424→    activeFiltersRef.current = activeFilters;
   425→   425→    allRecordsRef.current = allRecords;
   426→   426→  }, [activeFilters, allRecords]);
   427→   427→
   428→   428→  // Server-side filtering: debounce typed values, then re-query the service.
   429→   429→  useEffect(() => {
   430→   430→    const handle = window.setTimeout(() => {
   431→   431→      if (!activeFilters.length) {
   432→   432→        setRecords(allRecordsRef.current);
   433→   433→        setIsFiltering(false);
   434→   434→        return;
   435→   435→      }
   436→   436→      const seq = ++filterSeqRef.current;
   437→   437→      setIsFiltering(true);
   438→   438→      fetchRecordsPage(activeFilters)
   439→   439→        .then((next) => {
   440→   440→          if (filterSeqRef.current !== seq) return;
   441→   441→          setRecords(next);
   442→   442→        })
   443→   443→        .catch((error) => {
   444→   444→          if (filterSeqRef.current !== seq) return;
   445→   445→          if (error instanceof SessionExpiredError) return;
   446→   446→          toast({
   447→   447→            title: t('dataList.loadError'),
   448→   448→            description: error instanceof Error ? error.message : undefined,
   449→   449→            variant: 'destructive',
   450→   450→          });
   451→   451→        })
   452→   452→        .finally(() => {
   453→   453→          if (filterSeqRef.current === seq) setIsFiltering(false);
   454→   454→        });
   455→   455→    }, 250);
   456→   456→    return () => window.clearTimeout(handle);
   457→   457→  }, [activeFilters, fetchRecordsPage, t]);
   458→   458→
   459→   459→  /** The displayed rows — already filtered by the service. */
   460→   460→  const filteredRecords = records;
   461→   461→
   462→   462→  // ── Column filter operator menu ──
   463→   463→  const toggleFilterMenu = useCallback(
   464→   464→    (fieldKey: string, button: HTMLButtonElement) => {
   465→   465→      setFilterMenu((prev) => {
   466→   466→        if (prev?.fieldKey === fieldKey) return null;
   467→   467→        const rect = button.getBoundingClientRect();
   468→   468→        const menuWidth = 184;
   469→   469→        const menuHeight = 290;
   470→   470→        return {
   471→   471→          fieldKey,
   472→   472→          rect,
   473→   473→          left: Math.max(
   474→   474→            8,
   475→   475→            Math.min(rect.left, window.innerWidth - menuWidth - 8),
   476→   476→          ),
   477→   477→          openUp:
   478→   478→            rect.bottom + menuHeight > window.innerHeight - 8 &&
