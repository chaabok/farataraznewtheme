    90→    90→      title: t(`app.${appKey}` as never),

   175→   175→
   176→   176→  /* measure the icons layer for clamping (viewport resizes keep icons inside).
   177→   177→     Task 39: re-attach on every grid (re)mount — the layer unmounts while the
   178→   178→     Task-38 sidebar is open, so a one-shot observer would leave a stale rect. */
   179→   179→  React.useEffect(() => {
   180→   180→    if (iconMode !== "grid") return;
   181→   181→    const el = layerRef.current;
   182→   182→    if (!el) return;
   183→   183→    const update = () => setRect({ w: el.clientWidth, h: el.clientHeight });
   184→   184→    update();
   185→   185→    const ro = new ResizeObserver(update);
   186→   186→    ro.observe(el);
   187→   187→    return () => ro.disconnect();
   188→   188→  }, [iconMode]);
   189→   189→
   190→   190→  const clampPos = React.useCallback(
   191→   191→    (p: { x: number; y: number }): { x: number; y: number } => {
   192→   192→      const w = rect?.w ?? (typeof window !== "undefined" ? Math.max(0, window.innerWidth - 16) : 1264);
   193→   193→      const h = rect?.h ?? (typeof window !== "undefined" ? Math.max(0, window.innerHeight - 68) : 732);
   194→   194→      return {
   195→   195→        x: Math.min(Math.max(p.x, 0), Math.max(0, w - ICON_W)),
   196→   196→        y: Math.min(Math.max(p.y, 0), Math.max(0, h - ICON_H)),
   197→   197→      };
   198→   198→    },
   199→   199→    [rect],
   200→   200→  );
   201→   201→
   202→   202→  /* default slot = column-major grid anchored to the start edge (right in RTL, left in LTR) */
   203→   203→  const defaultPos = React.useCallback(
   204→   204→    (index: number): { x: number; y: number } => {
   205→   205→      const h = rect?.h ?? (typeof window !== "undefined" ? Math.max(1, window.innerHeight - 68) : 732);
   206→   206→      const w = rect?.w ?? (typeof window !== "undefined" ? Math.max(1, window.innerWidth - 16) : 1264);
   207→   207→      const rows = Math.max(1, Math.floor((h - GRID_PAD) / GRID_STEP_Y));
   208→   208→      const col = Math.floor(index / rows);
   209→   209→      const row = index % rows;
   210→   210→      const x = dir === "rtl" ? w - GRID_PAD - ICON_W - col * GRID_STEP_X : GRID_PAD + col * GRID_STEP_X;
   211→   211→      return { x, y: GRID_PAD + row * GRID_STEP_Y };
   212→   212→    },
   213→   213→    [rect, dir],
   214→   214→  );
   215→   215→
   216→   216→  /* Task 35 — the BusinessStructure shortcut left the desktop: the form
   217→   217→     moved into the Settings window as a sub-page (owner request). Icons:
   218→   218→     the Task-24/Task-30 form-shell samples, then the catalog apps minus
   219→   219→     userManagement — it moved into the Settings window too (Task 37). */
   220→   220→  const items = React.useMemo<IconEntry[]>(() => {
   221→   221→    const base: IconEntry[] = [
   222→   222→      { id: "form-demo", kind: "form-demo", label: t("fdf.iconLabel") },
   223→   223→      { id: "tree-demo", kind: "tree-demo", label: t("stf.iconLabel") },
   224→   224→      ...apps
   225→   225→        .filter((a) => a.appKey !== "userManagement")
   226→   226→        .map((a) => ({
   227→   227→          id: a.appKey,
   228→   228→          kind: "app" as WindowKind,
   229→   229→          appKey: a.appKey,
   230→   230→          label: t(a.titleKey as never),
   231→   231→        })),
   232→   232→    ];
   233→   233→    /* Task 39 — owner request: order by the active language's collation in
   234→   234→       BOTH presentations, so the desktop grid and the sidebar always tell
   235→   235→       the same story (labels are unique, the id is just a stable tiebreak). */
   236→   236→    return base.sort(
   237→   237→      (a, b) => collator.compare(a.label, b.label) || collator.compare(a.id, b.id),
   238→   238→    );
   239→   239→  }, [apps, t, collator]);
   240→   240→
   241→   241→  /* Task 36 — the catalog "settings" icon opens the real Settings window too:

   244→   244→  const activate = (entry: IconEntry) => openWindow(entry.kind, entry.appKey);
   245→   245→
   246→   246→  /* Task 39 — self-heal: a persisted icon layout whose icons visually overlap
   247→   247→     (a legacy pile, or positions clamped together by a resize) is discarded in
   248→   248→     favor of the clean language-sorted grid. Deliberate drag positions that do
   249→   249→     NOT overlap are left untouched, and plain re-renders (selection, drag
   250→   250→     ghost…) never re-evaluate this — only grid (re)mounts and real layer
   251→   251→     geometry changes do. */
   252→   252→  React.useEffect(() => {
   253→   253→    if (iconMode !== "grid" || !freeMode || !rect) return;
   254→   254→    const saved = useDesktop.getState().iconPositions;
   255→   255→    const pts = items.map((entry, i) => clampPos(saved[entry.id] ?? defaultPos(i)));
   256→   256→    for (let a = 0; a < pts.length; a++) {
   257→   257→      for (let b = a + 1; b < pts.length; b++) {
   258→   258→        const p = pts[a];
   259→   259→        const q = pts[b];
   260→   260→        if (p.x < q.x + ICON_W && q.x < p.x + ICON_W && p.y < q.y + ICON_H && q.y < p.y + ICON_H) {
   261→   261→          resetIconLayout();
   262→   262→          return;
   263→   263→        }
   264→   264→      }
   265→   265→    }
   266→   266→  }, [iconMode, freeMode, rect, items, clampPos, defaultPos, resetIconLayout]);
   267→   267→
   268→   268→  const beginDragTracking = (e: React.PointerEvent, entry: IconEntry) => {
   269→   269→    if (!freeMode || e.button !== 0) return;
   270→   270→    const btn = (e.currentTarget as HTMLElement).getBoundingClientRect();
   271→   271→    const grab = { x: e.clientX - btn.left, y: e.clientY - btn.top };
   272→   272→    const startX = e.clientX;
   273→   273→    const startY = e.clientY;
   274→   274→    const pointerId = e.pointerId;
   275→   275→    const taskbarTop =
   276→   276→      document.querySelector<HTMLElement>("[data-taskbar]")?.getBoundingClientRect().top ??
   277→   277→      (typeof window !== "undefined" ? window.innerHeight - 52 : 748);
   278→   278→    let active = false;
   279→   279→    let over = false;
   280→   280→
   281→   281→    const onMove = (ev: PointerEvent) => {
   282→   282→      if (ev.pointerId !== pointerId) return;
   283→   283→      if (!active) {
   284→   284→        if (Math.hypot(ev.clientX - startX, ev.clientY - startY) < DRAG_THRESHOLD) return;
   285→   285→        active = true;
   286→   286→        setSelected(entry.id);
   287→   287→      }
   288→   288→      const o = ev.clientY >= taskbarTop - 12;
   289→   289→      if (o !== over) {
   290→   290→        over = o;
   291→   291→        setDragOverTaskbar(o);
   292→   292→      }
   293→   293→      setGhost({ entry, x: ev.clientX - grab.x, y: ev.clientY - grab.y, overTaskbar: o });
   294→   294→    };
   295→   295→
   296→   296→    const finish = (ev: PointerEvent | null, commit: boolean) => {
   297→   297→      window.removeEventListener("pointermove", onMove);
   298→   298→      window.removeEventListener("pointerup", onUp);
   299→   299→      window.removeEventListener("pointercancel", onCancel);
   300→   300→      setDragOverTaskbar(false);
   301→   301→      setGhost(null);
   302→   302→      if (!active) return; /* plain click — let onClick/dblclick behave normally */
   303→   303→      suppressClickRef.current = true;
   304→   304→      window.setTimeout(() => {
   305→   305→        suppressClickRef.current = false;
   306→   306→      }, 0);
   307→   307→      if (!commit || !ev) return;
   308→   308→      if (over) {
   309→   309→        if (pinToTaskbar(entry.id)) toast.success(t("taskbar.pinnedToast"));
   310→   310→        else toast.info(t("taskbar.alreadyPinned"));
   311→   311→        return;
   312→   312→      }
   313→   313→      const layer = layerRef.current?.getBoundingClientRect();
   314→   314→      if (!layer) return;
   315→   315→      setIconPos(
   316→   316→        entry.id,
   317→   317→        clampPos({
   318→   318→          x: ev.clientX - grab.x - layer.left,
   319→   319→          y: ev.clientY - grab.y - layer.top,
   320→   320→        }),
   321→   321→      );
   322→   322→    };
   323→   323→
   324→   324→    const onUp = (ev: PointerEvent) => finish(ev, true);
   325→   325→    const onCancel = () => finish(null, false);
   326→   326→
   327→   327→    window.addEventListener("pointermove", onMove);
   328→   328→    window.addEventListener("pointerup", onUp);
   329→   329→    window.addEventListener("pointercancel", onCancel);
   330→   330→  };
   331→   331→
   332→   332→  const handleClick = (e: React.MouseEvent, entry: IconEntry) => {
   333→   333→    if (suppressClickRef.current) return;
   334→   334→    setSelected(entry.id);
   335→   335→    /* touch devices: single tap opens (desktop convention: double-click) */
   336→   336→    if (window.matchMedia?.("(pointer: coarse)").matches) activate(entry);
   337→   337→  };
   338→   338→
   339→   339→  const handleContextMenu = (e: React.MouseEvent, entry: IconEntry) => {
   340→   340→    e.preventDefault();
   341→   341→    setSelected(entry.id);
   342→   342→    setMenu({ x: e.clientX, y: e.clientY, entry, source: "grid" });
   343→   343→  };
   344→   344→
   345→   345→  const iconGlyph = (entry: IconEntry) =>
   346→   346→    entry.id === "form-demo" ? (
   347→   347→      <span className="flex h-12 w-12 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-teal-500 to-emerald-600 text-white shadow-lg">
   348→   348→        <FilePlus2 className="h-6 w-6" />
   349→   349→      </span>
   350→   350→    ) : entry.id === "tree-demo" ? (
   351→   351→      <span className="flex h-12 w-12 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-amber-500 to-orange-600 text-white shadow-lg">
   352→   352→        <ListTree className="h-6 w-6" />
   353→   353→      </span>
   354→   354→    ) : (
   355→   355→      <AppTile appKey={entry.appKey ?? ""} className="h-12 w-12 rounded-2xl" iconClass="h-6 w-6" />
   356→   356→    );
   357→   357→
   358→   358→  /* Task 38 — sidebar geometry: the panel docks on the inline-start edge,
   359→   359→     sliding clear of a dashboard header docked on the same side (§1.4:
   360→   360→     logical positioning only — no physical classes). */
   361→   361→  const sidebarOpen = iconMode === "sidebar";
   362→   362→  const startDockedHeader =
   363→   363→    (dir === "rtl" && headerDock === "right") || (dir === "ltr" && headerDock === "left");
   364→   364→  const startOffset = startDockedHeader ? HEADER_THICK + GRID_PAD : GRID_PAD;
   365→   365→  const topOffset = headerDock === "top" ? HEADER_THICK + GRID_PAD : GRID_PAD;
   366→   366→  const SIDEBAR_W = "min(272px, 100vw - 72px)";
   367→   367→  const toggleStart = sidebarOpen
   368→   368→    ? `calc(${startOffset}px + ${sidebarCollapsed ? "64px" : SIDEBAR_W} + 8px)`
   369→   369→    : `${startOffset + 4}px`;
   370→   370→  const ToggleIcon = sidebarOpen
   371→   371→    ? dir === "rtl"
   372→   372→      ? PanelRightClose
   373→   373→      : PanelLeftClose
   374→   374→    : dir === "rtl"
   375→   375→      ? PanelRightOpen
   376→   376→      : PanelLeftOpen;
   377→   377→  const SidebarCollapseIcon = dir === "rtl" ? ChevronRight : ChevronLeft;
   378→   378→  const SidebarExpandIcon = dir === "rtl" ? ChevronLeft : ChevronRight;
   379→   379→
   380→   380→  /* Task 38 — sidebar icon style: compact tightly-rounded tiles (vs the
   381→   381→     desktop's large rounded glyphs) so the presentation visibly changes
   382→   382→     when switching modes (owner request). */
   383→   383→  const sidebarGlyph = (entry: IconEntry) =>
   384→   384→    entry.id === "form-demo" ? (
   385→   385→      <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br from-teal-500 to-emerald-600 text-white shadow-md">
   386→   386→        <FilePlus2 className="h-4 w-4" />
   387→   387→      </span>
   388→   388→    ) : entry.id === "tree-demo" ? (
   389→   389→      <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br from-amber-500 to-orange-600 text-white shadow-md">
   390→   390→        <ListTree className="h-4 w-4" />
   391→   391→      </span>
   392→   392→    ) : (
   393→   393→      <AppTile appKey={entry.appKey ?? ""} className="h-8 w-8 rounded-lg" iconClass="h-4 w-4" />
   394→   394→    );
   395→   395→
   396→   396→  return (
   397→   397→    <>
   398→   398→      {iconMode === "grid" && (
   399→   399→      <div
   400→   400→        ref={layerRef}
   401→   401→        aria-label={t("startmenu.pinned")}
   402→   402→      style={{
   403→   403→        top: headerDock === "top" ? HEADER_THICK + GRID_PAD : GRID_PAD,
   404→   404→        bottom: 60,
   405→   405→        left: headerDock === "left" ? HEADER_THICK + GRID_PAD : GRID_PAD,
   406→   406→        right: headerDock === "right" ? HEADER_THICK + GRID_PAD : GRID_PAD,
