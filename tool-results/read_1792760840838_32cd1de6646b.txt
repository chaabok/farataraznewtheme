   244→function FieldCell({
   245→  field,
   246→  onChange,
   247→}: {
   248→  field: SimpleFormField;
   249→  onChange?: (id: string, value: string | boolean) => void;
   250→}) {
   251→  const type = field.type ?? "text";
   252→  /* Task 27 — compact vertical metrics: field height 32px → 22px (−31%) so
   253→     the tabs section and the form footer gain room. Horizontal metrics
   254→     (label alignment, --label-w, gaps) are untouched. */
   255→  const inputBase = "h-5.5 min-w-0 text-xs";
   256→
   257→  /* checkbox — inline toggle pushed to the row end (legacy «فعال» placement) */
   258→  if (type === "checkbox") {
   259→    const cid = `sft-chk-${field.id}`;
   260→    return (
   261→      <label
   262→        htmlFor={cid}
   263→        className="flex cursor-pointer items-center justify-end gap-2 text-xs text-foreground/90"
   264→      >
   265→        <Checkbox
   266→          id={cid}
   267→          checked={field.checked === true}
   268→          onCheckedChange={(v) => onChange?.(field.id, v === true)}
   269→          disabled={field.disabled}
   270→          aria-label={field.label}
   271→        />
   272→        <span className="whitespace-nowrap">{field.label}</span>
   273→      </label>
   274→    );
   275→  }
   276→
   277→  return (
   278→    <div data-form-cell className="flex min-w-0 items-center gap-2">
   279→      {/* label unit: [text-box (var width)][colon][asterisk slot] — the tail is
   280→          uniform (asterisk always occupies space) so every input of the form
   281→          starts on one shared edge — column-continuous vertical alignment
   282→          (Task 25.1, useAlignedLabels) */}
   283→      <span
   284→        data-form-label
   285→        className="flex shrink-0 items-center gap-0.5 whitespace-nowrap text-xs text-muted-foreground"
   286→      >
   287→        <span className="w-[var(--label-w,auto)] shrink-0 text-end">
   288→          <span data-form-label-text className="inline-block whitespace-nowrap">
   289→            {field.label}
   290→          </span>
   291→        </span>
   292→        <span className="opacity-70" aria-hidden>
   293→          :
   294→        </span>
   295→        <span
   296→          aria-hidden
   297→          className={cn("text-rose-500", !field.required && "invisible")}
   298→        >
   299→          *
   300→        </span>
   301→      </span>
   302→
   303→      {type === "select" ? (
   304→        <Select
   305→          value={field.value}
   306→          onValueChange={(v) => onChange?.(field.id, v)}
   307→          disabled={field.disabled}
   308→        >
   309→          <SelectTrigger aria-label={field.label} className={cn(inputBase, "flex-1")}>
   310→            <SelectValue placeholder={field.placeholder} />
   311→          </SelectTrigger>
   312→          <SelectContent>
   313→            {(field.options ?? []).map((o) => (
   314→              <SelectItem key={o.value} value={o.value} className="text-xs">
   315→                {o.label}
   316→              </SelectItem>
   317→            ))}
   318→          </SelectContent>
   319→        </Select>
   320→      ) : type === "lookup" ? (
   321→        <div className="flex min-w-0 flex-1 items-center gap-1.5">
   322→          <Input
   323→            value={field.value ?? ""}
   324→            placeholder={field.placeholder}
   325→            readOnly={field.readOnly}
   326→            disabled={field.disabled}
   327→            aria-label={field.label}
   328→            onChange={(e) => onChange?.(field.id, e.target.value)}
   329→            className={inputBase}
   330→          />
   331→          {field.onLookup && (
   332→            <button
   333→              type="button"
   334→              aria-label={`${field.label} — +`}
   335→              onClick={field.onLookup}
   336→              className="flex h-5.5 w-5.5 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br from-teal-500 to-emerald-600 text-white shadow-md transition-all hover:brightness-110 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring cursor-pointer"
   337→            >
   338→              <Plus className="h-3.5 w-3.5" aria-hidden />
   339→            </button>
   340→          )}
   341→        </div>
   342→      ) : (
   343→        <div className="relative min-w-0 flex-1">
   344→          <Input
   345→            value={field.value ?? ""}
   346→            inputMode={type === "number" ? "decimal" : undefined}
   347→            placeholder={field.placeholder}
   348→            readOnly={field.readOnly}
   349→            disabled={field.disabled}
   350→            aria-label={field.label}
   351→            onChange={(e) => onChange?.(field.id, e.target.value)}
   352→            className={cn(inputBase, field.suffixChip && "pe-16")}
   353→          />
   354→          {field.suffixChip && (
   355→            <span className="pointer-events-none absolute end-1 top-1/2 flex h-4 -translate-y-1/2 items-center rounded-md bg-rose-500/15 px-1.5 text-[10px] font-semibold text-rose-600 dark:text-rose-400">
   356→              {field.suffixChip}
   357→            </span>
   358→          )}
   359→        </div>
   360→      )}
   361→    </div>
   362→  );
   363→}

   365→/**
   366→ * SimpleFormTwoRowTabbed (Task 24) — the legacy «فرم سادهٔ دو ردیفه» shell:
   367→ * gradient form header (title + Task-28 window caption: minimize / maximize /
   368→ * close — the strip doubles as the drag handle of a chromeless host window),
   369→ * the Task-20 legacy action bar (plus the Task-28 redesigned help tile) and
   370→ * record navigation, a two-row field grid (with Task-25 automatic label
   371→ * alignment via useAlignedLabels — default scope "container": straight
   372→ * vertical input columns; opt-in labelScope="row" for per-row alignment),
   373→ * and OPTIONAL tabbed detail grids with an empty state and a footer
   374→ * row-actions strip (+ / × / copy / save). All action-bar handlers are
   375→ * pass-throughs — the caller owns persistence and toasts.
   376→ */
   377→export function SimpleFormTwoRowTabbed({
   378→  title,
   379→  badge,
   380→  icon,
   381→  fields,
   382→  onFieldChange,
   383→  labelScope = "container",
   384→  tabs,
   385→  defaultTabId,
   386→  onTabRowsChange,
   387→  onSaveRows,
   388→  onFirst,
   389→  onPrev,
   390→  onNext,
   391→  onLast,
   392→  onReset,
   393→  onSave,
   394→  onSaveNew,
   395→  onSaveClose,
   396→  onCopy,
   397→  onDelete,
   398→  onDeleteNew,
   399→  onDeleteClose,
   400→  onPrint,
   401→  onClose,
   402→  onHelp,
   403→  listItems,
   404→  listLoading,
   405→  onOpenListItem,
   406→  canSave = true,
   407→  canCopy = true,
   408→  canDelete = true,
   409→  canPrint = true,
   410→  onHeaderClose,
   411→  onHeaderMaximize,
   412→  onHeaderMinimize,
   413→  headerMaximized,
   414→  onHeaderDragStart,
   415→  onHeaderDragMove,
   416→  onHeaderDragEnd,
   417→  onHeaderDoubleClick,
   418→  className,
   419→}: SimpleFormTwoRowTabbedProps) {
   420→  const t = useT();
   421→  const tabsProp = React.useMemo(() => tabs ?? [], [tabs]);
   422→
   423→  /* Task 25.1 — automatic label alignment for the field grid. Default scope
   424→     "container": one shared label width ⇒ every input starts on the same
   425→     edge within its cell AND the edges stack into straight vertical columns
   426→     across rows (the owner's expectation). Pass labelScope="row" for
   427→     per-row alignment. */
   428→  const fieldsGridRef = React.useRef<HTMLDivElement | null>(null);
   429→  useAlignedLabels(fieldsGridRef, { scope: labelScope });
   430→
   431→  /* detail-grid rows: initialized once per mount from the tabs prop */
   432→  const [rowsByTab, setRowsByTab] = React.useState<Record<string, Record<string, string>[]>>(
   433→    () =>
   434→      Object.fromEntries(
   435→        tabsProp.map((tb) => [tb.id, (tb.rows ?? []).map((r) => ({ ...r }))]),
   436→      ),
   437→  );
   438→  const [activeTab, setActiveTab] = React.useState(
   439→    () => defaultTabId ?? tabsProp[0]?.id ?? "",
   440→  );
   441→  const [selected, setSelected] = React.useState<{ tabId: string; index: number } | null>(null);
   442→
   443→  const hasTabs = tabsProp.length > 0;
   444→  const hasNav = Boolean(onFirst || onPrev || onNext || onLast);
   445→
   446→  const commitRows = (tabId: string, rows: Record<string, string>[]) => {
   447→    setRowsByTab((prev) => ({ ...prev, [tabId]: rows }));
   448→    onTabRowsChange?.(tabId, rows);
   449→  };
   450→
   451→  const addRow = () => {
   452→    const tb = tabsProp.find((x) => x.id === activeTab);
   453→    if (!tb) return;
   454→    const rows = rowsByTab[activeTab] ?? [];
   455→    const blank = Object.fromEntries(tb.columns.map((c) => [c.id, ""])) as Record<string, string>;
   456→    const next = [...rows, blank];
   457→    commitRows(activeTab, next);
   458→    setSelected({ tabId: activeTab, index: next.length - 1 });
   459→  };
   460→
   461→  const removeSelectedRow = () => {
   462→    if (!selected || selected.tabId !== activeTab) return;
   463→    const rows = [...(rowsByTab[activeTab] ?? [])];
   464→    rows.splice(selected.index, 1);
   465→    commitRows(activeTab, rows);
   466→    setSelected(null);
   467→  };
   468→
   469→  const copySelectedRow = () => {
   470→    if (!selected || selected.tabId !== activeTab) return;
   471→    const rows = [...(rowsByTab[activeTab] ?? [])];
   472→    const clone = { ...rows[selected.index] };
   473→    rows.splice(selected.index + 1, 0, clone);
   474→    commitRows(activeTab, rows);
   475→    setSelected({ tabId: activeTab, index: selected.index + 1 });
   476→  };
   477→
   478→  const setCell = (tabId: string, rowIndex: number, colId: string, value: string) => {
   479→    const rows = [...(rowsByTab[tabId] ?? [])];
   480→    if (!rows[rowIndex]) return;
   481→    rows[rowIndex] = { ...rows[rowIndex], [colId]: value };
   482→    commitRows(tabId, rows);
   483→  };
   484→
   485→  const selectedInActive = Boolean(selected && selected.tabId === activeTab);
   486→
   487→  return (
   488→    <section
   489→      data-simple-form
   490→      aria-label={title}
   491→      className={cn("flex h-full min-h-0 flex-col overflow-hidden bg-background/25", className)}
   492→    >
   493→      {/* 1) form header — gradient identity strip; Task 26: emerald end sits
   494→          on the inline END side in both directions (mirrored, §1.4).
   495→          Task 28: the strip is also the drag handle of a chromeless host
   496→          window and hosts the caption buttons [– □ ✕] (inline end). */}
   497→      <header
   498→        className="relative flex h-11 shrink-0 select-none items-center gap-2.5 rtl:bg-gradient-to-l ltr:bg-gradient-to-r from-teal-600 via-teal-600 to-emerald-600 px-3 text-white"
   499→        onPointerDown={onHeaderDragStart}
   500→        onPointerMove={onHeaderDragMove}
   501→        onPointerUp={onHeaderDragEnd}
   502→        onPointerCancel={onHeaderDragEnd}
   503→        onDoubleClick={(e) => {
   504→          if ((e.target as HTMLElement).closest("[data-window-btn]")) return;
   505→          onHeaderDoubleClick?.();
   506→        }}
   507→      >
   508→        {icon && (
   509→          <span className="flex h-6 w-6 shrink-0 items-center justify-center rounded-lg bg-white/20 shadow-inner">
   510→            {icon}
   511→          </span>
   512→        )}
   513→        <h2 className="truncate text-sm font-bold tracking-tight drop-shadow-sm">{title}</h2>
   514→        {badge && (
   515→          <span className="rounded-full bg-white/15 px-2 py-0.5 text-[10px] font-medium text-white/90 ring-1 ring-inset ring-white/25">
   516→            {badge}
   517→          </span>
   518→        )}
   519→        <div className="ms-auto flex items-center gap-0.5">
   520→          {onHeaderMinimize && (
   521→            <HeaderButton onClick={onHeaderMinimize} label={t("common.minimize")}>
   522→              <Minus className="h-4 w-4" aria-hidden />
   523→            </HeaderButton>
   524→          )}
   525→          {onHeaderMaximize && (
   526→            <HeaderButton onClick={onHeaderMaximize} label={t("common.maximize")}>
   527→              {headerMaximized ? (
   528→                <Copy className="h-3 w-3 -scale-x-100" aria-hidden />
   529→              ) : (
   530→                <Square className="h-3 w-3" aria-hidden />
   531→              )}
   532→            </HeaderButton>
   533→          )}
   534→          {onHeaderClose && (
   535→            <HeaderButton onClick={onHeaderClose} label={t("common.close")}>
   536→              <X className="h-4 w-4" aria-hidden />
   537→            </HeaderButton>
   538→          )}
   539→        </div>
   540→        <span
   541→          aria-hidden
   542→          className="pointer-events-none absolute inset-x-0 bottom-0 h-px bg-white/25"
   543→        />
   544→      </header>
   545→
   546→      {/* 2) action strip — legacy FormActionBar (start) + record nav (end);
   547→          Task 28: help rides INSIDE the action bar as its 11th gradient tile */}
   548→      <div className="flex shrink-0 flex-wrap items-center gap-2 border-b border-border/60 bg-muted/40 px-3 py-2">
   549→        <FormActionBar
   550→          className="min-w-0 flex-1"
   551→          onReset={onReset}
   552→          onSave={onSave}
   553→          onSaveNew={onSaveNew}
   554→          onSaveClose={onSaveClose}
   555→          onCopy={onCopy}
   556→          onDelete={onDelete}
   557→          onDeleteNew={onDeleteNew}
   558→          onDeleteClose={onDeleteClose}
   559→          onPrint={onPrint}
   560→          onClose={onClose}
   561→          onHelp={onHelp}
   562→          listItems={listItems}
   563→          listLoading={listLoading}
   564→          onOpenListItem={onOpenListItem}
   565→          canSave={canSave}
   566→          canCopy={canCopy}
   567→          canDelete={canDelete}
   568→          canPrint={canPrint}
   569→        />
   570→        {/* Task 26 — per-button RTL flip (NOT on the group): the RTL flex
   571→            flow already puts «اولین» at the reading start (right edge), and
   572→            mirroring each button shows correctly-flipped glyphs. A group
   573→            flip would push First to the group's LEFT edge — wrong reading
   574→            order for RTL. */}
   575→        {hasNav && (
   576→          <div
   577→            role="group"
   578→            aria-label={t("sft.first")}
