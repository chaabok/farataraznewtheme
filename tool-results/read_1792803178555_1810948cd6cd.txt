    86→function resolveWindowMeta(win: WinState, t: ReturnType<typeof useT>) {
    87→  if (win.kind === "app") {
    88→    const appKey = win.appKey ?? "";
    89→    return {
    90→      title: t(`app.${appKey}` as never),
    91→      icon: <AppTile appKey={appKey} className="h-5 w-5 rounded-md" iconClass="h-3 w-3" />,
    92→      content: <AppPlaceholderWindow appKey={appKey} />,
    93→    };
    94→  }
    95→  const meta = STATIC_WINDOWS[win.kind];
    96→  return { title: t(meta.titleKey as never), icon: meta.icon, content: meta.content };
    97→}

   101→const ICON_W = 84;
   102→const ICON_H = 100;
   103→const GRID_STEP_X = 88;
   104→const GRID_STEP_Y = 106;
   105→const GRID_PAD = 8;
   106→/* Task 22: header thickness = 52 × 0.7 = 36.4px (owner height contract).

   118→function DesktopIcons() {
   119→  const t = useT();
   120→  const dir = useSettings((s) => s.dir());
   121→  const locale = useSettings((s) => s.locale);
   122→  /* Task 39 — owner request: icons are ordered by the selected language
   123→     (fa → Persian alphabet, en → A-Z, …) via Intl collation. */
   124→  const collator = React.useMemo(() => new Intl.Collator(locale, { numeric: true }), [locale]);
   125→  const openWindow = useDesktop((s) => s.openWindow);
   126→  const iconPositions = useDesktop((s) => s.iconPositions);
   127→  const pinned = useDesktop((s) => s.pinned);
   128→  const setIconPos = useDesktop((s) => s.setIconPos);
   129→  const resetIconLayout = useDesktop((s) => s.resetIconLayout);
   130→  const pinToTaskbar = useDesktop((s) => s.pinToTaskbar);
   131→  const setDragOverTaskbar = useDesktop((s) => s.setDragOverTaskbar);
   132→  const headerDock = useDesktop((s) => s.headerDock);
   133→  /* Task 38 — sidebar presentation state (owner request) */
   134→  const iconMode = useDesktop((s) => s.iconMode);
   135→  const sidebarCollapsed = useDesktop((s) => s.sidebarCollapsed);
   136→  const setIconMode = useDesktop((s) => s.setIconMode);
   137→  const setSidebarCollapsed = useDesktop((s) => s.setSidebarCollapsed);
   138→  const { apps, load } = useAppRegistry();
   139→
   140→  const layerRef = React.useRef<HTMLDivElement>(null);
   141→  const [rect, setRect] = React.useState<{ w: number; h: number } | null>(null);
   142→  const [freeMode, setFreeMode] = React.useState(false);
   143→  const [selected, setSelected] = React.useState<string | null>(null);
   144→  const [menu, setMenu] = React.useState<{
   145→    x: number;
   146→    y: number;
   147→    entry: IconEntry;
   148→    source: "grid" | "sidebar";
   149→  } | null>(null);
   150→  const [ghost, setGhost] = React.useState<{
   151→    entry: IconEntry;
   152→    x: number;
   153→    y: number;
   154→    overTaskbar: boolean;
   155→  } | null>(null);
   156→  const suppressClickRef = React.useRef(false);
   157→
   158→  React.useEffect(() => {
   159→    void load();
   160→  }, [load]);
   161→
   162→  /* SSR-safe: persisted layout (icon positions + footer pins) rehydrated after mount */
   163→  React.useEffect(() => {
   164→    void useDesktop.persist.rehydrate();
   165→  }, []);
   166→
   167→  /* free positioning + dragging on ≥640px viewports; small screens keep the wrap flow */
   168→  React.useEffect(() => {
   169→    const mq = window.matchMedia("(min-width: 640px)");
   170→    const apply = () => setFreeMode(mq.matches);
   171→    apply();
   172→    mq.addEventListener("change", apply);
   173→    return () => mq.removeEventListener("change", apply);
   174→  }, []);
   175→
   176→  /* measure the icons layer for clamping (viewport resizes keep icons inside).
   177→     Task 39: re-attach on every grid (re)mount — the layer unmounts while the
   178→     Task-38 sidebar is open, so a one-shot observer would leave a stale rect. */
   179→  React.useEffect(() => {
   180→    if (iconMode !== "grid") return;
   181→    const el = layerRef.current;
   182→    if (!el) return;
   183→    const update = () => setRect({ w: el.clientWidth, h: el.clientHeight });
   184→    update();
   185→    const ro = new ResizeObserver(update);
   186→    ro.observe(el);
   187→    return () => ro.disconnect();
   188→  }, [iconMode]);
   189→
   190→  const clampPos = React.useCallback(
   191→    (p: { x: number; y: number }): { x: number; y: number } => {
   192→      const w = rect?.w ?? (typeof window !== "undefined" ? Math.max(0, window.innerWidth - 16) : 1264);
   193→      const h = rect?.h ?? (typeof window !== "undefined" ? Math.max(0, window.innerHeight - 68) : 732);
   194→      return {
   195→        x: Math.min(Math.max(p.x, 0), Math.max(0, w - ICON_W)),
   196→        y: Math.min(Math.max(p.y, 0), Math.max(0, h - ICON_H)),
   197→      };
   198→    },
   199→    [rect],
   200→  );
   201→
   202→  /* default slot = column-major grid anchored to the start edge (right in RTL, left in LTR) */
   203→  const defaultPos = React.useCallback(
   204→    (index: number): { x: number; y: number } => {
   205→      const h = rect?.h ?? (typeof window !== "undefined" ? Math.max(1, window.innerHeight - 68) : 732);
   206→      const w = rect?.w ?? (typeof window !== "undefined" ? Math.max(1, window.innerWidth - 16) : 1264);
   207→      const rows = Math.max(1, Math.floor((h - GRID_PAD) / GRID_STEP_Y));
   208→      const col = Math.floor(index / rows);
   209→      const row = index % rows;
   210→      const x = dir === "rtl" ? w - GRID_PAD - ICON_W - col * GRID_STEP_X : GRID_PAD + col * GRID_STEP_X;
   211→      return { x, y: GRID_PAD + row * GRID_STEP_Y };
   212→    },
   213→    [rect, dir],
   214→  );
   215→
   216→  /* Task 35 — the BusinessStructure shortcut left the desktop: the form
   217→     moved into the Settings window as a sub-page (owner request). Icons:
   218→     the Task-24/Task-30 form-shell samples, then the catalog apps minus
   219→     userManagement — it moved into the Settings window too (Task 37). */
   220→  const items = React.useMemo<IconEntry[]>(() => {
   221→    const base: IconEntry[] = [
   222→      { id: "form-demo", kind: "form-demo", label: t("fdf.iconLabel") },
   223→      { id: "tree-demo", kind: "tree-demo", label: t("stf.iconLabel") },
   224→      ...apps
   225→        .filter((a) => a.appKey !== "userManagement")
   226→        .map((a) => ({
   227→          id: a.appKey,
   228→          kind: "app" as WindowKind,
   229→          appKey: a.appKey,
   230→          label: t(a.titleKey as never),
   231→        })),
   232→    ];
   233→    /* Task 39 — owner request: order by the active language's collation in
   234→       BOTH presentations, so the desktop grid and the sidebar always tell
   235→       the same story (labels are unique, the id is just a stable tiebreak). */
   236→    return base.sort(
   237→      (a, b) => collator.compare(a.label, b.label) || collator.compare(a.id, b.id),
   238→    );
   239→  }, [apps, t, collator]);
   240→
   241→  /* Task 36 — the catalog "settings" icon opens the real Settings window too:
   242→     openWindow redirects (app, "settings") → kind "settings", so this icon,
   243→     the taskbar pin and the start-menu tile all share one Settings window. */
   244→  const activate = (entry: IconEntry) => openWindow(entry.kind, entry.appKey);
   245→
   246→  /* Task 39 — self-heal: a persisted icon layout whose icons visually overlap
   247→     (a legacy pile, or positions clamped together by a resize) is discarded in
   248→     favor of the clean language-sorted grid. Deliberate drag positions that do
   249→     NOT overlap are left untouched, and plain re-renders (selection, drag
   250→     ghost…) never re-evaluate this — only grid (re)mounts and real layer
   251→     geometry changes do. */
   252→  React.useEffect(() => {
   253→    if (iconMode !== "grid" || !freeMode || !rect) return;
   254→    const saved = useDesktop.getState().iconPositions;
   255→    const pts = items.map((entry, i) => clampPos(saved[entry.id] ?? defaultPos(i)));
   256→    for (let a = 0; a < pts.length; a++) {
   257→      for (let b = a + 1; b < pts.length; b++) {
   258→        const p = pts[a];
   259→        const q = pts[b];
   260→        if (p.x < q.x + ICON_W && q.x < p.x + ICON_W && p.y < q.y + ICON_H && q.y < p.y + ICON_H) {
   261→          resetIconLayout();
   262→          return;
   263→        }
   264→      }
   265→    }
   266→  }, [iconMode, freeMode, rect, items, clampPos, defaultPos, resetIconLayout]);
   267→
   268→  const beginDragTracking = (e: React.PointerEvent, entry: IconEntry) => {
   269→    if (!freeMode || e.button !== 0) return;
   270→    const btn = (e.currentTarget as HTMLElement).getBoundingClientRect();
   271→    const grab = { x: e.clientX - btn.left, y: e.clientY - btn.top };
   272→    const startX = e.clientX;
   273→    const startY = e.clientY;
   274→    const pointerId = e.pointerId;
   275→    const taskbarTop =
   276→      document.querySelector<HTMLElement>("[data-taskbar]")?.getBoundingClientRect().top ??
   277→      (typeof window !== "undefined" ? window.innerHeight - 52 : 748);
   278→    let active = false;
   279→    let over = false;
   280→
   281→    const onMove = (ev: PointerEvent) => {
   282→      if (ev.pointerId !== pointerId) return;
   283→      if (!active) {
   284→        if (Math.hypot(ev.clientX - startX, ev.clientY - startY) < DRAG_THRESHOLD) return;
   285→        active = true;
   286→        setSelected(entry.id);
   287→      }
   288→      const o = ev.clientY >= taskbarTop - 12;
   289→      if (o !== over) {
   290→        over = o;
   291→        setDragOverTaskbar(o);
   292→      }
   293→      setGhost({ entry, x: ev.clientX - grab.x, y: ev.clientY - grab.y, overTaskbar: o });
   294→    };
   295→
   296→    const finish = (ev: PointerEvent | null, commit: boolean) => {
   297→      window.removeEventListener("pointermove", onMove);
   298→      window.removeEventListener("pointerup", onUp);
   299→      window.removeEventListener("pointercancel", onCancel);
   300→      setDragOverTaskbar(false);
   301→      setGhost(null);
   302→      if (!active) return; /* plain click — let onClick/dblclick behave normally */
   303→      suppressClickRef.current = true;
   304→      window.setTimeout(() => {
   305→        suppressClickRef.current = false;
   306→      }, 0);
   307→      if (!commit || !ev) return;
   308→      if (over) {
   309→        if (pinToTaskbar(entry.id)) toast.success(t("taskbar.pinnedToast"));
   310→        else toast.info(t("taskbar.alreadyPinned"));
   311→        return;
   312→      }
   313→      const layer = layerRef.current?.getBoundingClientRect();
   314→      if (!layer) return;
   315→      setIconPos(
   316→        entry.id,
   317→        clampPos({
   318→          x: ev.clientX - grab.x - layer.left,
   319→          y: ev.clientY - grab.y - layer.top,
   320→        }),
   321→      );
   322→    };
   323→
   324→    const onUp = (ev: PointerEvent) => finish(ev, true);
   325→    const onCancel = () => finish(null, false);
   326→
   327→    window.addEventListener("pointermove", onMove);
   328→    window.addEventListener("pointerup", onUp);
   329→    window.addEventListener("pointercancel", onCancel);
   330→  };
   331→
   332→  const handleClick = (e: React.MouseEvent, entry: IconEntry) => {
   333→    if (suppressClickRef.current) return;
   334→    setSelected(entry.id);
   335→    /* touch devices: single tap opens (desktop convention: double-click) */
   336→    if (window.matchMedia?.("(pointer: coarse)").matches) activate(entry);
   337→  };
   338→
   339→  const handleContextMenu = (e: React.MouseEvent, entry: IconEntry) => {
   340→    e.preventDefault();
   341→    setSelected(entry.id);
   342→    setMenu({ x: e.clientX, y: e.clientY, entry, source: "grid" });
   343→  };
   344→
   345→  const iconGlyph = (entry: IconEntry) =>
   346→    entry.id === "form-demo" ? (
   347→      <span className="flex h-12 w-12 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-teal-500 to-emerald-600 text-white shadow-lg">
   348→        <FilePlus2 className="h-6 w-6" />
   349→      </span>
   350→    ) : entry.id === "tree-demo" ? (
   351→      <span className="flex h-12 w-12 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-amber-500 to-orange-600 text-white shadow-lg">
   352→        <ListTree className="h-6 w-6" />
   353→      </span>
   354→    ) : (
   355→      <AppTile appKey={entry.appKey ?? ""} className="h-12 w-12 rounded-2xl" iconClass="h-6 w-6" />
   356→    );
   357→
   358→  /* Task 38 — sidebar geometry: the panel docks on the inline-start edge,
   359→     sliding clear of a dashboard header docked on the same side (§1.4:
   360→     logical positioning only — no physical classes). */
   361→  const sidebarOpen = iconMode === "sidebar";
   362→  const startDockedHeader =
   363→    (dir === "rtl" && headerDock === "right") || (dir === "ltr" && headerDock === "left");
   364→  const startOffset = startDockedHeader ? HEADER_THICK + GRID_PAD : GRID_PAD;
   365→  const topOffset = headerDock === "top" ? HEADER_THICK + GRID_PAD : GRID_PAD;
   366→  const SIDEBAR_W = "min(272px, 100vw - 72px)";
   367→  const toggleStart = sidebarOpen
   368→    ? `calc(${startOffset}px + ${sidebarCollapsed ? "64px" : SIDEBAR_W} + 8px)`
   369→    : `${startOffset + 4}px`;
   370→  const ToggleIcon = sidebarOpen
   371→    ? dir === "rtl"
   372→      ? PanelRightClose
   373→      : PanelLeftClose
   374→    : dir === "rtl"
   375→      ? PanelRightOpen
   376→      : PanelLeftOpen;
   377→  const SidebarCollapseIcon = dir === "rtl" ? ChevronRight : ChevronLeft;
   378→  const SidebarExpandIcon = dir === "rtl" ? ChevronLeft : ChevronRight;
   379→
   380→  /* Task 38 — sidebar icon style: compact tightly-rounded tiles (vs the
   381→     desktop's large rounded glyphs) so the presentation visibly changes
   382→     when switching modes (owner request). */
   383→  const sidebarGlyph = (entry: IconEntry) =>
   384→    entry.id === "form-demo" ? (
   385→      <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br from-teal-500 to-emerald-600 text-white shadow-md">
   386→        <FilePlus2 className="h-4 w-4" />
   387→      </span>
   388→    ) : entry.id === "tree-demo" ? (
   389→      <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br from-amber-500 to-orange-600 text-white shadow-md">
   390→        <ListTree className="h-4 w-4" />
   391→      </span>
   392→    ) : (
   393→      <AppTile appKey={entry.appKey ?? ""} className="h-8 w-8 rounded-lg" iconClass="h-4 w-4" />
   394→    );
   395→
   396→  return (
   397→    <>
   398→      {iconMode === "grid" && (
   399→      <div
   400→        ref={layerRef}
   401→        aria-label={t("startmenu.pinned")}
   402→      style={{
   403→        top: headerDock === "top" ? HEADER_THICK + GRID_PAD : GRID_PAD,
   404→        bottom: 60,
   405→        left: headerDock === "left" ? HEADER_THICK + GRID_PAD : GRID_PAD,
   406→        right: headerDock === "right" ? HEADER_THICK + GRID_PAD : GRID_PAD,
   407→      }}
   408→      className={cn(
   409→        "absolute z-0 select-none",
   410→        freeMode
   411→          ? "block"
   412→          : "flex flex-row flex-wrap content-start gap-0.5 sm:flex-col sm:flex-wrap",
   413→      )}
   414→    >
   415→      {items.map((entry, i) => {
   416→        const pos = freeMode ? clampPos(iconPositions[entry.id] ?? defaultPos(i)) : null;
   417→        const dragging = ghost?.entry.id === entry.id;
   418→        return (
   419→          <button
   420→            key={entry.id}
   421→            aria-pressed={selected === entry.id}
   422→            onClick={(e) => handleClick(e, entry)}
   423→            onDoubleClick={() => activate(entry)}
   424→            onContextMenu={(e) => handleContextMenu(e, entry)}
   425→            onPointerDown={(e) => beginDragTracking(e, entry)}
   426→            style={pos ? { left: pos.x, top: pos.y } : undefined}
   427→            className={cn(
   428→              "group flex w-[84px] shrink-0 flex-col items-center gap-1.5 rounded-xl p-2 transition-colors hover:bg-white/12 focus:outline-none cursor-pointer",
   429→              freeMode && "absolute touch-none",
   430→              selected === entry.id && "bg-white/18 ring-1 ring-white/30",
   431→              dragging && "opacity-30",
   432→            )}
   433→          >
   434→            {iconGlyph(entry)}
   435→            <span className="line-clamp-2 rounded-md px-1 text-center text-[11px] font-medium leading-tight text-white drop-shadow group-focus-visible:bg-white/20">
   436→              {entry.label}
   437→            </span>
   438→          </button>
   439→        );
   440→      })}
   441→
   442→      {/* drag ghost following the pointer (fixed overlay, never intercepts events) */}
   443→      {ghost && (
   444→        <div aria-hidden className="pointer-events-none fixed z-[9600] w-[84px]" style={{ left: ghost.x, top: ghost.y }}>
   445→          <div
   446→            className={cn(
   447→              "flex flex-col items-center gap-1.5 rounded-xl p-2 transition-transform duration-100",
   448→              ghost.overTaskbar ? "scale-[0.62] opacity-90" : "scale-105 opacity-95 drop-shadow-2xl",
   449→            )}
   450→          >
   451→            {iconGlyph(ghost.entry)}
   452→            <span className="line-clamp-2 rounded-md bg-black/30 px-1 text-center text-[11px] font-medium leading-tight text-white">
   453→              {ghost.entry.label}
   454→            </span>
   455→          </div>
   456→        </div>
   457→      )}
   458→
   459→      </div>
   460→      )}
   461→
   462→      {/* Task 38 — collapsible acrylic sidebar fed by the very same icon
   463→          entries (owner request): compact tiles + labels, rail/expanded
   464→          states, single-click launch, same context-menu contract. */}
   465→      {sidebarOpen && (
   466→        <aside
   467→          aria-label={t("sidebar.title")}
   468→          style={{
   469→            insetInlineStart: startOffset,
   470→            top: topOffset,
   471→            bottom: 60,
   472→            width: sidebarCollapsed ? 64 : SIDEBAR_W,
   473→          }}
   474→          className="acrylic-strong menu-in win-shadow absolute z-[80] flex flex-col overflow-hidden rounded-2xl border border-border/60 shadow-xl transition-[width] duration-300"
   475→        >
   476→          {/* header: title + count + rail toggle */}
   477→          <div className="flex h-12 shrink-0 items-center gap-1 border-b border-border/60 px-2">
   478→            {!sidebarCollapsed && (
   479→              <>
   480→                <span className="min-w-0 flex-1 truncate px-1 text-xs font-semibold text-foreground/90">
   481→                  {t("sidebar.title")}
   482→                </span>
   483→                <Badge variant="secondary" className="shrink-0 font-mono text-[10px]">
   484→                  {items.length}
   485→                </Badge>
   486→              </>
   487→            )}
   488→            <button
   489→              type="button"
   490→              onClick={() => setSidebarCollapsed(!sidebarCollapsed)}
   491→              aria-expanded={!sidebarCollapsed}
   492→              aria-label={sidebarCollapsed ? t("sidebar.expand") : t("sidebar.collapse")}
   493→              title={sidebarCollapsed ? t("sidebar.expand") : t("sidebar.collapse")}
   494→              className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg text-muted-foreground transition-colors hover:bg-foreground/10 hover:text-foreground cursor-pointer"
   495→            >
   496→              {sidebarCollapsed ? (
   497→                <SidebarExpandIcon className="h-4 w-4" aria-hidden />
   498→              ) : (
   499→                <SidebarCollapseIcon className="h-4 w-4" aria-hidden />
   500→              )}
   501→            </button>
   502→          </div>
   503→
   504→          {/* items — same entries as the desktop grid, new presentation */}
   505→          <nav className="min-h-0 flex-1 overflow-y-auto scrollbar-slim p-2">
   506→            <ul className={cn("flex flex-col gap-0.5", sidebarCollapsed && "items-center")}>
   507→              {items.map((entry) => (
   508→                <li key={entry.id} className="w-full">
   509→                  <button
   510→                    type="button"
   511→                    aria-pressed={selected === entry.id}
   512→                    title={sidebarCollapsed ? entry.label : undefined}
   513→                    onClick={() => {
   514→                      setSelected(entry.id);
   515→                      activate(entry);
   516→                    }}
   517→                    onContextMenu={(e) => {
   518→                      e.preventDefault();
   519→                      setSelected(entry.id);
   520→                      setMenu({ x: e.clientX, y: e.clientY, entry, source: "sidebar" });
   521→                    }}
   522→                    className={cn(
   523→                      sidebarCollapsed
   524→                        ? "flex h-10 w-10 items-center justify-center rounded-xl"
   525→                        : "flex w-full items-center gap-2.5 rounded-xl p-2 text-start",
   526→                      "transition-colors hover:bg-foreground/8 cursor-pointer focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",
   527→                      selected === entry.id && "bg-foreground/10 ring-1 ring-foreground/15",
   528→                    )}
   529→                  >
   530→                    {sidebarGlyph(entry)}
   531→                    {!sidebarCollapsed && (
   532→                      <span className="min-w-0 truncate text-xs font-medium text-foreground/90">
   533→                        {entry.label}
   534→                      </span>
   535→                    )}
   536→                  </button>
   537→                </li>
   538→              ))}
   539→            </ul>
   540→          </nav>
   541→        </aside>
   542→      )}
   543→
   544→      {/* Task 38 — floating view toggle: opens the sidebar / restores the
   545→          desktop icon grid; rides the sidebar edge with a smooth transition.
   546→          Task 40 — owner request: the button shows ONLY its icon; the title
   547→          lives in a beautiful shadcn/ui tooltip (instead of inline text and
   548→          the native title attribute). */}
   549→      <Tooltip>
   550→        <TooltipTrigger asChild>
   551→          <button
   552→            type="button"
   553→            onClick={() => {
   554→              if (!sidebarOpen) {
   555→                setIconMode("sidebar");
   556→                return;
   557→              }
   558→              /* Task 39 — owner request: returning to the desktop icon view must
   559→                 never resurrect a stale/piled layout. The icons are laid out
   560→                 fresh: sorted by the active language, sequential columns anchored
   561→                 to the start edge (right in RTL, left in LTR) — no manual
   562→                 rearranging needed. Free dragging still works inside the grid
   563→                 session; the next round-trip re-arranges again. */
   564→              resetIconLayout();
   565→              setIconMode("grid");
   566→            }}
   567→            aria-pressed={sidebarOpen}
   568→            aria-label={sidebarOpen ? t("sidebar.toGrid") : t("sidebar.toSidebar")}
   569→            style={{ bottom: 64, insetInlineStart: toggleStart }}
   570→            className="acrylic-strong win-shadow absolute z-[9300] flex h-10 w-10 items-center justify-center rounded-full border text-foreground/90 shadow-lg transition-[inset-inline-start] duration-300 hover:bg-foreground/10 cursor-pointer"
   571→          >
   572→            <ToggleIcon className="h-4 w-4 shrink-0" aria-hidden />
   573→          </button>
   574→        </TooltipTrigger>
   575→        <TooltipContent side="top" sideOffset={10} className="rounded-full px-3.5 font-medium">
   576→          {sidebarOpen ? t("sidebar.toGrid") : t("sidebar.toSidebar")}
   577→        </TooltipContent>
   578→      </Tooltip>
   579→
   580→      {/* icon context menu: open / pin–unpin taskbar / reset layout */}
   581→      {menu && (
   582→        <FloatingMenu
   583→          x={menu.x}
   584→          y={menu.y}
   585→          onClose={() => setMenu(null)}
   586→          items={[
   587→            {
   588→              label: t("ctx.open"),
   589→              icon: <ExternalLink className="h-3.5 w-3.5" />,
   590→              onSelect: () => activate(menu.entry),
   591→            },
   592→            pinned.includes(menu.entry.id)
   593→              ? {
   594→                  label: t("ctx.unpinTaskbar"),
   595→                  icon: <PinOff className="h-3.5 w-3.5" />,
   596→                  onSelect: () => {
   597→                    useDesktop.getState().unpinFromTaskbar(menu.entry.id);
   598→                    toast.success(t("taskbar.unpinnedToast"));
   599→                  },
   600→                }
   601→              : {
   602→                  label: t("ctx.pinTaskbar"),
   603→                  icon: <Pin className="h-3.5 w-3.5" />,
   604→                  onSelect: () => {
   605→                    if (pinToTaskbar(menu.entry.id)) toast.success(t("taskbar.pinnedToast"));
   606→                    else toast.info(t("taskbar.alreadyPinned"));
   607→                  },
   608→                },
   609→            ...(freeMode && menu.source === "grid"
   610→              ? [
   611→                  {
   612→                    label: t("ctx.resetLayout"),
   613→                    icon: <RotateCcw className="h-3.5 w-3.5" />,
   614→                    onSelect: () => resetIconLayout(),
   615→                  },
   616→                ]
   617→              : []),
   618→          ]}
   619→        />
   620→      )}
   621→    </>
   622→  );
   623→}
