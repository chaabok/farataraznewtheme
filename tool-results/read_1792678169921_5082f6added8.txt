     1→'use client';
     2→
     3→import {
     4→  useCallback,
     5→  useEffect,
     6→  useMemo,
     7→  useState,
     8→  type ReactNode,
     9→} from 'react';
    10→import { AnimatePresence, motion } from 'framer-motion';
    11→import { ChevronUp, List, Loader2, RefreshCw, Trash2 } from 'lucide-react';
    12→import {
    13→  AlertDialog,
    14→  AlertDialogAction,
    15→  AlertDialogCancel,
    16→  AlertDialogContent,
    17→  AlertDialogDescription,
    18→  AlertDialogFooter,
    19→  AlertDialogHeader,
    20→  AlertDialogTitle,
    21→} from '@/components/ui/alert-dialog';
    22→import { Skeleton } from '@/components/ui/skeleton';
    23→import { toast } from '@/hooks/use-toast';
    24→import { useTranslation } from '@/i18n';
    25→import { apiFetch, SessionExpiredError } from '@/lib/api-client';
    26→import type { FormWindowInfo } from '@/store/desktop-store';
    27→import type {
    28→  FormDataMap,
    29→  FormDef,
    30→  FormFieldValue,
    31→  FormRecord,
    32→} from '@/types/forms';
    33→import { cn } from '@/lib/utils';

    35→import { FieldRenderer } from './field-renderer';
    36→import {
    37→  DEFAULT_FORM_ACCENT,
    38→  fieldsGridClasses,
    39→} from './form-styles';
    40→import { FormFooter, type FormStatus } from './form-footer';
    41→import { FormShell } from './form-shell';
    42→import { FormTabs } from './form-tabs';
    43→import {
    44→  buildDefaults,
    45→  deriveTabs,
    46→  fieldLabel,
    47→  formTitle,
    48→  isEmptyRequiredValue,
    49→  isChangedValue,
    50→  formatFieldValue,
    51→  pickListColumns,
    52→} from './form-utils';
    53→import { TreePanel } from './tree-panel';
    54→
    55→export interface DataFormWindowProps {
    56→  /** Window id — `form:<formKey>` (moduleKey defaults to `settings`). */
    57→  moduleId: string;
    58→  /** Descriptor attached by `openFormWindow` in the desktop store. */
    59→  form?: FormWindowInfo;
    60→  /** Closes the hosting window ("save & close" / "delete & close"). */
    61→  onClose?: () => void;
    62→}
    63→
    64→/** Which combined action a pending delete confirmation belongs to. */
    65→type DeleteIntent = 'current' | 'new' | 'close' | null;
    66→
    67→/** Outcome of the shared save pipeline. */
    68→type SaveResult = 'saved' | 'clean' | 'invalid' | 'error';
    69→
    70→/** Fallback parsing of the window id when no descriptor is attached. */
    71→function parseModuleId(moduleId: string): {
    72→  moduleKey: string | null;
    73→  formKey: string | null;
    74→} {
    75→  if (!moduleId.startsWith('form:')) {
    76→    return { moduleKey: null, formKey: null };
    77→  }
    78→  const formKey = moduleId.slice('form:'.length);
    79→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    80→}
    81→
    82→/** Sentinel kept in state so load errors render through the dictionaries. */
    83→const GENERIC_LOAD_ERROR = 'formWindow.loadError';
    84→
    85→/**
    86→ * Generic, database-driven form window: loads a FormDef + its records and
    87→ * drives the whole view/new/edit state machine (dirty tracking, CRUD,
    88→ * record navigation, Ctrl+S, records drawer) through the shared form
    89→ * component library. One implementation renders every ERP form.
    90→ */
    91→export function DataFormWindow({ moduleId, form, onClose }: DataFormWindowProps) {
    92→  const { locale, direction, t, tf, formatNumber } = useTranslation();
    93→
    94→  const fallback = useMemo(() => parseModuleId(moduleId), [moduleId]);
    95→  const moduleKey = form?.moduleKey ?? fallback.moduleKey;
    96→  const formKey = form?.formKey ?? fallback.formKey;
    97→  const basePath =
    98→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
    99→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   100→
   101→  // ── Data ──
   102→  const [def, setDef] = useState<FormDef | null>(null);
   103→  const [records, setRecords] = useState<FormRecord[]>([]);
   104→  const [isLoading, setIsLoading] = useState(true);
   105→  const [loadError, setLoadError] = useState<string | null>(null);
   106→
   107→  // ── Record state machine ──
   108→  const [mode, setMode] = useState<FormMode>('view');
   109→  const [currentIndex, setCurrentIndex] = useState(-1);
   110→  const [draft, setDraft] = useState<FormDataMap>({});
   111→  const [baseline, setBaseline] = useState<FormDataMap>({});
   112→  const [isSaving, setIsSaving] = useState(false);
   113→  const [isDeleting, setIsDeleting] = useState(false);
   114→  /**
   115→   * Pending delete request awaiting the confirm dialog — `current` =
   116→   * delete and stay, `new` = delete then reset for a fresh entry,
   117→   * `close` = delete then close the window.
   118→   */
   119→  const [deleteIntent, setDeleteIntent] = useState<DeleteIntent>(null);
   120→  const [isListOpen, setIsListOpen] = useState(false);
   121→  const [activeTabKey, setActiveTabKey] = useState<string | null>(null);
   122→
   123→  const current =
   124→    currentIndex >= 0 && currentIndex < records.length
   125→      ? records[currentIndex]
   126→      : null;
   127→
   128→  const isDirty = useMemo(() => {
   129→    const keys = new Set([...Object.keys(baseline), ...Object.keys(draft)]);
   130→    for (const key of keys) {
   131→      if (isChangedValue(baseline[key], draft[key])) return true;
   132→    }
   133→    return false;
   134→  }, [baseline, draft]);
   135→
   136→  const status: FormStatus =
   137→    mode === 'new' ? (isDirty ? 'unsaved' : 'new') : isDirty ? 'unsaved' : 'saved';
   138→
   139→  // ── Loading ──
   140→  const load = useCallback(async () => {
   141→    if (!basePath) {
   142→      setLoadError(GENERIC_LOAD_ERROR);
   143→      setIsLoading(false);
   144→      return;
   145→    }
   146→    setIsLoading(true);
   147→    setLoadError(null);
   148→    try {
   149→      const [defRes, recordsRes] = await Promise.all([
   150→        apiFetch<{ data: FormDef }>(
   151→          `${basePath}?locale=${encodeURIComponent(locale)}`,
   152→        ),
   153→        apiFetch<{ data: FormRecord[] }>(`${basePath}/records`),
   154→      ]);
   155→      const nextDef = defRes.data;
   156→      const nextRecords = recordsRes.data ?? [];
   157→      setDef(nextDef);
   158→      setRecords(nextRecords);
   159→
   160→      if (nextRecords.length) {
   161→        const lastIndex = nextRecords.length - 1;
   162→        const last = nextRecords[lastIndex];
   163→        setCurrentIndex(lastIndex);
   164→        setDraft({ ...last.data });
   165→        setBaseline({ ...last.data });
   166→        setMode('view');
   167→      } else {
   168→        const defaults = buildDefaults(nextDef.fields);
   169→        setCurrentIndex(-1);
   170→        setDraft({ ...defaults });
   171→        setBaseline({ ...defaults });
   172→        setMode('new');
   173→      }
   174→
   175→      const firstTab = deriveTabs(nextDef, locale)[0] ?? null;
   176→      setActiveTabKey(firstTab ? firstTab.key : null);
   177→    } catch (error) {
   178→      if (error instanceof SessionExpiredError) return;
   179→      setLoadError(
   180→        error instanceof Error && error.message
   181→          ? error.message
   182→          : GENERIC_LOAD_ERROR,
   183→      );
   184→    } finally {
   185→      setIsLoading(false);
   186→    }
   187→  }, [basePath, locale]);
   188→
   189→  useEffect(() => {
   190→    void load();
   191→  }, [load]);
   192→
   193→  const refreshRecords = useCallback(async (): Promise<FormRecord[]> => {
   194→    if (!basePath) return [];
   195→    const res = await apiFetch<{ data: FormRecord[] }>(`${basePath}/records`);
   196→    const list = res.data ?? [];
   197→    setRecords(list);
   198→    return list;
   199→  }, [basePath]);
   200→
   201→  // ── Record transitions ──
   202→  const applyRecord = useCallback((index: number, list: FormRecord[]) => {
   203→    const target = list[index];
   204→    setCurrentIndex(index);
   205→    setMode('view');
   206→    if (target) {
   207→      setDraft({ ...target.data });
   208→      setBaseline({ ...target.data });
   209→    }
   210→  }, []);
   211→
   212→  const navigateTo = useCallback(
   213→    (index: number) => {
   214→      if (isSaving || isDeleting) return;
   215→      if (index < 0 || index >= records.length) return;
   216→      if (mode !== 'view' && isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) {
   217→        return;
   218→      }
   219→      applyRecord(index, records);
   220→    },
   221→    [applyRecord, isDeleting, isDirty, isSaving, mode, records, t],
   222→  );
   223→
   224→  const startNew = useCallback(
   225→    (preset?: FormDataMap) => {
   226→      if (!def) return;
   227→      const base = { ...buildDefaults(def.fields), ...(preset ?? {}) };
   228→      setDraft(base);
   229→      setBaseline({ ...base });
   230→      setMode('new');
   231→      setCurrentIndex(-1);
   232→    },
   233→    [def],
   234→  );
   235→
   236→  const handleNew = useCallback(() => {
   237→    if (mode === 'new') return;
   238→    if (isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) return;
   239→    startNew();
   240→  }, [isDirty, mode, startNew, t]);
   241→
   242→  const handleCopy = useCallback(() => {
   243→    if (mode !== 'view' || !current) return;
   244→    startNew({ ...current.data });
   245→    toast({ title: t('formWindow.copyToast') });
   246→  }, [current, mode, startNew, t]);
   247→
   248→  /**
   249→   * Shared save pipeline for every save action: validates required
   250→   * fields, writes the draft to the database and refreshes the record
   251→   * list. `keepRecord: false` skips re-selecting the saved record
   252→   * (used by "save & new", which resets the form right after).
   253→   */
   254→  const performSave = useCallback(
   255→    async (options?: { keepRecord?: boolean }): Promise<SaveResult> => {
   256→      if (!def || !basePath || isSaving || isDeleting) return 'error';
   257→      if (mode === 'view' || !isDirty) return 'clean';
   258→
   259→      const missingField = def.fields.find((field) =>
   260→        isEmptyRequiredValue(field, draft[field.fieldKey]),
   261→      );
   262→      if (missingField) {
   263→        toast({
   264→          title: fieldLabel(missingField, locale),
   265→          description: t('formWindow.requiredHint'),
   266→          variant: 'destructive',
   267→        });
   268→        return 'invalid';
   269→      }
   270→
   271→      setIsSaving(true);
   272→      try {
   273→        const saved =
   274→          mode === 'new'
   275→            ? (
   276→                await apiFetch<{ data: FormRecord }>(`${basePath}/records`, {
   277→                  method: 'POST',
   278→                  body: { data: draft },
   279→                })
   280→              ).data
   281→            : (
   282→                await apiFetch<{ data: FormRecord }>(
   283→                  `${basePath}/records/${encodeURIComponent(current?.id ?? '')}`,
   284→                  { method: 'PUT', body: { data: draft } },
   285→                )
   286→              ).data;
   287→
   288→        const list = await refreshRecords();
   289→        if (options?.keepRecord !== false) {
   290→          const index = list.findIndex((record) => record.id === saved.id);
   291→          if (index >= 0) {
   292→            applyRecord(index, list);
   293→          } else {
   294→            setMode('view');
   295→          }
   296→        }
   297→        return 'saved';
   298→      } catch (error) {
   299→        if (!(error instanceof SessionExpiredError)) {
   300→          toast({
   301→            title: t('formWindow.saveError'),
   302→            description: error instanceof Error ? error.message : undefined,
   303→            variant: 'destructive',
   304→          });
   305→        }
   306→        return 'error';
   307→      } finally {
   308→        setIsSaving(false);
   309→      }
   310→    },
   311→    [
   312→      applyRecord,
   313→      basePath,
   314→      current,
   315→      def,
   316→      draft,
   317→      isDeleting,
   318→      isDirty,
   319→      isSaving,
   320→      locale,
   321→      mode,
   322→      refreshRecords,
   323→      t,
   324→    ],
   325→  );
   326→
   327→  const handleSave = useCallback(async () => {
   328→    const result = await performSave();
   329→    if (result === 'saved') toast({ title: t('formWindow.saveSuccess') });
   330→  }, [performSave, t]);
   331→
   332→  /** Save & new: write the record, then reset the form for a fresh entry. */
   333→  const handleSaveNew = useCallback(async () => {
   334→    if (isSaving || isDeleting) return;
   335→    const result = await performSave({ keepRecord: false });
   336→    if (result === 'error' || result === 'invalid') return;
   337→    startNew();
   338→    if (result === 'saved') toast({ title: t('formWindow.saveNewSuccess') });
   339→  }, [isDeleting, isSaving, performSave, startNew, t]);
   340→
   341→  /** Save & close: write the record after validation, then close. */
   342→  const handleSaveClose = useCallback(async () => {
   343→    if (isSaving || isDeleting) return;
   344→    const result = await performSave();
   345→    if (result === 'error' || result === 'invalid') return;
   346→    if (result === 'saved') toast({ title: t('formWindow.saveCloseSuccess') });
   347→    onClose?.();
   348→  }, [isDeleting, isSaving, onClose, performSave, t]);
   349→
   350→  /** First click on any delete action opens the confirm dialog. */
   351→  const requestDelete = useCallback(
   352→    (intent: Exclude<DeleteIntent, null>) => {
   353→      if (!current || mode === 'new' || isSaving || isDeleting) return;
   354→      setDeleteIntent(intent);
   355→    },
   356→    [current, isDeleting, isSaving, mode],
   357→  );
   358→
   359→  /** Runs after the user presses «بلی» in the confirm dialog. */
   360→  const confirmDelete = useCallback(async () => {
   361→    if (isDeleting || !deleteIntent) return;
   362→    const intent = deleteIntent;
   363→    if (!basePath || !current) {
   364→      setDeleteIntent(null);
   365→      return;
   366→    }
   367→    setIsDeleting(true);
   368→    try {
   369→      // DELETE answers 204 with an empty body — nothing to parse.
   370→      await apiFetch(
   371→        `${basePath}/records/${encodeURIComponent(current.id)}`,
   372→        {
   373→          method: 'DELETE',
   374→        },
   375→      );
   376→      setDeleteIntent(null);
   377→      if (intent === 'close') {
   378→        toast({ title: t('formWindow.deleteCloseSuccess') });
   379→        onClose?.();
   380→        return;
   381→      }
   382→      const list = await refreshRecords();
   383→      if (intent === 'new') {
   384→        startNew();
   385→        toast({ title: t('formWindow.deleteNewSuccess') });
   386→      } else if (list.length) {
   387→        applyRecord(Math.min(currentIndex, list.length - 1), list);
   388→        toast({ title: t('formWindow.deleteSuccess') });
   389→      } else {
   390→        setCurrentIndex(-1);
   391→        startNew();
   392→        toast({ title: t('formWindow.deleteSuccess') });
   393→      }
   394→    } catch (error) {
   395→      setDeleteIntent(null);
   396→      if (error instanceof SessionExpiredError) return;
   397→      toast({
   398→        title: error instanceof Error && error.message
   399→          ? error.message
   400→          : t('formWindow.loadError'),
   401→        variant: 'destructive',
   402→      });
   403→    } finally {
   404→      setIsDeleting(false);
   405→    }
   406→  }, [
   407→    applyRecord,
   408→    basePath,
   409→    currentIndex,
   410→    current,
   411→    deleteIntent,
   412→    isDeleting,
   413→    onClose,
   414→    refreshRecords,
   415→    startNew,
   416→    t,
   417→  ]);
   418→
   419→  const handlePrint = useCallback(() => {
   420→    toast({ title: t('formWindow.printToast') });
   421→  }, [t]);
   422→
   423→  const handleFieldChange = useCallback(
   424→    (fieldKey: string, value: FormFieldValue) => {
   425→      setDraft((prev) => ({ ...prev, [fieldKey]: value }));
   426→      setMode((prev) => (prev === 'view' ? 'edit' : prev));
   427→    },
   428→    [],
   429→  );
   430→
   431→  // Ctrl+S / Cmd+S saves the active draft (Escape is handled by the dialog).
   432→  useEffect(() => {
   433→    const onKeyDown = (event: KeyboardEvent) => {
   434→      if (
   435→        (event.ctrlKey || event.metaKey) &&
   436→        !event.shiftKey &&
   437→        !event.altKey &&
   438→        event.key.toLowerCase() === 's'
   439→      ) {
   440→        event.preventDefault();
   441→        void handleSave();
   442→      }
   443→    };
   444→    window.addEventListener('keydown', onKeyDown);
   445→    return () => window.removeEventListener('keydown', onKeyDown);
   446→  }, [handleSave]);
   447→
   448→  // ── Derived view models ──
   449→  const sortedFields = useMemo(
   450→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   451→    [def],
   452→  );
   453→
   454→  const tabs = useMemo(() => (def ? deriveTabs(def, locale) : []), [def, locale]);
   455→  const activeTab =
   456→    tabs.find((tab) => (tab.key ?? null) === (activeTabKey ?? null)) ??
   457→    tabs[0] ??
   458→    null;
   459→
   460→  const listColumns = useMemo(
   461→    () => pickListColumns(sortedFields),
   462→    [sortedFields],
   463→  );
   464→
   465→  const title = def
   466→    ? formTitle(def, locale)
   467→    : form?.name ?? t('common.loading');
   468→  const subtitle =
   469→    mode === 'new'
   470→      ? t('formWindow.newMode')
   471→      : current
   472→        ? tf('formWindow.recordOf', {
   473→            n: formatNumber(currentIndex + 1),
   474→            total: formatNumber(records.length),
   475→          })
   476→        : undefined;
   477→
   478→  const accentColor = def?.accentColor || DEFAULT_FORM_ACCENT;
   479→
   480→  /** The fields grid shared by simple forms and the active tab. */
   481→  const renderFields = (fields: typeof sortedFields): ReactNode => (
   482→    <div className={fieldsGridClasses}>
   483→      {fields.map((field) => (
   484→        <FieldRenderer
   485→          key={field.fieldKey}
   486→          field={field}
   487→          value={draft[field.fieldKey] ?? null}
   488→          onChange={(value) => handleFieldChange(field.fieldKey, value)}
   489→          disabled={isSaving || isDeleting}
   490→          lookupUrlBase={lookupUrlBase}
   491→        />
   492→      ))}
   493→    </div>
   494→  );
   495→
   496→  let content: ReactNode;
   497→  if (!def) {
   498→    // Loading skeleton (error handled below).
   499→    content = (
   500→      <div className={fieldsGridClasses}>
   501→        {Array.from({ length: 6 }).map((_, index) => (
   502→          <div key={`form-skeleton-${index}`} className="space-y-1.5">
   503→            <Skeleton className="h-3 w-24 bg-white/[0.08]" />
   504→            <Skeleton className="h-10 w-full rounded-lg bg-white/[0.05]" />
   505→          </div>
   506→        ))}
   507→      </div>
   508→    );
   509→  } else if (def.type === 'tree') {
   510→    content = (
   511→      <TreePanel
   512→        records={records}
   513→        fields={sortedFields}
   514→        lookupUrlBase={lookupUrlBase}
   515→      />
   516→    );
   517→  } else if (def.type === 'tabbed' && tabs.length > 0) {
   518→    content = (
   519→      <FormTabs
   520→        tabs={tabs.map((tab) => ({
   521→          key: tab.key,
   522→          name: tab.name,
   523→          count: tab.fields.length,
   524→        }))}
   525→        active={activeTab?.key ?? null}
   526→        onChange={setActiveTabKey}
   527→        accentColor={accentColor}
   528→        direction={direction}
   529→      >
   530→        {renderFields(activeTab?.fields ?? [])}
   531→      </FormTabs>
   532→    );
   533→  } else {
   534→    content = renderFields(sortedFields);
   535→  }
   536→
   537→  const errorText =
   538→    loadError === GENERIC_LOAD_ERROR ? t('formWindow.loadError') : loadError;
   539→
   540→  return (
   541→    <>
   542→    <FormShell
   543→      title={title}
   544→      subtitle={subtitle}
   545→      iconKey={def?.iconKey ?? form?.iconKey ?? 'fileText'}
   546→      accentColor={accentColor}
   547→      actionBar={
   548→        <FormActionBar
   549→          mode={mode}
   550→          isDirty={isDirty}
   551→          isSaving={isSaving}
   552→          isDeleting={isDeleting}
   553→          hasRecord={Boolean(current)}
   554→          recordIndex={currentIndex}
   555→          recordCount={records.length}
   556→          isListOpen={isListOpen}
   557→          onNew={handleNew}
   558→          onSave={() => void handleSave()}
   559→          onSaveNew={() => void handleSaveNew()}
   560→          onSaveClose={() => void handleSaveClose()}
   561→          onCopy={handleCopy}
   562→          onDelete={() => requestDelete('current')}
   563→          onDeleteNew={() => requestDelete('new')}
   564→          onDeleteClose={() => requestDelete('close')}
   565→          onPrint={handlePrint}
   566→          onToggleList={() => setIsListOpen((open) => !open)}
   567→          onFirst={() => navigateTo(0)}
   568→          onPrev={() => navigateTo(Math.max(0, currentIndex - 1))}
   569→          onNext={() => navigateTo(Math.min(records.length - 1, currentIndex + 1))}
   570→          onLast={() => navigateTo(records.length - 1)}
   571→        />
   572→      }
   573→      subPanel={
   574→        <AnimatePresence initial={false}>
   575→          {isListOpen && (
   576→            <RecordsDrawer
   577→              key="records-drawer"
   578→              records={records}
   579→              currentIndex={currentIndex}
   580→              codeField={listColumns.codeField}
   581→              titleField={listColumns.titleField}
   582→              onJump={navigateTo}
   583→              onClose={() => setIsListOpen(false)}
   584→            />
   585→          )}
   586→        </AnimatePresence>
   587→      }
   588→      footer={
   589→        <FormFooter
   590→          status={status}
   591→          position={
   592→            current && mode !== 'new'
   593→              ? { n: currentIndex + 1, total: records.length }
   594→              : null
   595→          }
   596→          updatedAt={current?.updatedAt ?? null}
   597→        />
   598→      }
   599→    >
   600→      {errorText && !def ? (
   601→        <div className="flex min-h-[220px] flex-col items-center justify-center gap-3 p-6 text-center">
   602→          <p className="max-w-sm text-sm text-red-200/90">{errorText}</p>
   603→          <button
   604→            type="button"
   605→            onClick={() => void load()}
   606→            className="flex items-center gap-2 rounded-lg bg-white/[0.07] px-4 py-2 text-[13px] text-white/80 transition-colors hover:bg-white/[0.12] focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-white/40"
   607→          >
   608→            <RefreshCw className="h-3.5 w-3.5" />
   609→            {t('common.retry')}
   610→          </button>
   611→        </div>
   612→      ) : (
   613→        <>
   614→          {errorText && def && (
   615→            <div className="mx-4 mt-4 rounded-lg border border-red-400/30 bg-red-950/40 px-3 py-2 text-xs text-red-200/90">
   616→              {errorText}
   617→            </div>
   618→          )}
   619→          {records.length === 0 && def && def.type !== 'tree' ? (
   620→            <div className="px-4 pb-2 pt-6 text-center">
   621→              <p className="text-[13px] text-white/55">
   622→                {t('formWindow.emptyRecords')}
   623→              </p>
   624→              <p className="mt-1 text-[11px] text-white/35">
   625→                {t('formWindow.emptyRecordsHint')}
   626→              </p>
   627→            </div>
   628→          ) : null}
   629→          {content}
   630→        </>
   631→      )}
   632→    </FormShell>
   633→
   634→    {/* Single delete-confirmation dialog shared by all three delete actions. */}
   635→    <AlertDialog
   636→      open={deleteIntent !== null}
   637→      onOpenChange={(open) => {
   638→        if (!open) setDeleteIntent(null);
   639→      }}
   640→    >
   641→      <AlertDialogContent className="max-w-[calc(100vw-2rem)] rounded-2xl border border-white/15 bg-[#1e2128]/95 text-white shadow-2xl shadow-black/60 backdrop-blur-xl sm:max-w-md">
   642→        <AlertDialogHeader className="sm:text-start">
   643→          <AlertDialogTitle className="flex items-center gap-2.5 text-base text-white">
   644→            <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-rose-500/15 text-rose-400 ring-1 ring-rose-400/30">
   645→              <Trash2 className="h-4 w-4" aria-hidden="true" />
   646→            </span>
   647→            {t('formWindow.deleteDialogTitle')}
   648→          </AlertDialogTitle>
   649→          <AlertDialogDescription className="text-[13px] leading-relaxed text-white/65">
   650→            {t('formWindow.deleteDialogMessage')}
   651→          </AlertDialogDescription>
   652→        </AlertDialogHeader>
   653→        <AlertDialogFooter className="gap-2 sm:justify-start">
   654→          <AlertDialogCancel
   655→            disabled={isDeleting}
   656→            className="h-9 rounded-lg border-white/15 bg-white/[0.06] px-4 text-[13px] text-white/80 transition-colors hover:bg-white/[0.12] hover:text-white focus-visible:ring-white/40"
   657→          >
   658→            {t('formWindow.confirmNo')}
   659→          </AlertDialogCancel>
   660→          <AlertDialogAction
   661→            disabled={isDeleting}
   662→            onClick={(event) => {
   663→              // Keep the dialog open while the delete request runs.
   664→              event.preventDefault();
   665→              void confirmDelete();
   666→            }}
   667→            className="flex h-9 items-center gap-1.5 rounded-lg bg-rose-500/90 px-4 text-[13px] text-white shadow-sm shadow-rose-950/40 transition-colors hover:bg-rose-500 focus-visible:ring-rose-300/60"
   668→          >
   669→            {isDeleting ? (
   670→              <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" />
   671→            ) : null}
   672→            {t('formWindow.confirmYes')}
   673→          </AlertDialogAction>
   674→        </AlertDialogFooter>

   676→    </AlertDialog>
   677→    </>
   678→  );
