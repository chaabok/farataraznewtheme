     1→'use client';
     2→
     3→import {
     4→  useCallback,
     5→  useEffect,
     6→  useMemo,
     7→  useState,
     8→  type ReactNode,
     9→} from 'react';
    10→import { AnimatePresence, motion } from 'framer-motion';
    11→import { ChevronUp, List, Loader2, RefreshCw, Trash2 } from 'lucide-react';
    12→import {
    13→  AlertDialog,
    14→  AlertDialogAction,
    15→  AlertDialogCancel,
    16→  AlertDialogContent,
    17→  AlertDialogDescription,
    18→  AlertDialogFooter,
    19→  AlertDialogHeader,
    20→  AlertDialogTitle,
    21→} from '@/components/ui/alert-dialog';
    22→import { Skeleton } from '@/components/ui/skeleton';
    23→import { toast } from '@/hooks/use-toast';
    24→import { useTranslation } from '@/i18n';
    25→import { apiFetch, SessionExpiredError } from '@/lib/api-client';
    26→import type { FormWindowInfo } from '@/store/desktop-store';
    27→import type {
    28→  FormDataMap,
    29→  FormDef,
    30→  FormFieldValue,
    31→  FormRecord,
    32→} from '@/types/forms';
    33→import { cn } from '@/lib/utils';
    34→import { FormActionBar, type FormMode } from './action-bar';
    35→import { FieldRenderer } from './field-renderer';
    36→import {
    37→  DEFAULT_FORM_ACCENT,
    38→  fieldsGridClasses,
    39→} from './form-styles';
    40→import { FormFooter, type FormStatus } from './form-footer';
    41→import { FormShell } from './form-shell';
    42→import { FormTabs } from './form-tabs';
    43→import {
    44→  buildDefaults,
    45→  deriveTabs,
    46→  fieldLabel,
    47→  formTitle,
    48→  isEmptyRequiredValue,
    49→  isChangedValue,
    50→  formatFieldValue,
    51→  pickListColumns,
    52→} from './form-utils';
    53→import { TreePanel } from './tree-panel';
    54→
    55→export interface DataFormWindowProps {
    56→  /** Window id — `form:<formKey>` (moduleKey defaults to `settings`). */
    57→  moduleId: string;
    58→  /** Descriptor attached by `openFormWindow` in the desktop store. */
    59→  form?: FormWindowInfo;
    60→  /** Closes the hosting window ("save & close" / "delete & close"). */
    61→  onClose?: () => void;
    62→}
    63→
    64→/** Which combined action a pending delete confirmation belongs to. */
    65→type DeleteIntent = 'current' | 'new' | 'close' | null;
    66→
    67→/** Outcome of the shared save pipeline. */
    68→type SaveResult = 'saved' | 'clean' | 'invalid' | 'error';
    69→
    70→/** Fallback parsing of the window id when no descriptor is attached. */
    71→function parseModuleId(moduleId: string): {
    72→  moduleKey: string | null;
    73→  formKey: string | null;
    74→} {
    75→  if (!moduleId.startsWith('form:')) {
    76→    return { moduleKey: null, formKey: null };
    77→  }
    78→  const formKey = moduleId.slice('form:'.length);
    79→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    80→}
    81→
    82→/** Sentinel kept in state so load errors render through the dictionaries. */
    83→const GENERIC_LOAD_ERROR = 'formWindow.loadError';
    84→
    85→/**
    86→ * Generic, database-driven form window: loads a FormDef + its records and
    87→ * drives the whole view/new/edit state machine (dirty tracking, CRUD,
    88→ * record navigation, Ctrl+S, records drawer) through the shared form
    89→ * component library. One implementation renders every ERP form.
    90→ */
    91→export function DataFormWindow({ moduleId, form, onClose }: DataFormWindowProps) {
    92→  const { locale, direction, t, tf, formatNumber } = useTranslation();
    93→
    94→  const fallback = useMemo(() => parseModuleId(moduleId), [moduleId]);
    95→  const moduleKey = form?.moduleKey ?? fallback.moduleKey;
    96→  const formKey = form?.formKey ?? fallback.formKey;
    97→  const basePath =
    98→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
    99→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   100→
   101→  // ── Data ──
   102→  const [def, setDef] = useState<FormDef | null>(null);
   103→  const [records, setRecords] = useState<FormRecord[]>([]);
   104→  const [isLoading, setIsLoading] = useState(true);
   105→  const [loadError, setLoadError] = useState<string | null>(null);
   106→
   107→  // ── Record state machine ──
   108→  const [mode, setMode] = useState<FormMode>('view');
   109→  const [currentIndex, setCurrentIndex] = useState(-1);
   110→  const [draft, setDraft] = useState<FormDataMap>({});
   111→  const [baseline, setBaseline] = useState<FormDataMap>({});
   112→  const [isSaving, setIsSaving] = useState(false);
   113→  const [isDeleting, setIsDeleting] = useState(false);
   114→  /**
   115→   * Pending delete request awaiting the confirm dialog — `current` =
   116→   * delete and stay, `new` = delete then reset for a fresh entry,
   117→   * `close` = delete then close the window.
   118→   */
   119→  const [deleteIntent, setDeleteIntent] = useState<DeleteIntent>(null);
   120→  const [isListOpen, setIsListOpen] = useState(false);
   121→  const [activeTabKey, setActiveTabKey] = useState<string | null>(null);
   122→
   123→  const current =
   124→    currentIndex >= 0 && currentIndex < records.length
   125→      ? records[currentIndex]
   126→      : null;
   127→
   128→  const isDirty = useMemo(() => {
   129→    const keys = new Set([...Object.keys(baseline), ...Object.keys(draft)]);
   130→    for (const key of keys) {
   131→      if (isChangedValue(baseline[key], draft[key])) return true;
   132→    }
   133→    return false;
   134→  }, [baseline, draft]);
   135→
   136→  const status: FormStatus =
   137→    mode === 'new' ? (isDirty ? 'unsaved' : 'new') : isDirty ? 'unsaved' : 'saved';
   138→
   139→  // ── Loading ──
   140→  const load = useCallback(async () => {
   141→    if (!basePath) {
   142→      setLoadError(GENERIC_LOAD_ERROR);
   143→      setIsLoading(false);
   144→      return;
   145→    }
   146→    setIsLoading(true);
   147→    setLoadError(null);
   148→    try {
   149→      const [defRes, recordsRes] = await Promise.all([
   150→        apiFetch<{ data: FormDef }>(
   151→          `${basePath}?locale=${encodeURIComponent(locale)}`,
   152→        ),
   153→        apiFetch<{ data: FormRecord[] }>(`${basePath}/records`),
   154→      ]);
   155→      const nextDef = defRes.data;
   156→      const nextRecords = recordsRes.data ?? [];
   157→      setDef(nextDef);
   158→      setRecords(nextRecords);
   159→
   160→      if (nextRecords.length) {
   161→        const lastIndex = nextRecords.length - 1;
   162→        const last = nextRecords[lastIndex];
   163→        setCurrentIndex(lastIndex);
   164→        setDraft({ ...last.data });
   165→        setBaseline({ ...last.data });
   166→        setMode('view');
   167→      } else {
   168→        const defaults = buildDefaults(nextDef.fields);
   169→        setCurrentIndex(-1);
   170→        setDraft({ ...defaults });
   171→        setBaseline({ ...defaults });
   172→        setMode('new');
   173→      }
   174→
   175→      const derived = deriveTabs(nextDef, locale);
   176→      // The first group is the always-visible header section; the active
   177→      // tab state only drives the remaining body tabs (default: first one).
   178→      const firstBodyTab = derived[1] ?? null;
   179→      setActiveTabKey(firstBodyTab ? firstBodyTab.key : null);
   180→    } catch (error) {
   181→      if (error instanceof SessionExpiredError) return;
   182→      setLoadError(
   183→        error instanceof Error && error.message
   184→          ? error.message
   185→          : GENERIC_LOAD_ERROR,
   186→      );
   187→    } finally {
   188→      setIsLoading(false);
   189→    }
   190→  }, [basePath, locale]);
   191→
   192→  useEffect(() => {
   193→    void load();
   194→  }, [load]);
   195→
   196→  const refreshRecords = useCallback(async (): Promise<FormRecord[]> => {
   197→    if (!basePath) return [];
   198→    const res = await apiFetch<{ data: FormRecord[] }>(`${basePath}/records`);
   199→    const list = res.data ?? [];
   200→    setRecords(list);
   201→    return list;
   202→  }, [basePath]);
   203→
   204→  // ── Record transitions ──
   205→  const applyRecord = useCallback((index: number, list: FormRecord[]) => {
   206→    const target = list[index];
   207→    setCurrentIndex(index);
   208→    setMode('view');
   209→    if (target) {
   210→      setDraft({ ...target.data });
   211→      setBaseline({ ...target.data });
   212→    }
   213→  }, []);
   214→
   215→  const navigateTo = useCallback(
   216→    (index: number) => {
   217→      if (isSaving || isDeleting) return;
   218→      if (index < 0 || index >= records.length) return;
   219→      if (mode !== 'view' && isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) {
   220→        return;
   221→      }
   222→      applyRecord(index, records);
   223→    },
   224→    [applyRecord, isDeleting, isDirty, isSaving, mode, records, t],
   225→  );
   226→
   227→  const startNew = useCallback(
   228→    (preset?: FormDataMap) => {
   229→      if (!def) return;
   230→      const base = { ...buildDefaults(def.fields), ...(preset ?? {}) };
   231→      setDraft(base);
   232→      setBaseline({ ...base });
   233→      setMode('new');
   234→      setCurrentIndex(-1);
   235→    },
   236→    [def],
   237→  );
   238→
   239→  const handleNew = useCallback(() => {
   240→    if (mode === 'new') return;
   241→    if (isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) return;
   242→    startNew();
   243→  }, [isDirty, mode, startNew, t]);
   244→
   245→  const handleCopy = useCallback(() => {
   246→    if (mode !== 'view' || !current) return;
   247→    startNew({ ...current.data });
   248→    toast({ title: t('formWindow.copyToast') });
   249→  }, [current, mode, startNew, t]);
   250→
   251→  /**
   252→   * Shared save pipeline for every save action: validates required
   253→   * fields, writes the draft to the database and refreshes the record
   254→   * list. `keepRecord: false` skips re-selecting the saved record
   255→   * (used by "save & new", which resets the form right after).
   256→   */
   257→  const performSave = useCallback(
   258→    async (options?: { keepRecord?: boolean }): Promise<SaveResult> => {
   259→      if (!def || !basePath || isSaving || isDeleting) return 'error';
   260→      if (mode === 'view' || !isDirty) return 'clean';
   261→
   262→      const missingField = def.fields.find((field) =>
   263→        isEmptyRequiredValue(field, draft[field.fieldKey]),
   264→      );
   265→      if (missingField) {
   266→        toast({
   267→          title: fieldLabel(missingField, locale),
   268→          description: t('formWindow.requiredHint'),
   269→          variant: 'destructive',
   270→        });
   271→        return 'invalid';
   272→      }
   273→
   274→      setIsSaving(true);
   275→      try {
   276→        const saved =
   277→          mode === 'new'
   278→            ? (
   279→                await apiFetch<{ data: FormRecord }>(`${basePath}/records`, {
   280→                  method: 'POST',
   281→                  body: { data: draft },
   282→                })
   283→              ).data
   284→            : (
   285→                await apiFetch<{ data: FormRecord }>(
   286→                  `${basePath}/records/${encodeURIComponent(current?.id ?? '')}`,
   287→                  { method: 'PUT', body: { data: draft } },
   288→                )
   289→              ).data;
   290→
   291→        const list = await refreshRecords();
   292→        if (options?.keepRecord !== false) {
   293→          const index = list.findIndex((record) => record.id === saved.id);
   294→          if (index >= 0) {
   295→            applyRecord(index, list);
   296→          } else {
   297→            setMode('view');
   298→          }
   299→        }
   300→        return 'saved';
   301→      } catch (error) {
   302→        if (!(error instanceof SessionExpiredError)) {
   303→          toast({
   304→            title: t('formWindow.saveError'),
   305→            description: error instanceof Error ? error.message : undefined,
   306→            variant: 'destructive',
   307→          });
   308→        }
   309→        return 'error';
   310→      } finally {
   311→        setIsSaving(false);
   312→      }
   313→    },
   314→    [
   315→      applyRecord,
   316→      basePath,
   317→      current,
   318→      def,
   319→      draft,
   320→      isDeleting,
   321→      isDirty,
   322→      isSaving,
   323→      locale,
   324→      mode,
   325→      refreshRecords,
   326→      t,
   327→    ],
   328→  );
   329→
   330→  const handleSave = useCallback(async () => {
   331→    const result = await performSave();
   332→    if (result === 'saved') toast({ title: t('formWindow.saveSuccess') });
   333→  }, [performSave, t]);
   334→
   335→  /** Save & new: write the record, then reset the form for a fresh entry. */
   336→  const handleSaveNew = useCallback(async () => {
   337→    if (isSaving || isDeleting) return;
   338→    const result = await performSave({ keepRecord: false });
   339→    if (result === 'error' || result === 'invalid') return;
   340→    startNew();
   341→    if (result === 'saved') toast({ title: t('formWindow.saveNewSuccess') });
   342→  }, [isDeleting, isSaving, performSave, startNew, t]);
   343→
   344→  /** Save & close: write the record after validation, then close. */
   345→  const handleSaveClose = useCallback(async () => {
   346→    if (isSaving || isDeleting) return;
   347→    const result = await performSave();
   348→    if (result === 'error' || result === 'invalid') return;
   349→    if (result === 'saved') toast({ title: t('formWindow.saveCloseSuccess') });
   350→    onClose?.();
   351→  }, [isDeleting, isSaving, onClose, performSave, t]);
   352→
   353→  /** First click on any delete action opens the confirm dialog. */
   354→  const requestDelete = useCallback(
   355→    (intent: Exclude<DeleteIntent, null>) => {
   356→      if (!current || mode === 'new' || isSaving || isDeleting) return;
   357→      setDeleteIntent(intent);
   358→    },
   359→    [current, isDeleting, isSaving, mode],
   360→  );
   361→
   362→  /** Runs after the user presses «بلی» in the confirm dialog. */
   363→  const confirmDelete = useCallback(async () => {
   364→    if (isDeleting || !deleteIntent) return;
   365→    const intent = deleteIntent;
   366→    if (!basePath || !current) {
   367→      setDeleteIntent(null);
   368→      return;
   369→    }
   370→    setIsDeleting(true);
   371→    try {
   372→      // DELETE answers 204 with an empty body — nothing to parse.
   373→      await apiFetch(
   374→        `${basePath}/records/${encodeURIComponent(current.id)}`,
   375→        {
   376→          method: 'DELETE',
   377→        },
   378→      );
   379→      setDeleteIntent(null);
   380→      if (intent === 'close') {
   381→        toast({ title: t('formWindow.deleteCloseSuccess') });
   382→        onClose?.();
   383→        return;
   384→      }
   385→      const list = await refreshRecords();
   386→      if (intent === 'new') {
   387→        startNew();
   388→        toast({ title: t('formWindow.deleteNewSuccess') });
   389→      } else if (list.length) {
   390→        applyRecord(Math.min(currentIndex, list.length - 1), list);
   391→        toast({ title: t('formWindow.deleteSuccess') });
   392→      } else {
   393→        setCurrentIndex(-1);
   394→        startNew();
   395→        toast({ title: t('formWindow.deleteSuccess') });
   396→      }
   397→    } catch (error) {
   398→      setDeleteIntent(null);
   399→      if (error instanceof SessionExpiredError) return;
   400→      toast({
   401→        title: error instanceof Error && error.message
   402→          ? error.message
   403→          : t('formWindow.loadError'),
   404→        variant: 'destructive',
   405→      });
   406→    } finally {
   407→      setIsDeleting(false);
   408→    }
   409→  }, [
   410→    applyRecord,
   411→    basePath,
   412→    currentIndex,
   413→    current,
   414→    deleteIntent,
   415→    isDeleting,
   416→    onClose,
   417→    refreshRecords,
   418→    startNew,
   419→    t,
   420→  ]);
   421→
   422→  const handlePrint = useCallback(() => {
   423→    toast({ title: t('formWindow.printToast') });
   424→  }, [t]);
   425→
   426→  const handleFieldChange = useCallback(
   427→    (fieldKey: string, value: FormFieldValue) => {
   428→      setDraft((prev) => ({ ...prev, [fieldKey]: value }));
   429→      setMode((prev) => (prev === 'view' ? 'edit' : prev));
   430→    },
   431→    [],
   432→  );
   433→
   434→  // Ctrl+S / Cmd+S saves the active draft (Escape is handled by the dialog).
   435→  useEffect(() => {
   436→    const onKeyDown = (event: KeyboardEvent) => {
   437→      if (
   438→        (event.ctrlKey || event.metaKey) &&
   439→        !event.shiftKey &&
   440→        !event.altKey &&
   441→        event.key.toLowerCase() === 's'
   442→      ) {
   443→        event.preventDefault();
   444→        void handleSave();
   445→      }
   446→    };
   447→    window.addEventListener('keydown', onKeyDown);
   448→    return () => window.removeEventListener('keydown', onKeyDown);
   449→  }, [handleSave]);
   450→
   451→  // ── Derived view models ──
   452→  const sortedFields = useMemo(
   453→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   454→    [def],
   455→  );
   456→
   457→  const tabs = useMemo(() => (def ? deriveTabs(def, locale) : []), [def, locale]);
   458→
   459→  /**
   460→   * Two-part layout of tabbed forms: the first group is the header
   461→   * section (its fields are always visible, without any tab strip) and
   462→   * the remaining groups build the tabbed body section below it.
   463→   */
   464→  const isTabbed = def?.type === 'tabbed' && tabs.length > 0;
   465→  const headerTab = isTabbed ? (tabs[0] ?? null) : null;
   466→  const bodyTabs = isTabbed ? tabs.slice(1) : [];
   467→  const activeBodyTab =
   468→    bodyTabs.find((tab) => (tab.key ?? null) === (activeTabKey ?? null)) ??
   469→    bodyTabs[0] ??
   470→    null;
   471→
   472→  const listColumns = useMemo(
   473→    () => pickListColumns(sortedFields),
   474→    [sortedFields],
   475→  );
   476→
   477→  const title = def
   478→    ? formTitle(def, locale)
   479→    : form?.name ?? t('common.loading');
   480→  const subtitle =
   481→    mode === 'new'
   482→      ? t('formWindow.newMode')
   483→      : current
   484→        ? tf('formWindow.recordOf', {
   485→            n: formatNumber(currentIndex + 1),
   486→            total: formatNumber(records.length),
   487→          })
   488→        : undefined;
   489→
   490→  const accentColor = def?.accentColor || DEFAULT_FORM_ACCENT;
   491→
   492→  /** The fields grid shared by simple forms and the active tab. */
   493→  const renderFields = (fields: typeof sortedFields): ReactNode => (
   494→    <div className={fieldsGridClasses}>
   495→      {fields.map((field) => (
   496→        <FieldRenderer
   497→          key={field.fieldKey}
   498→          field={field}
   499→          value={draft[field.fieldKey] ?? null}
   500→          onChange={(value) => handleFieldChange(field.fieldKey, value)}
   501→          disabled={isSaving || isDeleting}
   502→          lookupUrlBase={lookupUrlBase}
   503→        />
   504→      ))}
   505→    </div>
   506→  );
   507→
   508→  let content: ReactNode;
   509→  if (!def) {
   510→    // Loading skeleton (error handled below).
   511→    content = (
   512→      <div className={fieldsGridClasses}>
   513→        {Array.from({ length: 6 }).map((_, index) => (
   514→          <div key={`form-skeleton-${index}`} className="space-y-1.5">
   515→            <Skeleton className="h-3 w-24 bg-white/[0.08]" />
   516→            <Skeleton className="h-6 w-full rounded-md bg-white/[0.05]" />
   517→          </div>
   518→        ))}
   519→      </div>
   520→    );
   521→  } else if (def.type === 'tree') {
   522→    content = (
   523→      <TreePanel
   524→        records={records}
   525→        fields={sortedFields}
   526→        lookupUrlBase={lookupUrlBase}
   527→      />
   528→    );
   529→  } else if (isTabbed && headerTab) {
   530→    // Header section (main fields, no tab strip) + tabbed body section.
   531→    content = (
   532→      <div className="flex min-h-0 flex-col">
   533→        <section
   534→          aria-label={headerTab.name}
   535→          className={cn(
   536→            'shrink-0 bg-white/[0.02]',
   537→            bodyTabs.length > 0 && 'border-b border-white/10',
   538→          )}
   539→        >
   540→          {renderFields(headerTab.fields)}
   541→        </section>
   542→        {bodyTabs.length > 0 && (
   543→          <section aria-label={activeBodyTab?.name ?? undefined}>
   544→            <FormTabs
   545→              tabs={bodyTabs.map((tab) => ({
   546→                key: tab.key,
   547→                name: tab.name,
   548→                count: tab.fields.length,
   549→              }))}
   550→              active={activeBodyTab?.key ?? null}
   551→              onChange={setActiveTabKey}
   552→              accentColor={accentColor}
   553→              direction={direction}
   554→            >
   555→              {renderFields(activeBodyTab?.fields ?? [])}
   556→            </FormTabs>
   557→          </section>
   558→        )}
   559→      </div>
   560→    );
   561→  } else {
   562→    content = renderFields(sortedFields);
   563→  }
   564→
   565→  const errorText =
   566→    loadError === GENERIC_LOAD_ERROR ? t('formWindow.loadError') : loadError;
   567→
   568→  return (
   569→    <>
   570→    <FormShell
   571→      title={title}
   572→      subtitle={subtitle}
   573→      iconKey={def?.iconKey ?? form?.iconKey ?? 'fileText'}
   574→      accentColor={accentColor}
   575→      actionBar={
   576→        <FormActionBar
   577→          mode={mode}
   578→          isDirty={isDirty}
   579→          isSaving={isSaving}
   580→          isDeleting={isDeleting}
   581→          hasRecord={Boolean(current)}
   582→          recordIndex={currentIndex}
   583→          recordCount={records.length}
   584→          isListOpen={isListOpen}
   585→          onNew={handleNew}
   586→          onSave={() => void handleSave()}
   587→          onSaveNew={() => void handleSaveNew()}
   588→          onSaveClose={() => void handleSaveClose()}
   589→          onCopy={handleCopy}
   590→          onDelete={() => requestDelete('current')}
   591→          onDeleteNew={() => requestDelete('new')}
   592→          onDeleteClose={() => requestDelete('close')}
   593→          onPrint={handlePrint}
   594→          onToggleList={() => setIsListOpen((open) => !open)}
   595→          onFirst={() => navigateTo(0)}
   596→          onPrev={() => navigateTo(Math.max(0, currentIndex - 1))}
   597→          onNext={() => navigateTo(Math.min(records.length - 1, currentIndex + 1))}
   598→          onLast={() => navigateTo(records.length - 1)}
   599→        />
   600→      }
   601→      subPanel={
   602→        <AnimatePresence initial={false}>
   603→          {isListOpen && (
   604→            <RecordsDrawer
   605→              key="records-drawer"
   606→              records={records}
   607→              currentIndex={currentIndex}
   608→              codeField={listColumns.codeField}
   609→              titleField={listColumns.titleField}
   610→              onJump={navigateTo}
   611→              onClose={() => setIsListOpen(false)}
   612→            />
   613→          )}
   614→        </AnimatePresence>
   615→      }
   616→      footer={
   617→        <FormFooter
   618→          status={status}
   619→          position={
   620→            current && mode !== 'new'
   621→              ? { n: currentIndex + 1, total: records.length }
   622→              : null
   623→          }
   624→          updatedAt={current?.updatedAt ?? null}
   625→        />
   626→      }
   627→    >
   628→      {errorText && !def ? (
   629→        <div className="flex min-h-[220px] flex-col items-center justify-center gap-3 p-6 text-center">
   630→          <p className="max-w-sm text-sm text-red-200/90">{errorText}</p>
   631→          <button
   632→            type="button"
   633→            onClick={() => void load()}
   634→            className="flex items-center gap-2 rounded-lg bg-white/[0.07] px-4 py-2 text-[13px] text-white/80 transition-colors hover:bg-white/[0.12] focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-white/40"
   635→          >
   636→            <RefreshCw className="h-3.5 w-3.5" />
   637→            {t('common.retry')}
   638→          </button>
   639→        </div>
   640→      ) : (
   641→        <>
   642→          {errorText && def && (
   643→            <div className="mx-4 mt-4 rounded-lg border border-red-400/30 bg-red-950/40 px-3 py-2 text-xs text-red-200/90">
   644→              {errorText}
   645→            </div>
   646→          )}
   647→          {records.length === 0 && def && def.type !== 'tree' ? (
   648→            <div className="px-4 pb-2 pt-6 text-center">
   649→              <p className="text-[13px] text-white/55">
   650→                {t('formWindow.emptyRecords')}
   651→              </p>
   652→              <p className="mt-1 text-[11px] text-white/35">
   653→                {t('formWindow.emptyRecordsHint')}
   654→              </p>
   655→            </div>
   656→          ) : null}
   657→          {content}
   658→        </>
   659→      )}
   660→    </FormShell>
   661→
   662→    {/* Single delete-confirmation dialog shared by all three delete actions. */}
   663→    <AlertDialog
   664→      open={deleteIntent !== null}
   665→      onOpenChange={(open) => {
   666→        if (!open) setDeleteIntent(null);
   667→      }}
   668→    >
   669→      <AlertDialogContent className="max-w-[calc(100vw-2rem)] rounded-2xl border border-white/15 bg-[#1e2128]/95 text-white shadow-2xl shadow-black/60 backdrop-blur-xl sm:max-w-md">
   670→        <AlertDialogHeader className="sm:text-start">
   671→          <AlertDialogTitle className="flex items-center gap-2.5 text-base text-white">
   672→            <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-rose-500/15 text-rose-400 ring-1 ring-rose-400/30">
   673→              <Trash2 className="h-4 w-4" aria-hidden="true" />
   674→            </span>
   675→            {t('formWindow.deleteDialogTitle')}
   676→          </AlertDialogTitle>
   677→          <AlertDialogDescription className="text-[13px] leading-relaxed text-white/65">
   678→            {t('formWindow.deleteDialogMessage')}
   679→          </AlertDialogDescription>
   680→        </AlertDialogHeader>
   681→        <AlertDialogFooter className="gap-2 sm:justify-start">
   682→          <AlertDialogCancel
   683→            disabled={isDeleting}
   684→            className="h-9 rounded-lg border-white/15 bg-white/[0.06] px-4 text-[13px] text-white/80 transition-colors hover:bg-white/[0.12] hover:text-white focus-visible:ring-white/40"
   685→          >
   686→            {t('formWindow.confirmNo')}
   687→          </AlertDialogCancel>
   688→          <AlertDialogAction
   689→            disabled={isDeleting}
   690→            onClick={(event) => {
   691→              // Keep the dialog open while the delete request runs.
   692→              event.preventDefault();
   693→              void confirmDelete();
   694→            }}
   695→            className="flex h-9 items-center gap-1.5 rounded-lg bg-rose-500/90 px-4 text-[13px] text-white shadow-sm shadow-rose-950/40 transition-colors hover:bg-rose-500 focus-visible:ring-rose-300/60"
   696→          >
   697→            {isDeleting ? (
   698→              <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" />
   699→            ) : null}
   700→            {t('formWindow.confirmYes')}
   701→          </AlertDialogAction>
   702→        </AlertDialogFooter>
   703→      </AlertDialogContent>
   704→    </AlertDialog>
   705→    </>
   706→  );
   707→}
