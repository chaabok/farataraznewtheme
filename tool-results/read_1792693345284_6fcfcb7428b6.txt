    73→/** Fallback parsing of the tab id when no descriptor is attached. */
    74→function parseListModuleId(moduleId: string): {
    75→  moduleKey: string | null;
    76→  formKey: string | null;
    77→} {
    78→  if (!moduleId.startsWith('list:')) {
    79→    return { moduleKey: null, formKey: null };
    80→  }
    81→  const formKey = moduleId.slice('list:'.length);
    82→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    83→}

    85→/** Sentinel kept in state so load errors render through the dictionaries. */
    86→const GENERIC_LOAD_ERROR = 'dataList.loadError';
    87→
    88→const PERSIAN_DIGITS = '۰۱۲۳۴۵۶۷۸۹';
    89→const ARABIC_DIGITS = '٠١٢٣٤٥٦٧٨٩';
    90→
    91→/**
    92→ * Search normalization: lowercase, fold Persian/Arabic yeh & kaf variants
    93→ * and convert Persian/Arabic digits to Latin so typed filters match the
    94→ * stored data regardless of the keyboard used.
    95→ */
    96→function normalizeSearchText(value: string): string {
    97→  return value
    98→    .toLowerCase()
    99→    .replace(/[۰-۹]/g, (digit) => String(PERSIAN_DIGITS.indexOf(digit)))
   100→    .replace(/[٠-٩]/g, (digit) => String(ARABIC_DIGITS.indexOf(digit)))
   101→    .replace(/ي/g, 'ی')
   102→    .replace(/ك/g, 'ک')
   103→    .replace(/\u200c/g, ' ')
   104→    .trim();
   105→}

   107→/** Lightweight table checkbox (button-based, no Radix overhead per row). */
   108→function RowCheckbox({
   109→  checked,
   110→  onChange,
   111→  label,
   112→}: {
   113→  checked: boolean;
   114→  onChange: () => void;
   115→  label: string;
   116→}) {
   117→  return (
   118→    <button
   119→      type="button"
   120→      role="checkbox"
   121→      aria-checked={checked}
   122→      aria-label={label}
   123→      title={label}
   124→      onClick={(event) => {
   125→        event.stopPropagation();
   126→        onChange();
   127→      }}
   128→      onKeyDown={(event) => event.stopPropagation()}
   129→      onDoubleClick={(event) => event.stopPropagation()}
   130→      className={cn(
   131→        'mx-auto flex h-4 w-4 items-center justify-center rounded border transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-sky-400/60',
   132→        checked
   133→          ? 'border-sky-300/70 bg-sky-500/80 text-white shadow-[0_0_8px_-2px_rgba(56,189,248,0.8)]'
   134→          : 'border-white/25 bg-white/[0.04] hover:border-white/50',
   135→      )}
   136→    >
   137→      {checked && <Check className="h-3 w-3" strokeWidth={3.2} />}
   138→    </button>
   139→  );
   140→}

   142→/**
   143→ * Generic, database-driven records list («فهرست …»): every form saved
   144→ * through the unified form system automatically gets one of these windows
   145→ * with a filterable grid, multi-select, column chooser and the ERP-style
   146→ * footer icon strip (new / edit / delete / view / export / print /
   147→ * refresh / columns / filter).
   148→ */
   149→export function DataListWindow({ moduleId, list, onClose }: DataListWindowProps) {
   150→  const { locale, t, tf, formatNumber } = useTranslation();
   151→  const addFormTab = useDesktopStore((s) => s.addFormTab);
   152→
   153→  const fallback = useMemo(() => parseListModuleId(moduleId), [moduleId]);
   154→  const moduleKey = list?.moduleKey ?? fallback.moduleKey;
   155→  const formKey = list?.formKey ?? fallback.formKey;
   156→  const basePath =
   157→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
   158→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   159→
   160→  // ── Data ──
   161→  const [def, setDef] = useState<FormDef | null>(null);
   162→  const [records, setRecords] = useState<FormRecord[]>([]);
   163→  const [isLoading, setIsLoading] = useState(true);
   164→  const [isRefreshing, setIsRefreshing] = useState(false);
   165→  const [loadError, setLoadError] = useState<string | null>(null);
   166→  /** lookupKey → (option code → label). */
   167→  const [lookupLabels, setLookupLabels] = useState<
   168→    Record<string, Map<string, string>>
   169→  >({});
   170→
   171→  // ── View state ──
   172→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   173→  const [filters, setFilters] = useState<Record<string, string>>({});
   174→  const [isFilterRowVisible, setIsFilterRowVisible] = useState(true);
   175→  const [hiddenColumns, setHiddenColumns] = useState<Set<string>>(new Set());
   176→  const [isColumnsOpen, setIsColumnsOpen] = useState(false);
   177→  const [isDeleteConfirmOpen, setIsDeleteConfirmOpen] = useState(false);
   178→  const [isDeleting, setIsDeleting] = useState(false);
   179→
   180→  // ── Loading ──
   181→  const loadLookups = useCallback(
   182→    async (formDef: FormDef) => {
   183→      const keys = Array.from(
   184→        new Set(
   185→          formDef.fields
   186→            .filter((field) => field.fieldType === 'lookup' && field.lookupKey)
   187→            .map((field) => field.lookupKey as string),
   188→        ),
   189→      );
   190→      if (!keys.length || !lookupUrlBase) return;
   191→      const results = await Promise.all(
   192→        keys.map((key) =>
   193→          apiFetch<{ data: LookupOption[] }>(
   194→            `${lookupUrlBase}/${encodeURIComponent(key)}?locale=${encodeURIComponent(locale)}`,
   195→          )
   196→            .then((payload): [string, Map<string, string>] => [
   197→              key,
   198→              new Map(
   199→                (payload.data ?? []).map((option) => [
   200→                  option.code,
   201→                  locale === 'fa' ? option.label : (option.labelEn ?? option.label),
   202→                ]),
   203→              ),
   204→            ])
   205→            .catch((): [string, Map<string, string>] => [key, new Map()]),
   206→        ),
   207→      );
   208→      setLookupLabels(Object.fromEntries(results));
   209→    },
   210→    [locale, lookupUrlBase],
   211→  );
   212→
   213→  const load = useCallback(async () => {
   214→    if (!basePath) {
   215→      setLoadError(GENERIC_LOAD_ERROR);
   216→      setIsLoading(false);
   217→      return;
   218→    }
   219→    setIsLoading(true);
   220→    setLoadError(null);
   221→    try {
   222→      const defRes = await apiFetch<{ data: FormDef }>(
   223→        `${basePath}?locale=${encodeURIComponent(locale)}`,
   224→      );
   225→      const nextDef = defRes.data;
   226→      setDef(nextDef);
   227→      setSelectedIds(new Set());
   228→
   229→      const recordsRes = await apiFetch<{ data: FormRecord[] }>(
   230→        `${basePath}/records`,
   231→      );
   232→      setRecords(recordsRes.data ?? []);
   233→      void loadLookups(nextDef);
   234→    } catch (error) {
   235→      if (error instanceof SessionExpiredError) return;
   236→      setLoadError(
   237→        error instanceof Error && error.message
   238→          ? error.message
   239→          : GENERIC_LOAD_ERROR,
   240→      );
   241→    } finally {
   242→      setIsLoading(false);
   243→    }
   244→  }, [basePath, loadLookups, locale]);
   245→
   246→  useEffect(() => {
   247→    void load();
   248→  }, [load]);
   249→
   250→  const refreshRecords = useCallback(async () => {
   251→    if (!basePath) return;
   252→    setIsRefreshing(true);
   253→    try {
   254→      const res = await apiFetch<{ data: FormRecord[] }>(`${basePath}/records`);
   255→      const next = res.data ?? [];
   256→      setRecords(next);
   257→      setSelectedIds((prev) => {
   258→        const nextIds = new Set(
   259→          [...prev].filter((id) => next.some((record) => record.id === id)),
   260→        );
   261→        return nextIds.size === prev.size ? prev : nextIds;
   262→      });
   263→    } catch (error) {
   264→      if (error instanceof SessionExpiredError) return;
   265→      toast({
   266→        title: t('dataList.loadError'),
   267→        description: error instanceof Error ? error.message : undefined,
   268→        variant: 'destructive',
   269→      });
   270→    } finally {
   271→      setIsRefreshing(false);
   272→    }
   273→  }, [basePath, t]);
   274→
   275→  // Auto-refresh when a form window of the same form mutates records.
   276→  useEffect(() => {
   277→    if (!moduleKey || !formKey) return;
   278→    const handler = (event: Event) => {
   279→      const detail = (
   280→        event as CustomEvent<{ moduleKey?: string; formKey?: string }>
   281→      ).detail;
   282→      if (detail?.moduleKey === moduleKey && detail?.formKey === formKey) {
   283→        void refreshRecords();
   284→      }
   285→    };
   286→    window.addEventListener(RECORDS_CHANGED_EVENT, handler);
   287→    return () => window.removeEventListener(RECORDS_CHANGED_EVENT, handler);
   288→  }, [formKey, moduleKey, refreshRecords]);
   289→
   290→  // ── Derived view models ──
   291→  const sortedFields = useMemo(
   292→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   293→    [def],
   294→  );
   295→  const parentField = useMemo(
   296→    () => (def?.type === 'tree' ? pickParentField(sortedFields) : null),
   297→    [def, sortedFields],
   298→  );
   299→  const visibleFields = useMemo(
   300→    () => sortedFields.filter((field) => !hiddenColumns.has(field.fieldKey)),
   301→    [hiddenColumns, sortedFields],
   302→  );
   303→  const fieldByKey = useMemo(
   304→    () => new Map(sortedFields.map((field) => [field.fieldKey, field])),
   305→    [sortedFields],
   306→  );
   307→
   308→  /** recordId → display title of that record (tree parent columns). */
   309→  const parentTitles = useMemo(() => {
   310→    const map = new Map<string, string>();
   311→    if (!parentField) return map;
   312→    const titleField =
   313→      sortedFields.find(
   314→        (field) =>
   315→          field !== parentField && /title|name|desc/i.test(field.fieldKey),
   316→      ) ??
   317→      sortedFields.find((field) => field !== parentField) ??
   318→      null;
   319→    for (const record of records) {
   320→      const raw = titleField ? record.data[titleField.fieldKey] : null;
   321→      map.set(
   322→        record.id,
   323→        raw === null || raw === undefined || raw === ''
   324→          ? record.id
   325→          : String(raw),
   326→      );
   327→    }
   328→    return map;
   329→  }, [parentField, records, sortedFields]);
   330→
   331→  /** Display text of one cell (checkbox → بلی/خیر, lookup → label, …). */
   332→  const cellText = useCallback(
   333→    (field: FormFieldDef, record: FormRecord): string => {
   334→      const raw = record.data[field.fieldKey];
   335→      if (field.fieldType === 'checkbox') {
   336→        return raw === true ? t('dataList.yes') : t('dataList.no');
   337→      }
   338→      // Tree parent columns resolve the parent record's title (the
   339→      // stored value is the parent record id — must run before lookup).
   340→      if (parentField && field.fieldKey === parentField.fieldKey) {
   341→        if (raw === null || raw === undefined || raw === '') return '—';
   342→        return parentTitles.get(String(raw)) ?? String(raw);
   343→      }
   344→      if (field.fieldType === 'lookup') {
   345→        if (raw === null || raw === undefined || raw === '') return '—';
   346→        const labels = field.lookupKey
   347→          ? lookupLabels[field.lookupKey]
   348→          : undefined;
   349→        return labels?.get(String(raw)) ?? String(raw);
   350→      }
   351→      return formatFieldValue(field, raw, formatNumber);
   352→    },
   353→    [formatNumber, lookupLabels, parentField, parentTitles, t],
   354→  );
   355→
   356→  const filteredRecords = useMemo(() => {
   357→    const active = Object.entries(filters).filter(
   358→      ([, value]) => value.trim() !== '',
   359→    );
   360→    if (!active.length) return records;
   361→    const needles: Array<{ field: FormFieldDef; needle: string }> = [];
   362→    for (const [key, value] of active) {
   363→      const field = fieldByKey.get(key);
   364→      if (field) needles.push({ field, needle: normalizeSearchText(value) });
   365→    }
   366→    if (!needles.length) return records;
   367→    return records.filter((record) =>
   368→      needles.every(({ field, needle }) =>
   369→        normalizeSearchText(cellText(field, record)).includes(needle),
   370→      ),
   371→    );
   372→  }, [cellText, fieldByKey, filters, records]);
   373→
   374→  // ── Selection ──
   375→  const toggleSelected = useCallback((recordId: string, additive: boolean) => {
   376→    setSelectedIds((prev) => {
   377→      if (!additive) return prev.has(recordId) ? new Set([recordId]) : new Set([recordId]);
   378→      const next = new Set(prev);
   379→      if (next.has(recordId)) next.delete(recordId);
   380→      else next.add(recordId);
   381→      return next;
   382→    });
   383→  }, []);
   384→
   385→  const handleRowClick = useCallback(
   386→    (record: FormRecord, event: MouseEvent<HTMLTableRowElement>) => {
   387→      if (event.ctrlKey || event.metaKey) {
   388→        toggleSelected(record.id, true);
   389→      } else {
   390→        setSelectedIds((prev) =>
   391→          prev.size === 1 && prev.has(record.id) ? prev : new Set([record.id]),
   392→        );
   393→      }
   394→    },
   395→    [toggleSelected],
   396→  );
   397→
   398→  const allVisibleSelected =
   399→    filteredRecords.length > 0 &&
   400→    filteredRecords.every((record) => selectedIds.has(record.id));
   401→
   402→  const toggleSelectAll = useCallback(() => {
   403→    setSelectedIds((prev) => {
   404→      if (filteredRecords.length && filteredRecords.every((record) => prev.has(record.id))) {
   405→        return new Set();
   406→      }
   407→      return new Set(filteredRecords.map((record) => record.id));
   408→    });
   409→  }, [filteredRecords]);
   410→
   411→  // ── Form integration ──
   412→  const openForm = useCallback(
   413→    (options?: { recordId?: string; startInNew?: boolean; nonce?: number }) => {
   414→      if (!moduleKey || !formKey) return;
   415→      addFormTab({
   416→        moduleKey,
   417→        formKey,
   418→        name: def ? formTitle(def, locale) : (list?.name ?? formKey),
   419→        iconKey: def?.iconKey ?? list?.iconKey ?? 'fileText',
   420→        nonce: Date.now(),
   421→        ...options,
   422→      });
   423→    },
   424→    [addFormTab, def, formKey, list, locale, moduleKey],
   425→  );
   426→
   427→  const singleSelectedId =
   428→    selectedIds.size === 1 ? [...selectedIds][0] ?? null : null;
   429→
   430→  // ── Delete ──
   431→  const requestDelete = useCallback(() => {
   432→    if (!selectedIds.size || isDeleting || isRefreshing) return;
   433→    setIsDeleteConfirmOpen(true);
   434→  }, [isDeleting, isRefreshing, selectedIds.size]);
   435→
   436→  const confirmDelete = useCallback(async () => {
   437→    if (!basePath || !selectedIds.size || isDeleting) {
   438→      setIsDeleteConfirmOpen(false);
   439→      return;
   440→    }
   441→    setIsDeleting(true);
   442→    try {
   443→      for (const id of selectedIds) {
   444→        // DELETE answers 204 with an empty body — nothing to parse.
   445→        await apiFetch(`${basePath}/records/${encodeURIComponent(id)}`, {
   446→          method: 'DELETE',
   447→        });
   448→      }
   449→      setIsDeleteConfirmOpen(false);
   450→      const count = selectedIds.size;
   451→      setSelectedIds(new Set());
   452→      // The list listens to its own event too — one refresh for every
   453→      // open list/form of this formKey (dispatchEvent is synchronous).
   454→      notifyRecordsChanged({ moduleKey: moduleKey ?? '', formKey: formKey ?? '' });
   455→      toast({
   456→        title: tf('dataList.deleteSuccess', { count: formatNumber(count) }),
   457→      });
   458→    } catch (error) {
   459→      setIsDeleteConfirmOpen(false);
   460→      if (error instanceof SessionExpiredError) return;
   461→      toast({
   462→        title:
   463→          error instanceof Error && error.message
   464→            ? error.message
   465→            : t('dataList.deleteError'),
   466→        variant: 'destructive',
   467→      });
   468→    } finally {
   469→      setIsDeleting(false);
   470→    }
   471→  }, [
   472→    basePath,
   473→    formatNumber,
   474→    formKey,
   475→    isDeleting,
   476→    moduleKey,
   477→    selectedIds,
   478→    t,
   479→    tf,
   480→  ]);
   481→
   482→  // ── Export / print ──
   483→  const handleExport = useCallback(() => {
   484→    if (!visibleFields.length || !filteredRecords.length) {
   485→      toast({ title: t('dataList.exportEmpty') });
   486→      return;
   487→    }
   488→    const escapeCell = (value: string) => `"${value.replace(/"/g, '""')}"`;
   489→    const header = visibleFields.map((field) =>
   490→      escapeCell(fieldLabel(field, locale)),
   491→    );
   492→    const rows = filteredRecords.map((record) =>
   493→      visibleFields.map((field) => escapeCell(cellText(field, record))),
   494→    );
   495→    const csv =
   496→      '\uFEFF' + [header, ...rows].map((row) => row.join(',')).join('\r\n');
   497→    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
   498→    const url = URL.createObjectURL(blob);
   499→    const anchor = document.createElement('a');
   500→    anchor.href = url;
   501→    anchor.download = `${formKey ?? 'list'}-${new Date()
   502→      .toISOString()
   503→      .slice(0, 10)}.csv`;
   504→    anchor.click();
   505→    URL.revokeObjectURL(url);
   506→    toast({ title: t('dataList.exportSuccess') });
   507→  }, [cellText, filteredRecords, formKey, locale, t, visibleFields]);
   508→
   509→  const handlePrint = useCallback(() => {
   510→    toast({ title: t('formWindow.printToast') });
   511→  }, [t]);
   512→
   513→  const toggleColumn = useCallback((fieldKey: string) => {
   514→    setHiddenColumns((prev) => {
   515→      const next = new Set(prev);
   516→      if (next.has(fieldKey)) next.delete(fieldKey);
   517→      else next.add(fieldKey);
   518→      return next;
   519→    });
   520→  }, []);
   521→
   522→  // ── Chrome texts ──
   523→  const listName = def
   524→    ? formTitle(def, locale)
   525→    : (list?.name ?? formKey ?? t('common.loading'));
   526→  const title = tf('dataList.title', {
   527→    name: listName,
   528→    count: formatNumber(filteredRecords.length),
   529→  });
   530→  const subtitle = selectedIds.size
   531→    ? tf('dataList.selectedCount', { count: formatNumber(selectedIds.size) })
   532→    : tf('dataList.recordCount', { count: formatNumber(records.length) });
   533→
   534→  const accentColor = def?.accentColor || DEFAULT_FORM_ACCENT;
   535→  const hasActiveFilters = Object.values(filters).some(
   536→    (value) => value.trim() !== '',
   537→  );
   538→
   539→  const errorText =
   540→    loadError === GENERIC_LOAD_ERROR ? t('dataList.loadError') : loadError;
   541→
   542→  /** The grid (or its loading / error / empty states). */
   543→  let content: ReactNode;
   544→  if (!def) {
   545→    content = (
   546→      <div className="space-y-2 p-3">
   547→        {Array.from({ length: 6 }).map((_, index) => (
   548→          <Skeleton
   549→            key={`list-skeleton-${index}`}
   550→            className="h-8 w-full rounded-md bg-white/[0.05]"
   551→          />
   552→        ))}
   553→      </div>
   554→    );
   555→  } else {
   556→    content = (
   557→      <div className="relative h-full min-h-0">
   558→        <div className="h-full overflow-auto [scrollbar-width:thin]">
   559→          <table className="w-full min-w-max border-separate border-spacing-0 text-[12px]">
   560→            <thead>
   561→              <tr>
   562→                <th className="sticky top-0 z-30 h-9 w-10 border-b border-white/10 bg-[#1a1d26] px-2">
   563→                  <RowCheckbox
   564→                    checked={allVisibleSelected}
   565→                    onChange={toggleSelectAll}
   566→                    label={t('dataList.selectAll')}
   567→                  />
   568→                </th>
   569→                {visibleFields.map((field) => (
   570→                  <th
   571→                    key={field.fieldKey}
   572→                    scope="col"
   573→                    className="sticky top-0 z-30 h-9 min-w-[130px] max-w-[260px] border-b border-white/10 bg-[#1a1d26] px-3 text-center align-middle text-[11.5px] font-bold leading-tight text-white/85"
   574→                  >
