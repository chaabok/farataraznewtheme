    55→    55→export interface DataFormWindowProps {
    56→    56→  /** Window id — `form:<formKey>` (moduleKey defaults to `settings`). */
    57→    57→  moduleId: string;
    58→    58→  /** Descriptor attached by `openFormWindow` in the desktop store. */
    59→    59→  form?: FormWindowInfo;
    60→    60→  /** Closes the hosting window ("save & close" / "delete & close"). */
    61→    61→  onClose?: () => void;
    62→    62→}
    63→    63→
    64→    64→/** Which combined action a pending delete confirmation belongs to. */
    65→    65→type DeleteIntent = 'current' | 'new' | 'close' | null;
    66→    66→
    67→    67→/** Outcome of the shared save pipeline. */
    68→    68→type SaveResult = 'saved' | 'clean' | 'invalid' | 'error';
    69→    69→
    70→    70→/** Fallback parsing of the window id when no descriptor is attached. */
    71→    71→function parseModuleId(moduleId: string): {
    72→    72→  moduleKey: string | null;
    73→    73→  formKey: string | null;
    74→    74→} {
    75→    75→  if (!moduleId.startsWith('form:')) {
    76→    76→    return { moduleKey: null, formKey: null };
    77→    77→  }
    78→    78→  const formKey = moduleId.slice('form:'.length);
    79→    79→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    80→    80→}
    81→    81→
    82→    82→/** Sentinel kept in state so load errors render through the dictionaries. */
    83→    83→const GENERIC_LOAD_ERROR = 'formWindow.loadError';
    84→    84→
    85→    85→/**
    86→    86→ * Generic, database-driven form window: loads a FormDef + its records and
    87→    87→ * drives the whole view/new/edit state machine (dirty tracking, CRUD,
    88→    88→ * record navigation, Ctrl+S, records drawer) through the shared form
    89→    89→ * component library. One implementation renders every ERP form.
    90→    90→ */
    91→    91→export function DataFormWindow({ moduleId, form, onClose }: DataFormWindowProps) {
    92→    92→  const { locale, direction, t, tf, formatNumber } = useTranslation();
    93→    93→
    94→    94→  const fallback = useMemo(() => parseModuleId(moduleId), [moduleId]);
    95→    95→  const moduleKey = form?.moduleKey ?? fallback.moduleKey;
    96→    96→  const formKey = form?.formKey ?? fallback.formKey;
    97→    97→  const basePath =
    98→    98→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
    99→    99→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   100→   100→
   101→   101→  // ── Data ──
   102→   102→  const [def, setDef] = useState<FormDef | null>(null);
   103→   103→  const [records, setRecords] = useState<FormRecord[]>([]);
   104→   104→  const [isLoading, setIsLoading] = useState(true);
   105→   105→  const [loadError, setLoadError] = useState<string | null>(null);
   106→   106→
   107→   107→  // ── Record state machine ──
   108→   108→  const [mode, setMode] = useState<FormMode>('view');
   109→   109→  const [currentIndex, setCurrentIndex] = useState(-1);
   110→   110→  const [draft, setDraft] = useState<FormDataMap>({});
   111→   111→  const [baseline, setBaseline] = useState<FormDataMap>({});
   112→   112→  const [isSaving, setIsSaving] = useState(false);
   113→   113→  const [isDeleting, setIsDeleting] = useState(false);
   114→   114→  /**
   115→   115→   * Pending delete request awaiting the confirm dialog — `current` =
   116→   116→   * delete and stay, `new` = delete then reset for a fresh entry,
   117→   117→   * `close` = delete then close the window.
   118→   118→   */
   119→   119→  const [deleteIntent, setDeleteIntent] = useState<DeleteIntent>(null);
   120→   120→  const [isListOpen, setIsListOpen] = useState(false);
   121→   121→  const [activeTabKey, setActiveTabKey] = useState<string | null>(null);
   122→   122→
   123→   123→  const current =
   124→   124→    currentIndex >= 0 && currentIndex < records.length
   125→   125→      ? records[currentIndex]
   126→   126→      : null;
   127→   127→
   128→   128→  const isDirty = useMemo(() => {
   129→   129→    const keys = new Set([...Object.keys(baseline), ...Object.keys(draft)]);
   130→   130→    for (const key of keys) {
   131→   131→      if (isChangedValue(baseline[key], draft[key])) return true;
   132→   132→    }
   133→   133→    return false;
   134→   134→  }, [baseline, draft]);
   135→   135→
   136→   136→  const status: FormStatus =
   137→   137→    mode === 'new' ? (isDirty ? 'unsaved' : 'new') : isDirty ? 'unsaved' : 'saved';
   138→   138→
   139→   139→  // ── Loading ──
   140→   140→  const load = useCallback(async () => {
   141→   141→    if (!basePath) {
   142→   142→      setLoadError(GENERIC_LOAD_ERROR);
   143→   143→      setIsLoading(false);
   144→   144→      return;
   145→   145→    }
   146→   146→    setIsLoading(true);
   147→   147→    setLoadError(null);
   148→   148→    try {
   149→   149→      const [defRes, recordsRes] = await Promise.all([
   150→   150→        apiFetch<{ data: FormDef }>(
   151→   151→          `${basePath}?locale=${encodeURIComponent(locale)}`,
   152→   152→        ),
   153→   153→        apiFetch<{ data: FormRecord[] }>(`${basePath}/records`),
   154→   154→      ]);
   155→   155→      const nextDef = defRes.data;
   156→   156→      const nextRecords = recordsRes.data ?? [];
   157→   157→      setDef(nextDef);
   158→   158→      setRecords(nextRecords);
   159→   159→
   160→   160→      if (nextRecords.length) {
   161→   161→        const lastIndex = nextRecords.length - 1;
   162→   162→        const last = nextRecords[lastIndex];
   163→   163→        setCurrentIndex(lastIndex);
   164→   164→        setDraft({ ...last.data });
   165→   165→        setBaseline({ ...last.data });
   166→   166→        setMode('view');
   167→   167→      } else {
   168→   168→        const defaults = buildDefaults(nextDef.fields);
   169→   169→        setCurrentIndex(-1);
   170→   170→        setDraft({ ...defaults });
   171→   171→        setBaseline({ ...defaults });
   172→   172→        setMode('new');
   173→   173→      }
   174→   174→
   175→   175→      const firstTab = deriveTabs(nextDef, locale)[0] ?? null;
   176→   176→      setActiveTabKey(firstTab ? firstTab.key : null);
   177→   177→    } catch (error) {
   178→   178→      if (error instanceof SessionExpiredError) return;
   179→   179→      setLoadError(
   180→   180→        error instanceof Error && error.message
   181→   181→          ? error.message
   182→   182→          : GENERIC_LOAD_ERROR,
   183→   183→      );
   184→   184→    } finally {
   185→   185→      setIsLoading(false);
   186→   186→    }
   187→   187→  }, [basePath, locale]);
   188→   188→
   189→   189→  useEffect(() => {
   190→   190→    void load();
   191→   191→  }, [load]);
   192→   192→
   193→   193→  const refreshRecords = useCallback(async (): Promise<FormRecord[]> => {
   194→   194→    if (!basePath) return [];
   195→   195→    const res = await apiFetch<{ data: FormRecord[] }>(`${basePath}/records`);
   196→   196→    const list = res.data ?? [];
   197→   197→    setRecords(list);
   198→   198→    return list;
   199→   199→  }, [basePath]);
   200→   200→
   201→   201→  // ── Record transitions ──
   202→   202→  const applyRecord = useCallback((index: number, list: FormRecord[]) => {
   203→   203→    const target = list[index];
   204→   204→    setCurrentIndex(index);
   205→   205→    setMode('view');
   206→   206→    if (target) {
   207→   207→      setDraft({ ...target.data });
   208→   208→      setBaseline({ ...target.data });
   209→   209→    }
   210→   210→  }, []);
   211→   211→
   212→   212→  const navigateTo = useCallback(
   213→   213→    (index: number) => {
   214→   214→      if (isSaving || isDeleting) return;
   215→   215→      if (index < 0 || index >= records.length) return;
   216→   216→      if (mode !== 'view' && isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) {
   217→   217→        return;
   218→   218→      }
   219→   219→      applyRecord(index, records);
   220→   220→    },
   221→   221→    [applyRecord, isDeleting, isDirty, isSaving, mode, records, t],
   222→   222→  );
   223→   223→
   224→   224→  const startNew = useCallback(
   225→   225→    (preset?: FormDataMap) => {
   226→   226→      if (!def) return;
   227→   227→      const base = { ...buildDefaults(def.fields), ...(preset ?? {}) };
   228→   228→      setDraft(base);
   229→   229→      setBaseline({ ...base });
   230→   230→      setMode('new');
   231→   231→      setCurrentIndex(-1);
   232→   232→    },
   233→   233→    [def],
   234→   234→  );
   235→   235→
   236→   236→  const handleNew = useCallback(() => {
   237→   237→    if (mode === 'new') return;
   238→   238→    if (isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) return;
   239→   239→    startNew();
   240→   240→  }, [isDirty, mode, startNew, t]);
   241→   241→
   242→   242→  const handleCopy = useCallback(() => {
   243→   243→    if (mode !== 'view' || !current) return;
   244→   244→    startNew({ ...current.data });
   245→   245→    toast({ title: t('formWindow.copyToast') });
   246→   246→  }, [current, mode, startNew, t]);
   247→   247→
   248→   248→  /**
   249→   249→   * Shared save pipeline for every save action: validates required
   250→   250→   * fields, writes the draft to the database and refreshes the record
   251→   251→   * list. `keepRecord: false` skips re-selecting the saved record
   252→   252→   * (used by "save & new", which resets the form right after).
   253→   253→   */
   254→   254→  const performSave = useCallback(
   255→   255→    async (options?: { keepRecord?: boolean }): Promise<SaveResult> => {
   256→   256→      if (!def || !basePath || isSaving || isDeleting) return 'error';
   257→   257→      if (mode === 'view' || !isDirty) return 'clean';
   258→   258→
   259→   259→      const missingField = def.fields.find((field) =>
   260→   260→        isEmptyRequiredValue(field, draft[field.fieldKey]),
   261→   261→      );
   262→   262→      if (missingField) {
   263→   263→        toast({
   264→   264→          title: fieldLabel(missingField, locale),
   265→   265→          description: t('formWindow.requiredHint'),
   266→   266→          variant: 'destructive',
   267→   267→        });
   268→   268→        return 'invalid';
   269→   269→      }
   270→   270→
   271→   271→      setIsSaving(true);
   272→   272→      try {
   273→   273→        const saved =
   274→   274→          mode === 'new'
   275→   275→            ? (
   276→   276→                await apiFetch<{ data: FormRecord }>(`${basePath}/records`, {
   277→   277→                  method: 'POST',
   278→   278→                  body: { data: draft },
   279→   279→                })
   280→   280→              ).data
   281→   281→            : (
   282→   282→                await apiFetch<{ data: FormRecord }>(
   283→   283→                  `${basePath}/records/${encodeURIComponent(current?.id ?? '')}`,
   284→   284→                  { method: 'PUT', body: { data: draft } },
   285→   285→                )
   286→   286→              ).data;
   287→   287→
   288→   288→        const list = await refreshRecords();
   289→   289→        if (options?.keepRecord !== false) {
   290→   290→          const index = list.findIndex((record) => record.id === saved.id);
   291→   291→          if (index >= 0) {
   292→   292→            applyRecord(index, list);
   293→   293→          } else {
   294→   294→            setMode('view');
   295→   295→          }
   296→   296→        }
   297→   297→        return 'saved';
   298→   298→      } catch (error) {
   299→   299→        if (!(error instanceof SessionExpiredError)) {
   300→   300→          toast({
   301→   301→            title: t('formWindow.saveError'),
   302→   302→            description: error instanceof Error ? error.message : undefined,
   303→   303→            variant: 'destructive',
   304→   304→          });
   305→   305→        }
   306→   306→        return 'error';
   307→   307→      } finally {
   308→   308→        setIsSaving(false);
   309→   309→      }
   310→   310→    },
   311→   311→    [
   312→   312→      applyRecord,
   313→   313→      basePath,
   314→   314→      current,
   315→   315→      def,
   316→   316→      draft,
   317→   317→      isDeleting,
   318→   318→      isDirty,
   319→   319→      isSaving,
   320→   320→      locale,
   321→   321→      mode,
   322→   322→      refreshRecords,
   323→   323→      t,
   324→   324→    ],
   325→   325→  );
   326→   326→
   327→   327→  const handleSave = useCallback(async () => {
   328→   328→    const result = await performSave();
   329→   329→    if (result === 'saved') toast({ title: t('formWindow.saveSuccess') });
   330→   330→  }, [performSave, t]);
   331→   331→
   332→   332→  /** Save & new: write the record, then reset the form for a fresh entry. */
   333→   333→  const handleSaveNew = useCallback(async () => {
   334→   334→    if (isSaving || isDeleting) return;
   335→   335→    const result = await performSave({ keepRecord: false });
   336→   336→    if (result === 'error' || result === 'invalid') return;
   337→   337→    startNew();
   338→   338→    if (result === 'saved') toast({ title: t('formWindow.saveNewSuccess') });
   339→   339→  }, [isDeleting, isSaving, performSave, startNew, t]);
   340→   340→
   341→   341→  /** Save & close: write the record after validation, then close. */
   342→   342→  const handleSaveClose = useCallback(async () => {
   343→   343→    if (isSaving || isDeleting) return;
   344→   344→    const result = await performSave();
   345→   345→    if (result === 'error' || result === 'invalid') return;
   346→   346→    if (result === 'saved') toast({ title: t('formWindow.saveCloseSuccess') });
   347→   347→    onClose?.();
   348→   348→  }, [isDeleting, isSaving, onClose, performSave, t]);
   349→   349→
   350→   350→  /** First click on any delete action opens the confirm dialog. */
   351→   351→  const requestDelete = useCallback(
   352→   352→    (intent: Exclude<DeleteIntent, null>) => {
   353→   353→      if (!current || mode === 'new' || isSaving || isDeleting) return;
   354→   354→      setDeleteIntent(intent);
   355→   355→    },
   356→   356→    [current, isDeleting, isSaving, mode],
   357→   357→  );
   358→   358→
   359→   359→  /** Runs after the user presses «بلی» in the confirm dialog. */
   360→   360→  const confirmDelete = useCallback(async () => {
   361→   361→    if (isDeleting || !deleteIntent) return;
   362→   362→    const intent = deleteIntent;
   363→   363→    if (!basePath || !current) {
   364→   364→      setDeleteIntent(null);
   365→   365→      return;
   366→   366→    }
   367→   367→    setIsDeleting(true);
   368→   368→    try {
   369→   369→      // DELETE answers 204 with an empty body — nothing to parse.
   370→   370→      await apiFetch(
   371→   371→        `${basePath}/records/${encodeURIComponent(current.id)}`,
   372→   372→        {
   373→   373→          method: 'DELETE',
   374→   374→        },
   375→   375→      );
   376→   376→      setDeleteIntent(null);
   377→   377→      if (intent === 'close') {
   378→   378→        toast({ title: t('formWindow.deleteCloseSuccess') });
   379→   379→        onClose?.();
   380→   380→        return;
   381→   381→      }
   382→   382→      const list = await refreshRecords();
   383→   383→      if (intent === 'new') {
   384→   384→        startNew();
   385→   385→        toast({ title: t('formWindow.deleteNewSuccess') });
   386→   386→      } else if (list.length) {
   387→   387→        applyRecord(Math.min(currentIndex, list.length - 1), list);
   388→   388→        toast({ title: t('formWindow.deleteSuccess') });
   389→   389→      } else {
   390→   390→        setCurrentIndex(-1);
   391→   391→        startNew();
   392→   392→        toast({ title: t('formWindow.deleteSuccess') });
   393→   393→      }
   394→   394→    } catch (error) {
   395→   395→      setDeleteIntent(null);
   396→   396→      if (error instanceof SessionExpiredError) return;
   397→   397→      toast({
   398→   398→        title: error instanceof Error && error.message
   399→   399→          ? error.message
   400→   400→          : t('formWindow.loadError'),
   401→   401→        variant: 'destructive',
   402→   402→      });
   403→   403→    } finally {
   404→   404→      setIsDeleting(false);
   405→   405→    }
   406→   406→  }, [
   407→   407→    applyRecord,
   408→   408→    basePath,
   409→   409→    currentIndex,
   410→   410→    current,
   411→   411→    deleteIntent,
   412→   412→    isDeleting,
   413→   413→    onClose,
   414→   414→    refreshRecords,
   415→   415→    startNew,
   416→   416→    t,
   417→   417→  ]);
   418→   418→
   419→   419→  const handlePrint = useCallback(() => {
   420→   420→    toast({ title: t('formWindow.printToast') });
   421→   421→  }, [t]);
   422→   422→
   423→   423→  const handleFieldChange = useCallback(
   424→   424→    (fieldKey: string, value: FormFieldValue) => {
   425→   425→      setDraft((prev) => ({ ...prev, [fieldKey]: value }));
   426→   426→      setMode((prev) => (prev === 'view' ? 'edit' : prev));
   427→   427→    },
   428→   428→    [],
   429→   429→  );
   430→   430→
   431→   431→  // Ctrl+S / Cmd+S saves the active draft (Escape is handled by the dialog).
   432→   432→  useEffect(() => {
   433→   433→    const onKeyDown = (event: KeyboardEvent) => {
   434→   434→      if (
   435→   435→        (event.ctrlKey || event.metaKey) &&
   436→   436→        !event.shiftKey &&
   437→   437→        !event.altKey &&
   438→   438→        event.key.toLowerCase() === 's'
   439→   439→      ) {
   440→   440→        event.preventDefault();
   441→   441→        void handleSave();
   442→   442→      }
   443→   443→    };
   444→   444→    window.addEventListener('keydown', onKeyDown);
   445→   445→    return () => window.removeEventListener('keydown', onKeyDown);
   446→   446→  }, [handleSave]);
   447→   447→
   448→   448→  // ── Derived view models ──
   449→   449→  const sortedFields = useMemo(
   450→   450→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   451→   451→    [def],
   452→   452→  );
   453→   453→
   454→   454→  const tabs = useMemo(() => (def ? deriveTabs(def, locale) : []), [def, locale]);
   455→   455→  const activeTab =
   456→   456→    tabs.find((tab) => (tab.key ?? null) === (activeTabKey ?? null)) ??
   457→   457→    tabs[0] ??
   458→   458→    null;
   459→   459→
   460→   460→  const listColumns = useMemo(
   461→   461→    () => pickListColumns(sortedFields),
   462→   462→    [sortedFields],
   463→   463→  );
   464→   464→
   465→   465→  const title = def
   466→   466→    ? formTitle(def, locale)
   467→   467→    : form?.name ?? t('common.loading');
   468→   468→  const subtitle =
   469→   469→    mode === 'new'
   470→   470→      ? t('formWindow.newMode')
   471→   471→      : current
   472→   472→        ? tf('formWindow.recordOf', {
   473→   473→            n: formatNumber(currentIndex + 1),
   474→   474→            total: formatNumber(records.length),
   475→   475→          })
   476→   476→        : undefined;
   477→   477→
   478→   478→  const accentColor = def?.accentColor || DEFAULT_FORM_ACCENT;
   479→   479→
   480→   480→  /** The fields grid shared by simple forms and the active tab. */
   481→   481→  const renderFields = (fields: typeof sortedFields): ReactNode => (
   482→   482→    <div className={fieldsGridClasses}>
   483→   483→      {fields.map((field) => (
   484→   484→        <FieldRenderer
   485→   485→          key={field.fieldKey}
   486→   486→          field={field}
   487→   487→          value={draft[field.fieldKey] ?? null}
   488→   488→          onChange={(value) => handleFieldChange(field.fieldKey, value)}
   489→   489→          disabled={isSaving || isDeleting}
   490→   490→          lookupUrlBase={lookupUrlBase}
   491→   491→        />
   492→   492→      ))}
   493→   493→    </div>
   494→   494→  );
   495→   495→
   496→   496→  let content: ReactNode;
   497→   497→  if (!def) {
