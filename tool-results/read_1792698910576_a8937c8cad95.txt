    77→/** Fallback parsing of the tab id when no descriptor is attached. */
    78→function parseListModuleId(moduleId: string): {
    79→  moduleKey: string | null;
    80→  formKey: string | null;
    81→} {
    82→  if (!moduleId.startsWith('list:')) {
    83→    return { moduleKey: null, formKey: null };
    84→  }
    85→  const formKey = moduleId.slice('list:'.length);
    86→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    87→}

   101→/**
   102→ * Read the rendered width of every data column from its header cell.
   103→ * Used to freeze the whole grid the first time the user grabs a handle,
   104→ * so resizing one column never reflows its neighbours.
   105→ */
   106→function measureColumns(table: HTMLTableElement | null): Record<string, number> | null {
   107→  if (!table) return null;
   108→  const measured: Record<string, number> = {};
   109→  table
   110→    .querySelectorAll<HTMLTableCellElement>('th[data-col]')
   111→    .forEach((th) => {
   112→      const key = th.dataset.col;
   113→      if (key) measured[key] = th.getBoundingClientRect().width;
   114→    });
   115→  return Object.keys(measured).length ? measured : null;
   116→}

   186→ * Generic, database-driven records list («فهرست …»): every form saved
   187→ * through the unified form system automatically gets one of these windows
   188→ * with a filterable grid, multi-select, column chooser and the ERP-style
   189→ * footer icon strip (new / edit / delete / view / export / print /
   190→ * refresh / columns / filter).
   191→ */
   192→export function DataListWindow({ moduleId, list, onClose }: DataListWindowProps) {
   193→  const { locale, t, tf, formatNumber, direction } = useTranslation();
   194→  const addFormTab = useDesktopStore((s) => s.addFormTab);
   195→
   196→  const fallback = useMemo(() => parseListModuleId(moduleId), [moduleId]);
   197→  const moduleKey = list?.moduleKey ?? fallback.moduleKey;
   198→  const formKey = list?.formKey ?? fallback.formKey;
   199→  const basePath =
   200→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
   201→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   202→
   203→  // ── Data ──
   204→  const [def, setDef] = useState<FormDef | null>(null);
   205→  /** Full dataset of the form (parent-title context + total count). */
   206→  const [allRecords, setAllRecords] = useState<FormRecord[]>([]);
   207→  /** Rows currently displayed — the server-filtered slice. */
   208→  const [records, setRecords] = useState<FormRecord[]>([]);
   209→  const [isLoading, setIsLoading] = useState(true);
   210→  const [isRefreshing, setIsRefreshing] = useState(false);
   211→  /** True while a debounced server-side filter query is in flight. */
   212→  const [isFiltering, setIsFiltering] = useState(false);
   213→  const [loadError, setLoadError] = useState<string | null>(null);
   214→  /** lookupKey → (option code → label). */
   215→  const [lookupLabels, setLookupLabels] = useState<
   216→    Record<string, Map<string, string>>
   217→  >({});
   218→
   219→  // ── View state ──
   220→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   221→  const [filters, setFilters] = useState<Record<string, ColumnFilterState>>({});
   222→  const [isFilterRowVisible, setIsFilterRowVisible] = useState(true);
   223→  /** Open operator menu, anchored to the funnel button of one column. */
   224→  const [filterMenu, setFilterMenu] = useState<{
   225→    fieldKey: string;
   226→    rect: DOMRect;
   227→    left: number;
   228→    openUp: boolean;
   229→  } | null>(null);
   230→  const [hiddenColumns, setHiddenColumns] = useState<Set<string>>(new Set());
   231→  /**
   232→   * Explicit px width per data column — `null` = classic auto layout that
   233→   * fills the window. Set on the first resize interaction (all columns are
   234→   * frozen at their rendered widths, then only the grabbed one moves).
   235→   */
   236→  const [columnWidths, setColumnWidths] = useState<Record<string, number> | null>(null);
   237→  /** Field currently being drag-resized (drives cursors + handle hint). */
   238→  const [resizingField, setResizingField] = useState<string | null>(null);
   239→  const [isColumnsOpen, setIsColumnsOpen] = useState(false);
   240→  const [isDeleteConfirmOpen, setIsDeleteConfirmOpen] = useState(false);
   241→  const [isDeleting, setIsDeleting] = useState(false);
   242→
   243→  /** Latest values for callbacks that must not re-create on every change. */
   244→  const activeFiltersRef = useRef<ListColumnFilter[]>([]);
   245→  const allRecordsRef = useRef<FormRecord[]>([]);
   246→  const filterSeqRef = useRef(0);
   247→  const filterMenuRef = useRef<HTMLDivElement | null>(null);
   248→  const filterInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
   249→  const tableRef = useRef<HTMLTableElement | null>(null);
   250→
   251→  // ── Loading ──
   252→  /** Records URL with the JSON `?filters=` payload (server-side filtering). */
   253→  const fetchRecordsPage = useCallback(
   254→    async (columns: ListColumnFilter[]): Promise<FormRecord[]> => {
   255→      if (!basePath) return [];
   256→      const params = new URLSearchParams({ locale });
   257→      if (columns.length) {
   258→        params.set('filters', JSON.stringify(columns));
   259→      }
   260→      const res = await apiFetch<{ data: FormRecord[] }>(
   261→        `${basePath}/records?${params.toString()}`,
   262→      );
   263→      return res.data ?? [];
   264→    },
   265→    [basePath, locale],
   266→  );
   267→  const loadLookups = useCallback(
   268→    async (formDef: FormDef) => {
   269→      const keys = Array.from(
   270→        new Set(
   271→          formDef.fields
   272→            .filter((field) => field.fieldType === 'lookup' && field.lookupKey)
   273→            .map((field) => field.lookupKey as string),
   274→        ),
   275→      );
   276→      if (!keys.length || !lookupUrlBase) return;
   277→      const results = await Promise.all(
   278→        keys.map((key) =>
   279→          apiFetch<{ data: LookupOption[] }>(
   280→            `${lookupUrlBase}/${encodeURIComponent(key)}?locale=${encodeURIComponent(locale)}`,
   281→          )
   282→            .then((payload): [string, Map<string, string>] => [
   283→              key,
   284→              new Map(
   285→                (payload.data ?? []).map((option) => [
   286→                  option.code,
   287→                  locale === 'fa' ? option.label : (option.labelEn ?? option.label),
   288→                ]),
   289→              ),
   290→            ])
   291→            .catch((): [string, Map<string, string>] => [key, new Map()]),
   292→        ),
   293→      );
   294→      setLookupLabels(Object.fromEntries(results));
   295→    },
   296→    [locale, lookupUrlBase],
   297→  );
   298→
   299→  const load = useCallback(async () => {
   300→    if (!basePath) {
   301→      setLoadError(GENERIC_LOAD_ERROR);
   302→      setIsLoading(false);
   303→      return;
   304→    }
   305→    setIsLoading(true);
   306→    setLoadError(null);
   307→    try {
   308→      const defRes = await apiFetch<{ data: FormDef }>(
   309→        `${basePath}?locale=${encodeURIComponent(locale)}`,
   310→      );
   311→      const nextDef = defRes.data;
   312→      setDef(nextDef);
   313→      setSelectedIds(new Set());
   314→
   315→      const all = await fetchRecordsPage([]);
   316→      setAllRecords(all);
   317→      setRecords(all);
   318→      void loadLookups(nextDef);
   319→    } catch (error) {
   320→      if (error instanceof SessionExpiredError) return;
   321→      setLoadError(
   322→        error instanceof Error && error.message
   323→          ? error.message
   324→          : GENERIC_LOAD_ERROR,
   325→      );
   326→    } finally {
   327→      setIsLoading(false);
   328→    }
   329→  }, [basePath, fetchRecordsPage, loadLookups, locale]);
   330→
   331→  useEffect(() => {
   332→    void load();
   333→  }, [load]);
   334→
   335→  const refreshRecords = useCallback(async () => {
   336→    if (!basePath) return;
   337→    setIsRefreshing(true);
   338→    try {
   339→      const next = await fetchRecordsPage([]);
   340→      setAllRecords(next);
   341→      const columns = activeFiltersRef.current;
   342→      // Keep the visible slice consistent with the active column filters.
   343→      const visible = columns.length
   344→        ? await fetchRecordsPage(columns)
   345→        : next;
   346→      setRecords(visible);
   347→      setSelectedIds((prev) => {
   348→        const nextIds = new Set(
   349→          [...prev].filter((id) => visible.some((record) => record.id === id)),
   350→        );
   351→        return nextIds.size === prev.size ? prev : nextIds;
   352→      });
   353→    } catch (error) {
   354→      if (error instanceof SessionExpiredError) return;
   355→      toast({
   356→        title: t('dataList.loadError'),
   357→        description: error instanceof Error ? error.message : undefined,
   358→        variant: 'destructive',
   359→      });
   360→    } finally {
   361→      setIsRefreshing(false);
   362→    }
   363→  }, [basePath, fetchRecordsPage, t]);
   364→
   365→  // Auto-refresh when a form window of the same form mutates records.
   366→  useEffect(() => {
   367→    if (!moduleKey || !formKey) return;
   368→    const handler = (event: Event) => {
   369→      const detail = (
   370→        event as CustomEvent<{ moduleKey?: string; formKey?: string }>
   371→      ).detail;
   372→      if (detail?.moduleKey === moduleKey && detail?.formKey === formKey) {
   373→        void refreshRecords();
   374→      }
   375→    };
   376→    window.addEventListener(RECORDS_CHANGED_EVENT, handler);
   377→    return () => window.removeEventListener(RECORDS_CHANGED_EVENT, handler);
   378→  }, [formKey, moduleKey, refreshRecords]);
   379→
   380→  // ── Derived view models ──
   381→  const sortedFields = useMemo(
   382→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   383→    [def],
   384→  );
   385→  const parentField = useMemo(
   386→    () => (def?.type === 'tree' ? pickParentField(sortedFields) : null),
   387→    [def, sortedFields],
   388→  );
   389→  const visibleFields = useMemo(
   390→    () => sortedFields.filter((field) => !hiddenColumns.has(field.fieldKey)),
   391→    [hiddenColumns, sortedFields],
   392→  );
   393→
   394→  /** recordId → display title of that record (tree parent columns). */
   395→  const parentTitles = useMemo(() => {
   396→    const map = new Map<string, string>();
   397→    if (!parentField) return map;
   398→    const titleField =
   399→      sortedFields.find(
   400→        (field) =>
   401→          field !== parentField && /title|name|desc/i.test(field.fieldKey),
   402→      ) ??
   403→      sortedFields.find((field) => field !== parentField) ??
   404→      null;
   405→    for (const record of allRecords) {
   406→      const raw = titleField ? record.data[titleField.fieldKey] : null;
   407→      map.set(
   408→        record.id,
   409→        raw === null || raw === undefined || raw === ''
   410→          ? record.id
   411→          : String(raw),
   412→      );
   413→    }
   414→    return map;
   415→  }, [parentField, allRecords, sortedFields]);
   416→
   417→  /** Display text of one cell (checkbox → بلی/خیر, lookup → label, …). */
   418→  const cellText = useCallback(
   419→    (field: FormFieldDef, record: FormRecord): string => {
   420→      const raw = record.data[field.fieldKey];
   421→      if (field.fieldType === 'checkbox') {
   422→        return raw === true ? t('dataList.yes') : t('dataList.no');
   423→      }
   424→      // Tree parent columns resolve the parent record's title (the
   425→      // stored value is the parent record id — must run before lookup).
   426→      if (parentField && field.fieldKey === parentField.fieldKey) {
   427→        if (raw === null || raw === undefined || raw === '') return '—';
   428→        return parentTitles.get(String(raw)) ?? String(raw);
   429→      }
   430→      if (field.fieldType === 'lookup') {
   431→        if (raw === null || raw === undefined || raw === '') return '—';
   432→        const labels = field.lookupKey
   433→          ? lookupLabels[field.lookupKey]
   434→          : undefined;
   435→        return labels?.get(String(raw)) ?? String(raw);
   436→      }
   437→      return formatFieldValue(field, raw, formatNumber);
   438→    },
   439→    [formatNumber, lookupLabels, parentField, parentTitles, t],
   440→  );
   441→
   442→  /**
   443→   * Effective column filters sent to the service as `?filters=` JSON —
   444→   * the grid itself no longer filters client-side.
   445→   */
   446→  const activeFilters = useMemo<ListColumnFilter[]>(
   447→    () =>
   448→      Object.entries(filters)
   449→        .filter(([, filter]) => isColumnFilterActive(filter))
   450→        .map(([field, filter]) => ({
   451→          field,
   452→          op: filter.op,
   453→          value: filter.value,
   454→        })),
   455→    [filters],
   456→  );
   457→
   458→  useEffect(() => {
   459→    activeFiltersRef.current = activeFilters;
   460→    allRecordsRef.current = allRecords;
   461→  }, [activeFilters, allRecords]);
   462→
   463→  // Server-side filtering: debounce typed values, then re-query the service.
   464→  useEffect(() => {
   465→    const handle = window.setTimeout(() => {
   466→      if (!activeFilters.length) {
   467→        setRecords(allRecordsRef.current);
   468→        setIsFiltering(false);
   469→        return;
   470→      }
   471→      const seq = ++filterSeqRef.current;
   472→      setIsFiltering(true);
   473→      fetchRecordsPage(activeFilters)
   474→        .then((next) => {
   475→          if (filterSeqRef.current !== seq) return;
   476→          setRecords(next);
   477→        })
   478→        .catch((error) => {
   479→          if (filterSeqRef.current !== seq) return;
   480→          if (error instanceof SessionExpiredError) return;
   481→          toast({
   482→            title: t('dataList.loadError'),
   483→            description: error instanceof Error ? error.message : undefined,
   484→            variant: 'destructive',
   485→          });
   486→        })
   487→        .finally(() => {
   488→          if (filterSeqRef.current === seq) setIsFiltering(false);
   489→        });
   490→    }, 250);
   491→    return () => window.clearTimeout(handle);
   492→  }, [activeFilters, fetchRecordsPage, t]);
   493→
   494→  /** The displayed rows — already filtered by the service. */
   495→  const filteredRecords = records;
   496→
   497→  // ── Column filter operator menu ──
   498→  const toggleFilterMenu = useCallback(
   499→    (fieldKey: string, button: HTMLButtonElement) => {
   500→      setFilterMenu((prev) => {
   501→        if (prev?.fieldKey === fieldKey) return null;
   502→        const rect = button.getBoundingClientRect();
   503→        const menuWidth = 184;
   504→        const menuHeight = 290;
   505→        return {
   506→          fieldKey,
   507→          rect,
   508→          left: Math.max(
   509→            8,
   510→            Math.min(rect.left, window.innerWidth - menuWidth - 8),
   511→          ),
   512→          openUp:
   513→            rect.bottom + menuHeight > window.innerHeight - 8 &&
   514→            rect.top - menuHeight > 8,
   515→        };
   516→      });
   517→    },
   518→    [],
   519→  );
   520→
   521→  const setFilterValue = useCallback((fieldKey: string, value: string) => {
   522→    setFilters((prev) => {
   523→      const current = prev[fieldKey];
   524→      // Typing always implies value semantics: a fresh entry (and an entry
   525→      // whose operator carries no value — تمام / دارای مقدار / بدون مقدار)
   526→      // becomes «حاوی تایپ‌شده» so the input is never a dead end.
   527→      const op: ListFilterOp =
   528→        !current || current.op === 'hasValue' || current.op === 'noValue'
   529→          ? 'contains'
   530→          : current.op;
   531→      return { ...prev, [fieldKey]: { op, value } };
   532→    });
   533→  }, []);
   534→
   535→  const applyFilterOp = useCallback((fieldKey: string, op: ListFilterOp) => {
   536→    setFilterMenu(null);
   537→    setFilters((prev) => {
   538→      const next = { ...prev };
   539→      if (op === 'all') {
   540→        delete next[fieldKey];
   541→        return next;
   542→      }
   543→      const current = prev[fieldKey];
   544→      next[fieldKey] = {
   545→        op,
   546→        value:
   547→          op === 'hasValue' || op === 'noValue' ? '' : (current?.value ?? ''),
   548→      };
   549→      return next;
   550→    });
   551→    if (op !== 'hasValue' && op !== 'noValue' && op !== 'all') {
   552→      requestAnimationFrame(() => filterInputRefs.current[fieldKey]?.focus());
   553→    }
   554→  }, []);
   555→
   556→  // Dismiss the operator menu on outside click / Escape / scroll / resize.
   557→  useEffect(() => {
   558→    if (!filterMenu) return;
   559→    const close = () => setFilterMenu(null);
   560→    const onPointerDown = (event: PointerEvent) => {
   561→      const target = event.target;
   562→      if (!(target instanceof Node)) return;
   563→      if (filterMenuRef.current?.contains(target)) return;
   564→      if (
   565→        target instanceof Element &&
   566→        target.closest('[data-filter-op-button]')
   567→      ) {
   568→        return;
   569→      }
   570→      close();
   571→    };
   572→    const onKeyDown = (event: KeyboardEvent) => {
   573→      if (event.key === 'Escape') close();
   574→    };
   575→    window.addEventListener('pointerdown', onPointerDown);
   576→    window.addEventListener('keydown', onKeyDown);

   599→  const beginColumnResize = useCallback(

   601→      event.preventDefault();
   602→      event.stopPropagation();

   604→      // changes ONLY the grabbed column (no neighbour reflow).
   605→      const measured = measureColumns(tableRef.current);
   606→      if (!measured) return;
   607→      const startWidth = measured[fieldKey] ?? DEFAULT_COLUMN_WIDTH;
   608→      setColumnWidths(measured);
   609→      setResizingField(fieldKey);
   610→      // The handle sits on the inline-end edge: in RTL that edge grows
   611→      // leftwards, so the clientX delta flips sign.
   612→      const resizeDir = direction === 'rtl' ? -1 : 1;
   613→      const startX = event.clientX;
   614→      const onMove = (moveEvent: MouseEvent) => {
   615→        const next = Math.round(
   616→          startWidth + (moveEvent.clientX - startX) * resizeDir,
   617→        );
   618→        const clamped = Math.min(
   619→          MAX_COLUMN_WIDTH,
   620→          Math.max(MIN_COLUMN_WIDTH, next),
   621→        );
   622→        setColumnWidths((prev) =>
   623→          prev ? { ...prev, [fieldKey]: clamped } : prev,
   624→        );
   625→      };
   626→      const onUp = () => {
   627→        window.removeEventListener('mousemove', onMove);
   628→        window.removeEventListener('mouseup', onUp);
   629→        document.body.style.userSelect = '';
   630→        setResizingField(null);
   631→      };
   632→      document.body.style.userSelect = 'none';
   633→      window.addEventListener('mousemove', onMove);
   634→      window.addEventListener('mouseup', onUp);
   635→    },
   636→    [direction],
   637→  );

   639→  /** Keyboard resize (Arrow keys) — freezes the grid on first use too. */

   645→        MAX_COLUMN_WIDTH,
   646→        Math.max(MIN_COLUMN_WIDTH, current + sign * 24),
   647→      );
   648→      return { ...base, [fieldKey]: next };
   649→    });
   650→  }, []);

   653→  const resetColumnWidths = useCallback(() => setColumnWidths(null), []);

   663→    });

   748→      });
   749→    } finally {
   750→      setIsDeleting(false);
   751→    }
   752→  }, [
   753→    basePath,
   754→    formatNumber,
   755→    formKey,
   756→    isDeleting,
   757→    moduleKey,
   758→    selectedIds,

   771→      escapeCell(fieldLabel(field, locale)),

   774→      visibleFields.map((field) => escapeCell(cellText(field, record))),

   777→      '\uFEFF' + [header, ...rows].map((row) => row.join(',')).join('\r\n');

   794→  const toggleColumn = useCallback((fieldKey: string) => {

   803→  // ── Chrome texts ──
   804→  const listName = def
   805→    ? formTitle(def, locale)

   808→    name: listName,
   809→    count: formatNumber(filteredRecords.length),
   810→  });

   812→    ? tf('dataList.selectedCount', { count: formatNumber(selectedIds.size) })
   813→    : tf('dataList.recordCount', { count: formatNumber(allRecords.length) });
   814→
   815→  const accentColor = def?.accentColor || DEFAULT_FORM_ACCENT;
   816→  const hasActiveFilters = activeFilters.length > 0;
   817→
   818→  const errorText =
   819→    loadError === GENERIC_LOAD_ERROR ? t('dataList.loadError') : loadError;

   821→  /** The grid (or its loading / error / empty states). */

   832→      </div>
   833→    );

   838→          className={cn(

   866→            <thead>

   870→                    'sticky top-0 z-30 h-9 border-b border-white/10 bg-[#1a1d26] px-2',

   875→                    checked={allVisibleSelected}
   876→                    onChange={toggleSelectAll}

   880→                {visibleFields.map((field) => (

   882→                    key={field.fieldKey}

   914→                    >
   915→                      <span
   916→                        aria-hidden="true"
   917→                        className={cn(
   918→                          'h-full w-px transition-colors group-hover/th:bg-sky-400/60',
   919→                          resizingField === field.fieldKey
   920→                            ? 'bg-sky-400'
   921→                            : 'bg-white/10',
   922→                        )}
   923→                      />
   924→                    </button>
   925→                  </th>

   938→                    >
   939→                      <Filter className="h-3.5 w-3.5" />

  1083→  return (
