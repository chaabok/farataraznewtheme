    77→/** Fallback parsing of the tab id when no descriptor is attached. */
    78→function parseListModuleId(moduleId: string): {
    79→  moduleKey: string | null;
    80→  formKey: string | null;
    81→} {
    82→  if (!moduleId.startsWith('list:')) {
    83→    return { moduleKey: null, formKey: null };
    84→  }
    85→  const formKey = moduleId.slice('list:'.length);
    86→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    87→}

   114→/** True when a column contributes an effective server-side filter. */
   115→function isColumnFilterActive(filter?: ColumnFilterState): boolean {
   116→  return (
   117→    !!filter &&
   118→    (filter.op === 'hasValue' ||
   119→      filter.op === 'noValue' ||
   120→      filter.value.trim() !== '')
   121→  );
   122→}

   159→/**
   160→ * Generic, database-driven records list («فهرست …»): every form saved
   161→ * through the unified form system automatically gets one of these windows
   162→ * with a filterable grid, multi-select, column chooser and the ERP-style
   163→ * footer icon strip (new / edit / delete / view / export / print /
   164→ * refresh / columns / filter).
   165→ */
   166→export function DataListWindow({ moduleId, list, onClose }: DataListWindowProps) {
   167→  const { locale, t, tf, formatNumber } = useTranslation();
   168→  const addFormTab = useDesktopStore((s) => s.addFormTab);
   169→
   170→  const fallback = useMemo(() => parseListModuleId(moduleId), [moduleId]);
   171→  const moduleKey = list?.moduleKey ?? fallback.moduleKey;
   172→  const formKey = list?.formKey ?? fallback.formKey;
   173→  const basePath =
   174→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
   175→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   176→
   177→  // ── Data ──
   178→  const [def, setDef] = useState<FormDef | null>(null);
   179→  /** Full dataset of the form (parent-title context + total count). */
   180→  const [allRecords, setAllRecords] = useState<FormRecord[]>([]);
   181→  /** Rows currently displayed — the server-filtered slice. */
   182→  const [records, setRecords] = useState<FormRecord[]>([]);
   183→  const [isLoading, setIsLoading] = useState(true);
   184→  const [isRefreshing, setIsRefreshing] = useState(false);
   185→  /** True while a debounced server-side filter query is in flight. */
   186→  const [isFiltering, setIsFiltering] = useState(false);
   187→  const [loadError, setLoadError] = useState<string | null>(null);
   188→  /** lookupKey → (option code → label). */
   189→  const [lookupLabels, setLookupLabels] = useState<
   190→    Record<string, Map<string, string>>
   191→  >({});
   192→
   193→  // ── View state ──
   194→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   195→  const [filters, setFilters] = useState<Record<string, ColumnFilterState>>({});
   196→  const [isFilterRowVisible, setIsFilterRowVisible] = useState(true);
   197→  /** Open operator menu, anchored to the funnel button of one column. */
   198→  const [filterMenu, setFilterMenu] = useState<{
   199→    fieldKey: string;
   200→    rect: DOMRect;
   201→    left: number;
   202→    openUp: boolean;
   203→  } | null>(null);
   204→  const [hiddenColumns, setHiddenColumns] = useState<Set<string>>(new Set());
   205→  const [isColumnsOpen, setIsColumnsOpen] = useState(false);
   206→  const [isDeleteConfirmOpen, setIsDeleteConfirmOpen] = useState(false);
   207→  const [isDeleting, setIsDeleting] = useState(false);
   208→
   209→  /** Latest values for callbacks that must not re-create on every change. */
   210→  const activeFiltersRef = useRef<ListColumnFilter[]>([]);
   211→  const allRecordsRef = useRef<FormRecord[]>([]);
   212→  const filterSeqRef = useRef(0);
   213→  const filterMenuRef = useRef<HTMLDivElement | null>(null);
   214→  const filterInputRefs = useRef<Record<string, HTMLInputElement | null>>({});
   215→
   216→  // ── Loading ──
   217→  /** Records URL with the JSON `?filters=` payload (server-side filtering). */
   218→  const fetchRecordsPage = useCallback(
   219→    async (columns: ListColumnFilter[]): Promise<FormRecord[]> => {
   220→      if (!basePath) return [];
   221→      const params = new URLSearchParams({ locale });
   222→      if (columns.length) {
   223→        params.set('filters', JSON.stringify(columns));
   224→      }
   225→      const res = await apiFetch<{ data: FormRecord[] }>(
   226→        `${basePath}/records?${params.toString()}`,
   227→      );
   228→      return res.data ?? [];
   229→    },
   230→    [basePath, locale],
   231→  );
   232→  const loadLookups = useCallback(
   233→    async (formDef: FormDef) => {
   234→      const keys = Array.from(
   235→        new Set(
   236→          formDef.fields
   237→            .filter((field) => field.fieldType === 'lookup' && field.lookupKey)
   238→            .map((field) => field.lookupKey as string),
   239→        ),
   240→      );
   241→      if (!keys.length || !lookupUrlBase) return;
   242→      const results = await Promise.all(
   243→        keys.map((key) =>
   244→          apiFetch<{ data: LookupOption[] }>(
   245→            `${lookupUrlBase}/${encodeURIComponent(key)}?locale=${encodeURIComponent(locale)}`,
   246→          )
   247→            .then((payload): [string, Map<string, string>] => [
   248→              key,
   249→              new Map(
   250→                (payload.data ?? []).map((option) => [
   251→                  option.code,
   252→                  locale === 'fa' ? option.label : (option.labelEn ?? option.label),
   253→                ]),
   254→              ),
   255→            ])
   256→            .catch((): [string, Map<string, string>] => [key, new Map()]),
   257→        ),
   258→      );
   259→      setLookupLabels(Object.fromEntries(results));
   260→    },
   261→    [locale, lookupUrlBase],
   262→  );
   263→
   264→  const load = useCallback(async () => {
   265→    if (!basePath) {
   266→      setLoadError(GENERIC_LOAD_ERROR);
   267→      setIsLoading(false);
   268→      return;
   269→    }
   270→    setIsLoading(true);
   271→    setLoadError(null);
   272→    try {
   273→      const defRes = await apiFetch<{ data: FormDef }>(
   274→        `${basePath}?locale=${encodeURIComponent(locale)}`,
   275→      );
   276→      const nextDef = defRes.data;
   277→      setDef(nextDef);
   278→      setSelectedIds(new Set());
   279→
   280→      const all = await fetchRecordsPage([]);
   281→      setAllRecords(all);
   282→      setRecords(all);
   283→      void loadLookups(nextDef);
   284→    } catch (error) {
   285→      if (error instanceof SessionExpiredError) return;
   286→      setLoadError(
   287→        error instanceof Error && error.message
   288→          ? error.message
   289→          : GENERIC_LOAD_ERROR,
   290→      );
   291→    } finally {
   292→      setIsLoading(false);
   293→    }
   294→  }, [basePath, fetchRecordsPage, loadLookups, locale]);
   295→
   296→  useEffect(() => {
   297→    void load();
   298→  }, [load]);
   299→
   300→  const refreshRecords = useCallback(async () => {
   301→    if (!basePath) return;
   302→    setIsRefreshing(true);
   303→    try {
   304→      const next = await fetchRecordsPage([]);
   305→      setAllRecords(next);
   306→      const columns = activeFiltersRef.current;
   307→      // Keep the visible slice consistent with the active column filters.
   308→      const visible = columns.length
   309→        ? await fetchRecordsPage(columns)
   310→        : next;
   311→      setRecords(visible);
   312→      setSelectedIds((prev) => {
   313→        const nextIds = new Set(
   314→          [...prev].filter((id) => visible.some((record) => record.id === id)),
   315→        );
   316→        return nextIds.size === prev.size ? prev : nextIds;
   317→      });
   318→    } catch (error) {
   319→      if (error instanceof SessionExpiredError) return;
   320→      toast({
   321→        title: t('dataList.loadError'),
   322→        description: error instanceof Error ? error.message : undefined,
   323→        variant: 'destructive',
   324→      });
   325→    } finally {
   326→      setIsRefreshing(false);
   327→    }
   328→  }, [basePath, fetchRecordsPage, t]);
   329→
   330→  // Auto-refresh when a form window of the same form mutates records.
   331→  useEffect(() => {
   332→    if (!moduleKey || !formKey) return;
   333→    const handler = (event: Event) => {
   334→      const detail = (
   335→        event as CustomEvent<{ moduleKey?: string; formKey?: string }>
   336→      ).detail;
   337→      if (detail?.moduleKey === moduleKey && detail?.formKey === formKey) {
   338→        void refreshRecords();
   339→      }
   340→    };
   341→    window.addEventListener(RECORDS_CHANGED_EVENT, handler);
   342→    return () => window.removeEventListener(RECORDS_CHANGED_EVENT, handler);
   343→  }, [formKey, moduleKey, refreshRecords]);
   344→
   345→  // ── Derived view models ──
   346→  const sortedFields = useMemo(
   347→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   348→    [def],
   349→  );
   350→  const parentField = useMemo(
   351→    () => (def?.type === 'tree' ? pickParentField(sortedFields) : null),
   352→    [def, sortedFields],
   353→  );
   354→  const visibleFields = useMemo(
   355→    () => sortedFields.filter((field) => !hiddenColumns.has(field.fieldKey)),
   356→    [hiddenColumns, sortedFields],
   357→  );
   358→
   359→  /** recordId → display title of that record (tree parent columns). */
   360→  const parentTitles = useMemo(() => {
   361→    const map = new Map<string, string>();
   362→    if (!parentField) return map;
   363→    const titleField =
   364→      sortedFields.find(
   365→        (field) =>
   366→          field !== parentField && /title|name|desc/i.test(field.fieldKey),
   367→      ) ??
   368→      sortedFields.find((field) => field !== parentField) ??
   369→      null;
   370→    for (const record of allRecords) {
   371→      const raw = titleField ? record.data[titleField.fieldKey] : null;
   372→      map.set(
   373→        record.id,
   374→        raw === null || raw === undefined || raw === ''
   375→          ? record.id
   376→          : String(raw),
   377→      );
   378→    }
   379→    return map;
   380→  }, [parentField, allRecords, sortedFields]);
   381→
   382→  /** Display text of one cell (checkbox → بلی/خیر, lookup → label, …). */
   383→  const cellText = useCallback(
   384→    (field: FormFieldDef, record: FormRecord): string => {
   385→      const raw = record.data[field.fieldKey];
   386→      if (field.fieldType === 'checkbox') {
   387→        return raw === true ? t('dataList.yes') : t('dataList.no');
   388→      }
   389→      // Tree parent columns resolve the parent record's title (the
   390→      // stored value is the parent record id — must run before lookup).
   391→      if (parentField && field.fieldKey === parentField.fieldKey) {
   392→        if (raw === null || raw === undefined || raw === '') return '—';
   393→        return parentTitles.get(String(raw)) ?? String(raw);
   394→      }
   395→      if (field.fieldType === 'lookup') {
   396→        if (raw === null || raw === undefined || raw === '') return '—';
   397→        const labels = field.lookupKey
   398→          ? lookupLabels[field.lookupKey]
   399→          : undefined;
   400→        return labels?.get(String(raw)) ?? String(raw);
   401→      }
   402→      return formatFieldValue(field, raw, formatNumber);
   403→    },
   404→    [formatNumber, lookupLabels, parentField, parentTitles, t],
   405→  );
   406→
   407→  /**
   408→   * Effective column filters sent to the service as `?filters=` JSON —
   409→   * the grid itself no longer filters client-side.
   410→   */
   411→  const activeFilters = useMemo<ListColumnFilter[]>(
   412→    () =>
   413→      Object.entries(filters)
   414→        .filter(([, filter]) => isColumnFilterActive(filter))
   415→        .map(([field, filter]) => ({
   416→          field,
   417→          op: filter.op,
   418→          value: filter.value,
   419→        })),
   420→    [filters],
   421→  );
   422→
   423→  useEffect(() => {
   424→    activeFiltersRef.current = activeFilters;
   425→    allRecordsRef.current = allRecords;
   426→  }, [activeFilters, allRecords]);
   427→
   428→  // Server-side filtering: debounce typed values, then re-query the service.
   429→  useEffect(() => {
   430→    const handle = window.setTimeout(() => {
   431→      if (!activeFilters.length) {
   432→        setRecords(allRecordsRef.current);
   433→        setIsFiltering(false);
   434→        return;
   435→      }
   436→      const seq = ++filterSeqRef.current;
   437→      setIsFiltering(true);
   438→      fetchRecordsPage(activeFilters)
   439→        .then((next) => {
   440→          if (filterSeqRef.current !== seq) return;
   441→          setRecords(next);
   442→        })
   443→        .catch((error) => {
   444→          if (filterSeqRef.current !== seq) return;
   445→          if (error instanceof SessionExpiredError) return;
   446→          toast({
   447→            title: t('dataList.loadError'),
   448→            description: error instanceof Error ? error.message : undefined,
   449→            variant: 'destructive',
   450→          });
   451→        })
   452→        .finally(() => {
   453→          if (filterSeqRef.current === seq) setIsFiltering(false);
   454→        });
   455→    }, 250);
   456→    return () => window.clearTimeout(handle);
   457→  }, [activeFilters, fetchRecordsPage, t]);
   458→
   459→  /** The displayed rows — already filtered by the service. */
   460→  const filteredRecords = records;
   461→
   462→  // ── Column filter operator menu ──
   463→  const toggleFilterMenu = useCallback(
   464→    (fieldKey: string, button: HTMLButtonElement) => {
   465→      setFilterMenu((prev) => {
   466→        if (prev?.fieldKey === fieldKey) return null;
   467→        const rect = button.getBoundingClientRect();
   468→        const menuWidth = 184;
   469→        const menuHeight = 290;
   470→        return {
   471→          fieldKey,
   472→          rect,
   473→          left: Math.max(
   474→            8,
   475→            Math.min(rect.left, window.innerWidth - menuWidth - 8),
   476→          ),
   477→          openUp:
   478→            rect.bottom + menuHeight > window.innerHeight - 8 &&
   479→            rect.top - menuHeight > 8,
   480→        };
   481→      });
   482→    },
   483→    [],
   484→  );
   485→
   486→  const setFilterValue = useCallback((fieldKey: string, value: string) => {
   487→    setFilters((prev) => {
   488→      const current =
   489→        prev[fieldKey] ?? { op: 'contains' as ListFilterOp, value: '' };
   490→      return { ...prev, [fieldKey]: { ...current, value } };
   491→    });
   492→  }, []);
   493→
   494→  const applyFilterOp = useCallback((fieldKey: string, op: ListFilterOp) => {
   495→    setFilterMenu(null);
   496→    setFilters((prev) => {
   497→      const next = { ...prev };
   498→      if (op === 'all') {
   499→        delete next[fieldKey];
   500→        return next;
   501→      }
   502→      const current = prev[fieldKey];
   503→      next[fieldKey] = {
   504→        op,
   505→        value:
   506→          op === 'hasValue' || op === 'noValue' ? '' : (current?.value ?? ''),
   507→      };
   508→      return next;
   509→    });
   510→    if (op !== 'hasValue' && op !== 'noValue' && op !== 'all') {
   511→      requestAnimationFrame(() => filterInputRefs.current[fieldKey]?.focus());
   512→    }
   513→  }, []);
   514→
   515→  // Dismiss the operator menu on outside click / Escape / scroll / resize.
   516→  useEffect(() => {
   517→    if (!filterMenu) return;
   518→    const close = () => setFilterMenu(null);
   519→    const onPointerDown = (event: PointerEvent) => {
   520→      const target = event.target;
   521→      if (!(target instanceof Node)) return;
   522→      if (filterMenuRef.current?.contains(target)) return;
   523→      if (
   524→        target instanceof Element &&
   525→        target.closest('[data-filter-op-button]')
   526→      ) {
   527→        return;
   528→      }
   529→      close();
   530→    };
   531→    const onKeyDown = (event: KeyboardEvent) => {
   532→      if (event.key === 'Escape') close();
   533→    };
   534→    window.addEventListener('pointerdown', onPointerDown);
   535→    window.addEventListener('keydown', onKeyDown);
   536→    window.addEventListener('resize', close);
   537→    window.addEventListener('scroll', close, true);
   538→    return () => {
   539→      window.removeEventListener('pointerdown', onPointerDown);
   540→      window.removeEventListener('keydown', onKeyDown);
   541→      window.removeEventListener('resize', close);
   542→      window.removeEventListener('scroll', close, true);
   543→    };
   544→  }, [filterMenu]);
   545→
   546→  // ── Selection ──
   547→  const toggleSelected = useCallback((recordId: string, additive: boolean) => {
   548→    setSelectedIds((prev) => {
   549→      if (!additive) return prev.has(recordId) ? new Set([recordId]) : new Set([recordId]);
   550→      const next = new Set(prev);
   551→      if (next.has(recordId)) next.delete(recordId);
   552→      else next.add(recordId);
   553→      return next;
   554→    });
   555→  }, []);
   556→
   557→  const handleRowClick = useCallback(
   558→    (record: FormRecord, event: MouseEvent<HTMLTableRowElement>) => {
   559→      if (event.ctrlKey || event.metaKey) {
   560→        toggleSelected(record.id, true);
   561→      } else {
   562→        setSelectedIds((prev) =>
   563→          prev.size === 1 && prev.has(record.id) ? prev : new Set([record.id]),
   564→        );
   565→      }
   566→    },
   567→    [toggleSelected],
   568→  );
   569→
   570→  const allVisibleSelected =
   571→    filteredRecords.length > 0 &&
   572→    filteredRecords.every((record) => selectedIds.has(record.id));
   573→
   574→  const toggleSelectAll = useCallback(() => {
   575→    setSelectedIds((prev) => {
   576→      if (filteredRecords.length && filteredRecords.every((record) => prev.has(record.id))) {
   577→        return new Set();
   578→      }
   579→      return new Set(filteredRecords.map((record) => record.id));
   580→    });
   581→  }, [filteredRecords]);
   582→
   583→  // ── Form integration ──
