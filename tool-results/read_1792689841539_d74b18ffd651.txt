     4→  useCallback,
     5→  useEffect,
     6→  useMemo,
     7→  useState,
     8→  type ReactNode,
     9→} from 'react';
    10→import { AnimatePresence, motion } from 'framer-motion';
    11→import { ChevronUp, List, Loader2, RefreshCw, Trash2 } from 'lucide-react';
    12→import {
    13→  AlertDialog,
    14→  AlertDialogAction,
    15→  AlertDialogCancel,
    16→  AlertDialogContent,
    17→  AlertDialogDescription,
    18→  AlertDialogFooter,
    19→  AlertDialogHeader,

    21→} from '@/components/ui/alert-dialog';

    23→import { toast } from '@/hooks/use-toast';
    24→import { useTranslation } from '@/i18n';

    28→  FormDataMap,

    37→  DEFAULT_FORM_ACCENT,

    41→import { FormShell } from './form-shell';
    42→import { FormTabs } from './form-tabs';
    43→import {
    44→  buildDefaults,
    45→  deriveTabs,
    46→  fieldLabel,
    47→  formTitle,
    48→  isEmptyRequiredValue,
    49→  isChangedValue,
    50→  formatFieldValue,
    51→  pickListColumns,
    52→} from './form-utils';
    53→import { TreeForm } from './tree-form';
    54→
    55→export interface DataFormWindowProps {
    56→  /** Window id — `form:<formKey>` (moduleKey defaults to `settings`). */
    57→  moduleId: string;
    58→  /** Descriptor attached by `openFormWindow` in the desktop store. */
    59→  form?: FormWindowInfo;
    60→  /** Closes the hosting window ("save & close" / "delete & close"). */
    61→  onClose?: () => void;
    62→}
    63→
    64→/** Which combined action a pending delete confirmation belongs to. */
    65→type DeleteIntent = 'current' | 'new' | 'close' | null;
    66→
    67→/** Outcome of the shared save pipeline. */
    68→type SaveResult = 'saved' | 'clean' | 'invalid' | 'error';
    69→
    70→/** Fallback parsing of the window id when no descriptor is attached. */
    71→function parseModuleId(moduleId: string): {
    72→  moduleKey: string | null;
    73→  formKey: string | null;
    74→} {
    75→  if (!moduleId.startsWith('form:')) {
    76→    return { moduleKey: null, formKey: null };
    77→  }
    78→  const formKey = moduleId.slice('form:'.length);
    79→  return { moduleKey: formKey ? 'settings' : null, formKey: formKey || null };
    80→}
    81→
    82→/** Sentinel kept in state so load errors render through the dictionaries. */
    83→const GENERIC_LOAD_ERROR = 'formWindow.loadError';
    84→
    85→/**
    86→ * Generic, database-driven form window: loads a FormDef + its records and
    87→ * drives the whole view/new/edit state machine (dirty tracking, CRUD,
    88→ * record navigation, Ctrl+S, records drawer) through the shared form
    89→ * component library. One implementation renders every ERP form.
    90→ */
    91→export function DataFormWindow({ moduleId, form, onClose }: DataFormWindowProps) {
    92→  const { locale, direction, t, tf, formatNumber } = useTranslation();
    93→
    94→  const fallback = useMemo(() => parseModuleId(moduleId), [moduleId]);
    95→  const moduleKey = form?.moduleKey ?? fallback.moduleKey;
    96→  const formKey = form?.formKey ?? fallback.formKey;
    97→  const basePath =
    98→    moduleKey && formKey ? `/api/modules/${moduleKey}/forms/${formKey}` : null;
    99→  const lookupUrlBase = basePath ? `${basePath}/lookups` : '';
   100→
   101→  // ── Data ──
   102→  const [def, setDef] = useState<FormDef | null>(null);
   103→  const [records, setRecords] = useState<FormRecord[]>([]);
   104→  const [isLoading, setIsLoading] = useState(true);
   105→  const [loadError, setLoadError] = useState<string | null>(null);
   106→
   107→  // ── Record state machine ──
   108→  const [mode, setMode] = useState<FormMode>('view');
   109→  const [currentIndex, setCurrentIndex] = useState(-1);
   110→  const [draft, setDraft] = useState<FormDataMap>({});
   111→  const [baseline, setBaseline] = useState<FormDataMap>({});
   112→  const [isSaving, setIsSaving] = useState(false);
   113→  const [isDeleting, setIsDeleting] = useState(false);
   114→  /**
   115→   * Pending delete request awaiting the confirm dialog — `current` =
   116→   * delete and stay, `new` = delete then reset for a fresh entry,
   117→   * `close` = delete then close the window.
   118→   */
   119→  const [deleteIntent, setDeleteIntent] = useState<DeleteIntent>(null);
   120→  /** Record the confirmed delete targets (defaults to the loaded one). */
   121→  const [deleteTargetId, setDeleteTargetId] = useState<string | null>(null);
   122→  const [isListOpen, setIsListOpen] = useState(false);
   123→  const [activeTabKey, setActiveTabKey] = useState<string | null>(null);
   124→
   125→  const current =
   126→    currentIndex >= 0 && currentIndex < records.length
   127→      ? records[currentIndex]
   128→      : null;
   129→
   130→  const isDirty = useMemo(() => {
   131→    const keys = new Set([...Object.keys(baseline), ...Object.keys(draft)]);
   132→    for (const key of keys) {
   133→      if (isChangedValue(baseline[key], draft[key])) return true;
   134→    }
   135→    return false;
   136→  }, [baseline, draft]);
   137→
   138→  const status: FormStatus =
   139→    mode === 'new' ? (isDirty ? 'unsaved' : 'new') : isDirty ? 'unsaved' : 'saved';
   140→
   141→  // ── Loading ──
   142→  const load = useCallback(async () => {
   143→    if (!basePath) {
   144→      setLoadError(GENERIC_LOAD_ERROR);
   145→      setIsLoading(false);
   146→      return;
   147→    }
   148→    setIsLoading(true);
   149→    setLoadError(null);
   150→    try {
   151→      const [defRes, recordsRes] = await Promise.all([
   152→        apiFetch<{ data: FormDef }>(
   153→          `${basePath}?locale=${encodeURIComponent(locale)}`,
   154→        ),
   155→        apiFetch<{ data: FormRecord[] }>(`${basePath}/records`),
   156→      ]);
   157→      const nextDef = defRes.data;
   158→      const nextRecords = recordsRes.data ?? [];
   159→      setDef(nextDef);
   160→      setRecords(nextRecords);
   161→
   162→      if (nextRecords.length) {
   163→        const lastIndex = nextRecords.length - 1;
   164→        const last = nextRecords[lastIndex];
   165→        setCurrentIndex(lastIndex);
   166→        setDraft({ ...last.data });
   167→        setBaseline({ ...last.data });
   168→        setMode('view');
   169→      } else {
   170→        const defaults = buildDefaults(nextDef.fields);
   171→        setCurrentIndex(-1);
   172→        setDraft({ ...defaults });
   173→        setBaseline({ ...defaults });
   174→        setMode('new');
   175→      }
   176→
   177→      const derived = deriveTabs(nextDef, locale);
   178→      // The first group is the always-visible header section; the active
   179→      // tab state only drives the remaining body tabs (default: first one).
   180→      const firstBodyTab = derived[1] ?? null;
   181→      setActiveTabKey(firstBodyTab ? firstBodyTab.key : null);
   182→    } catch (error) {
   183→      if (error instanceof SessionExpiredError) return;
   184→      setLoadError(
   185→        error instanceof Error && error.message
   186→          ? error.message
   187→          : GENERIC_LOAD_ERROR,
   188→      );
   189→    } finally {
   190→      setIsLoading(false);
   191→    }
   192→  }, [basePath, locale]);
   193→
   194→  useEffect(() => {
   195→    void load();
   196→  }, [load]);
   197→
   198→  const refreshRecords = useCallback(async (): Promise<FormRecord[]> => {
   199→    if (!basePath) return [];
   200→    const res = await apiFetch<{ data: FormRecord[] }>(`${basePath}/records`);
   201→    const list = res.data ?? [];
   202→    setRecords(list);
   203→    return list;
   204→  }, [basePath]);
   205→
   206→  // ── Record transitions ──
   207→  const applyRecord = useCallback((index: number, list: FormRecord[]) => {
   208→    const target = list[index];
   209→    setCurrentIndex(index);
   210→    setMode('view');
   211→    if (target) {
   212→      setDraft({ ...target.data });
   213→      setBaseline({ ...target.data });
   214→    }
   215→  }, []);
   216→
   217→  const navigateTo = useCallback(
   218→    (index: number) => {
   219→      if (isSaving || isDeleting) return;
   220→      if (index < 0 || index >= records.length) return;
   221→      if (mode !== 'view' && isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) {
   222→        return;
   223→      }
   224→      applyRecord(index, records);
   225→    },
   226→    [applyRecord, isDeleting, isDirty, isSaving, mode, records, t],
   227→  );
   228→
   229→  const startNew = useCallback(
   230→    (preset?: FormDataMap) => {
   231→      if (!def) return;
   232→      const base = { ...buildDefaults(def.fields), ...(preset ?? {}) };
   233→      setDraft(base);
   234→      setBaseline({ ...base });
   235→      setMode('new');
   236→      setCurrentIndex(-1);
   237→    },
   238→    [def],
   239→  );
   240→
   241→  const handleNew = useCallback(() => {
   242→    if (mode === 'new') return;
   243→    if (isDirty && !window.confirm(t('formWindow.navDiscardConfirm'))) return;
   244→    startNew();
   245→  }, [isDirty, mode, startNew, t]);
   246→
   247→  const handleCopy = useCallback(() => {
   248→    if (mode !== 'view' || !current) return;
   249→    startNew({ ...current.data });
   250→    toast({ title: t('formWindow.copyToast') });
   251→  }, [current, mode, startNew, t]);
   252→
   253→  /**
   254→   * Shared save pipeline for every save action: validates required
   255→   * fields, writes the draft to the database and refreshes the record
   256→   * list. `keepRecord: false` skips re-selecting the saved record
   257→   * (used by "save & new", which resets the form right after).
   258→   */
   259→  const performSave = useCallback(
   260→    async (options?: { keepRecord?: boolean }): Promise<SaveResult> => {
   261→      if (!def || !basePath || isSaving || isDeleting) return 'error';
   262→      if (mode === 'view' || !isDirty) return 'clean';
   263→
   264→      const missingField = def.fields.find((field) =>
   265→        isEmptyRequiredValue(field, draft[field.fieldKey]),
   266→      );
   267→      if (missingField) {
   268→        toast({
   269→          title: fieldLabel(missingField, locale),
   270→          description: t('formWindow.requiredHint'),
   271→          variant: 'destructive',
   272→        });
   273→        return 'invalid';
   274→      }
   275→
   276→      setIsSaving(true);
   277→      try {
   278→        const saved =
   279→          mode === 'new'
   280→            ? (
   281→                await apiFetch<{ data: FormRecord }>(`${basePath}/records`, {
   282→                  method: 'POST',
   283→                  body: { data: draft },
   284→                })
   285→              ).data
   286→            : (
   287→                await apiFetch<{ data: FormRecord }>(
   288→                  `${basePath}/records/${encodeURIComponent(current?.id ?? '')}`,
   289→                  { method: 'PUT', body: { data: draft } },
   290→                )
   291→              ).data;
   292→
   293→        const list = await refreshRecords();
   294→        if (options?.keepRecord !== false) {
   295→          const index = list.findIndex((record) => record.id === saved.id);
   296→          if (index >= 0) {
   297→            applyRecord(index, list);
   298→          } else {
   299→            setMode('view');
   300→          }
   301→        }
   302→        return 'saved';
   303→      } catch (error) {
   304→        if (!(error instanceof SessionExpiredError)) {
   305→          toast({
   306→            title: t('formWindow.saveError'),
   307→            description: error instanceof Error ? error.message : undefined,
   308→            variant: 'destructive',
   309→          });
   310→        }
   311→        return 'error';
   312→      } finally {
   313→        setIsSaving(false);
   314→      }
   315→    },
   316→    [
   317→      applyRecord,
   318→      basePath,
   319→      current,
   320→      def,
   321→      draft,
   322→      isDeleting,
   323→      isDirty,
   324→      isSaving,
   325→      locale,
   326→      mode,
   327→      refreshRecords,
   328→      t,
   329→    ],
   330→  );
   331→
   332→  const handleSave = useCallback(async () => {
   333→    const result = await performSave();
   334→    if (result === 'saved') toast({ title: t('formWindow.saveSuccess') });
   335→  }, [performSave, t]);
   336→
   337→  /** Save & new: write the record, then reset the form for a fresh entry. */
   338→  const handleSaveNew = useCallback(async () => {
   339→    if (isSaving || isDeleting) return;
   340→    const result = await performSave({ keepRecord: false });
   341→    if (result === 'error' || result === 'invalid') return;
   342→    startNew();
   343→    if (result === 'saved') toast({ title: t('formWindow.saveNewSuccess') });
   344→  }, [isDeleting, isSaving, performSave, startNew, t]);
   345→
   346→  /** Save & close: write the record after validation, then close. */
   347→  const handleSaveClose = useCallback(async () => {
   348→    if (isSaving || isDeleting) return;
   349→    const result = await performSave();
   350→    if (result === 'error' || result === 'invalid') return;
   351→    if (result === 'saved') toast({ title: t('formWindow.saveCloseSuccess') });
   352→    onClose?.();
   353→  }, [isDeleting, isSaving, onClose, performSave, t]);
   354→
   355→  /** Resets the shared delete-confirmation dialog. */
   356→  const cancelDelete = useCallback(() => {
   357→    setDeleteIntent(null);
   358→    setDeleteTargetId(null);
   359→  }, []);
   360→
   361→  /**
   362→   * First click on any delete action opens the confirm dialog. The
   363→   * target defaults to the loaded record; the tree context menu passes
   364→   * the right-clicked branch id explicitly.
   365→   */
   366→  const requestDelete = useCallback(
   367→    (intent: Exclude<DeleteIntent, null>, targetId?: string) => {
   368→      if (mode === 'new' || isSaving || isDeleting) return;
   369→      const target = targetId
   370→        ? records.find((record) => record.id === targetId) ?? null
   371→        : current;
   372→      if (!target) return;
   373→      setDeleteTargetId(target.id);
   374→      setDeleteIntent(intent);
   375→    },
   376→    [current, isDeleting, isSaving, mode, records],
   377→  );
   378→
   379→  /** Runs after the user presses «بلی» in the confirm dialog. */
   380→  const confirmDelete = useCallback(async () => {
   381→    if (isDeleting || !deleteIntent) return;
   382→    const intent = deleteIntent;
   383→    const targetId = deleteTargetId ?? current?.id ?? '';
   384→    if (!basePath || !targetId) {
   385→      cancelDelete();
   386→      return;
   387→    }
   388→    setIsDeleting(true);
   389→    try {
   390→      // DELETE answers 204 with an empty body — nothing to parse.
   391→      await apiFetch(
   392→        `${basePath}/records/${encodeURIComponent(targetId)}`,
   393→        {
   394→          method: 'DELETE',
   395→        },
   396→      );
   397→      cancelDelete();
   398→      if (intent === 'close') {
   399→        toast({ title: t('formWindow.deleteCloseSuccess') });
   400→        onClose?.();
   401→        return;
   402→      }
   403→      const list = await refreshRecords();
   404→      if (intent === 'new') {
   405→        startNew();
   406→        toast({ title: t('formWindow.deleteNewSuccess') });
   407→      } else if (list.length) {
   408→        applyRecord(Math.min(currentIndex, list.length - 1), list);
   409→        toast({ title: t('formWindow.deleteSuccess') });
   410→      } else {
   411→        setCurrentIndex(-1);
   412→        startNew();
   413→        toast({ title: t('formWindow.deleteSuccess') });
   414→      }
   415→    } catch (error) {
   416→      cancelDelete();
   417→      if (error instanceof SessionExpiredError) return;
   418→      toast({
   419→        title: error instanceof Error && error.message
   420→          ? error.message
   421→          : t('formWindow.loadError'),
   422→        variant: 'destructive',
   423→      });
   424→    } finally {
   425→      setIsDeleting(false);
   426→    }
   427→  }, [
   428→    applyRecord,
   429→    basePath,
   430→    cancelDelete,
   431→    currentIndex,
   432→    current,
   433→    deleteIntent,
   434→    deleteTargetId,
   435→    isDeleting,
   436→    onClose,
   437→    refreshRecords,
   438→    startNew,
   439→    t,
   440→  ]);
   441→
   442→  const handlePrint = useCallback(() => {
   443→    toast({ title: t('formWindow.printToast') });
   444→  }, [t]);
   445→
   446→  const handleFieldChange = useCallback(
   447→    (fieldKey: string, value: FormFieldValue) => {
   448→      setDraft((prev) => ({ ...prev, [fieldKey]: value }));
   449→      setMode((prev) => (prev === 'view' ? 'edit' : prev));
   450→    },
   451→    [],
   452→  );
   453→
   454→  // Ctrl+S / Cmd+S saves the active draft (Escape is handled by the dialog).
   455→  useEffect(() => {
   456→    const onKeyDown = (event: KeyboardEvent) => {
   457→      if (
   458→        (event.ctrlKey || event.metaKey) &&
   459→        !event.shiftKey &&
   460→        !event.altKey &&
   461→        event.key.toLowerCase() === 's'
   462→      ) {
   463→        event.preventDefault();
   464→        void handleSave();
   465→      }
   466→    };
   467→    window.addEventListener('keydown', onKeyDown);
   468→    return () => window.removeEventListener('keydown', onKeyDown);
   469→  }, [handleSave]);
   470→
   471→  // ── Derived view models ──
   472→  const sortedFields = useMemo(
   473→    () => (def ? [...def.fields].sort((a, b) => a.sortOrder - b.sortOrder) : []),
   474→    [def],
   475→  );
   476→
   477→  const tabs = useMemo(() => (def ? deriveTabs(def, locale) : []), [def, locale]);
   478→
   479→  /**
   480→   * Two-part layout of tabbed forms: the first group is the header
   481→   * section (its fields are always visible, without any tab strip) and
   482→   * the remaining groups build the tabbed body section below it.
   483→   */
   484→  const isTabbed = def?.type === 'tabbed' && tabs.length > 0;
   485→  const headerTab = isTabbed ? (tabs[0] ?? null) : null;
   486→  const bodyTabs = isTabbed ? tabs.slice(1) : [];
   487→  const activeBodyTab =
   488→    bodyTabs.find((tab) => (tab.key ?? null) === (activeTabKey ?? null)) ??
   489→    bodyTabs[0] ??
   490→    null;
   491→
   492→  const listColumns = useMemo(
   493→    () => pickListColumns(sortedFields),
   494→    [sortedFields],
   495→  );
   496→
   497→  const title = def
   498→    ? formTitle(def, locale)
   499→    : form?.name ?? t('common.loading');
   500→  const subtitle =
   501→    mode === 'new'
   502→      ? t('formWindow.newMode')
   503→      : current
   504→        ? tf('formWindow.recordOf', {
   505→            n: formatNumber(currentIndex + 1),
   506→            total: formatNumber(records.length),
   507→          })
   508→        : undefined;
   509→
   510→  const accentColor = def?.accentColor || DEFAULT_FORM_ACCENT;
   511→
   512→  /** The fields grid shared by simple forms and the active tab. */
   513→  const renderFields = (fields: typeof sortedFields): ReactNode => (
   514→    <div className={fieldsGridClasses}>
   515→      {fields.map((field) => (
   516→        <FieldRenderer
   517→          key={field.fieldKey}
   518→          field={field}
   519→          value={draft[field.fieldKey] ?? null}
   520→          onChange={(value) => handleFieldChange(field.fieldKey, value)}
   521→          disabled={isSaving || isDeleting}
   522→          lookupUrlBase={lookupUrlBase}
   523→        />
   524→      ))}
   525→    </div>
   526→  );
   527→
   528→  let content: ReactNode;
   529→  if (!def) {
   530→    // Loading skeleton (error handled below).
   531→    content = (
   532→      <div className={fieldsGridClasses}>
   533→        {Array.from({ length: 6 }).map((_, index) => (
   534→          <div key={`form-skeleton-${index}`} className="space-y-1.5">
   535→            <Skeleton className="h-3 w-24 bg-white/[0.08]" />
   536→            <Skeleton className="h-6 w-full rounded-md bg-white/[0.05]" />
   537→          </div>
   538→        ))}
   539→      </div>
   540→    );
   541→  } else if (def.type === 'tree') {
   542→    content = (
   543→      <TreeForm
   544→        records={records}
   545→        fields={sortedFields}
   546→        draft={draft}
   547→        mode={mode}
   548→        selectedId={current?.id ?? null}
   549→        isBusy={isSaving || isDeleting}
   550→        accentColor={accentColor}
   551→        lookupUrlBase={lookupUrlBase}
   552→        onSelect={(recordId) => {
   553→          const index = records.findIndex((record) => record.id === recordId);
   554→          if (index >= 0) navigateTo(index);
   555→        }}
   556→        onFieldChange={handleFieldChange}
   557→        onNewChild={(preset) => {
   558→          if (
   559→            mode !== 'view' &&
   560→            isDirty &&
   561→            !window.confirm(t('formWindow.navDiscardConfirm'))
   562→          ) {
   563→            return;
   564→          }
   565→          startNew(preset);
   566→        }}
   567→        onDeleteNode={(recordId) => requestDelete('current', recordId)}
   568→        onCopyNode={(record) => {
   569→          if (
   570→            mode !== 'view' &&
   571→            isDirty &&
   572→            !window.confirm(t('formWindow.navDiscardConfirm'))
   573→          ) {
   574→            return;
   575→          }
   576→          startNew({ ...record.data });
   577→          toast({ title: t('formWindow.copyToast') });
   578→        }}
   579→      />
   580→    );
   581→  } else if (isTabbed && headerTab) {
   582→    // Header section (main fields, no tab strip) + tabbed body section.
   583→    content = (
   584→      <div className="flex min-h-0 flex-col">
   585→        <section
   586→          aria-label={headerTab.name}
   587→          className={cn(
   588→            'shrink-0 bg-white/[0.02]',
   589→            bodyTabs.length > 0 && 'border-b border-white/10',
   590→          )}
   591→        >
   592→          {renderFields(headerTab.fields)}
   593→        </section>
   594→        {bodyTabs.length > 0 && (
   595→          <section aria-label={activeBodyTab?.name ?? undefined}>
   596→            <FormTabs
   597→              tabs={bodyTabs.map((tab) => ({
   598→                key: tab.key,
   599→                name: tab.name,
   600→                count: tab.fields.length,
   601→              }))}
   602→              active={activeBodyTab?.key ?? null}
   603→              onChange={setActiveTabKey}
   604→              accentColor={accentColor}
   605→              direction={direction}
   606→            >
   607→              {renderFields(activeBodyTab?.fields ?? [])}
   608→            </FormTabs>
   609→          </section>
   610→        )}
   611→      </div>
   612→    );
   613→  } else {
   614→    content = renderFields(sortedFields);
   615→  }
   616→
   617→  const errorText =
   618→    loadError === GENERIC_LOAD_ERROR ? t('formWindow.loadError') : loadError;
   619→
   620→  return (
   621→    <>
   622→    <FormShell
   623→      title={title}
   624→      subtitle={subtitle}
   625→      iconKey={def?.iconKey ?? form?.iconKey ?? 'fileText'}
   626→      accentColor={accentColor}
   627→      actionBar={
   628→        <FormActionBar
   629→          mode={mode}
   630→          isDirty={isDirty}
   631→          isSaving={isSaving}
   632→          isDeleting={isDeleting}
   633→          hasRecord={Boolean(current)}
   634→          recordIndex={currentIndex}
   635→          recordCount={records.length}
   636→          isListOpen={isListOpen}
   637→          onNew={handleNew}
   638→          onSave={() => void handleSave()}
   639→          onSaveNew={() => void handleSaveNew()}
   640→          onSaveClose={() => void handleSaveClose()}
   641→          onCopy={handleCopy}
   642→          onDelete={() => requestDelete('current')}
   643→          onDeleteNew={() => requestDelete('new')}
   644→          onDeleteClose={() => requestDelete('close')}
   645→          onPrint={handlePrint}
   646→          onToggleList={() => setIsListOpen((open) => !open)}
   647→          onFirst={() => navigateTo(0)}
   648→          onPrev={() => navigateTo(Math.max(0, currentIndex - 1))}
   649→          onNext={() => navigateTo(Math.min(records.length - 1, currentIndex + 1))}
   650→          onLast={() => navigateTo(records.length - 1)}
   651→          hideNav={def?.type === 'tree'}
   652→        />
   653→      }
   654→      subPanel={
   655→        <AnimatePresence initial={false}>
   656→          {isListOpen && (
   657→            <RecordsDrawer
   658→              key="records-drawer"
   659→              records={records}
   660→              currentIndex={currentIndex}
   661→              codeField={listColumns.codeField}
   662→              titleField={listColumns.titleField}
   663→              onJump={navigateTo}
   664→              onClose={() => setIsListOpen(false)}
   665→            />
   666→          )}
   667→        </AnimatePresence>
   668→      }
   669→      footer={
   670→        <FormFooter
   671→          status={status}
   672→          position={
   673→            current && mode !== 'new'
   674→              ? { n: currentIndex + 1, total: records.length }
   675→              : null
   676→          }
   677→          updatedAt={current?.updatedAt ?? null}
   678→        />
   679→      }
   680→    >
   681→      {errorText && !def ? (
   682→        <div className="flex min-h-[220px] flex-col items-center justify-center gap-3 p-6 text-center">
   683→          <p className="max-w-sm text-sm text-red-200/90">{errorText}</p>
   684→          <button
   685→            type="button"
   686→            onClick={() => void load()}
   687→            className="flex items-center gap-2 rounded-lg bg-white/[0.07] px-4 py-2 text-[13px] text-white/80 transition-colors hover:bg-white/[0.12] focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-white/40"
   688→          >
   689→            <RefreshCw className="h-3.5 w-3.5" />
   690→            {t('common.retry')}
   691→          </button>
   692→        </div>
   693→      ) : (
   694→        <>
   695→          {errorText && def && (
   696→            <div className="mx-4 mt-4 rounded-lg border border-red-400/30 bg-red-950/40 px-3 py-2 text-xs text-red-200/90">
   697→              {errorText}
   698→            </div>
   699→          )}
   700→          {records.length === 0 && def && def.type !== 'tree' ? (
   701→            <div className="px-4 pb-2 pt-6 text-center">
   702→              <p className="text-[13px] text-white/55">
   703→                {t('formWindow.emptyRecords')}
   704→              </p>
   705→              <p className="mt-1 text-[11px] text-white/35">
   706→                {t('formWindow.emptyRecordsHint')}
   707→              </p>
   708→            </div>
   709→          ) : null}
   710→          {content}
   711→        </>
   712→      )}
   713→    </FormShell>
   714→
   715→    {/* Single delete-confirmation dialog shared by all three delete actions. */}
   716→    <AlertDialog
   717→      open={deleteIntent !== null}
   718→      onOpenChange={(open) => {
   719→        if (!open) cancelDelete();
   720→      }}
   721→    >
   722→      <AlertDialogContent className="max-w-[calc(100vw-2rem)] rounded-2xl border border-white/15 bg-[#1e2128]/95 text-white shadow-2xl shadow-black/60 backdrop-blur-xl sm:max-w-md">
   723→        <AlertDialogHeader className="sm:text-start">
   724→          <AlertDialogTitle className="flex items-center gap-2.5 text-base text-white">
   725→            <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-rose-500/15 text-rose-400 ring-1 ring-rose-400/30">
   726→              <Trash2 className="h-4 w-4" aria-hidden="true" />
   727→            </span>
   728→            {t('formWindow.deleteDialogTitle')}
   729→          </AlertDialogTitle>
   730→          <AlertDialogDescription className="text-[13px] leading-relaxed text-white/65">
   731→            {t('formWindow.deleteDialogMessage')}
   732→          </AlertDialogDescription>
   733→        </AlertDialogHeader>
   734→        <AlertDialogFooter className="gap-2 sm:justify-start">
   735→          <AlertDialogCancel
   736→            disabled={isDeleting}
   737→            className="h-9 rounded-lg border-white/15 bg-white/[0.06] px-4 text-[13px] text-white/80 transition-colors hover:bg-white/[0.12] hover:text-white focus-visible:ring-white/40"
   738→          >
   739→            {t('formWindow.confirmNo')}
   740→          </AlertDialogCancel>
   741→          <AlertDialogAction
   742→            disabled={isDeleting}
   743→            onClick={(event) => {
   744→              // Keep the dialog open while the delete request runs.
   745→              event.preventDefault();
   746→              void confirmDelete();
   747→            }}
   748→            className="flex h-9 items-center gap-1.5 rounded-lg bg-rose-500/90 px-4 text-[13px] text-white shadow-sm shadow-rose-950/40 transition-colors hover:bg-rose-500 focus-visible:ring-rose-300/60"
   749→          >
   750→            {isDeleting ? (
   751→              <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden="true" />
   752→            ) : null}
   753→            {t('formWindow.confirmYes')}
   754→          </AlertDialogAction>
   755→        </AlertDialogFooter>
   756→      </AlertDialogContent>
   757→    </AlertDialog>
   758→    </>
   759→  );
   760→}
